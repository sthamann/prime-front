\begin{abstract}\noindent
Round 378 attacks the named remainder COMPOSE of the terminal
lane: the slope chain
$M_3\to N_3\to N_2\to$ Fej\'er energy $\to$ van der Corput
$\to$ H5 $\to q_N<1$ is, at the Fej\'er/vdC$\to$H5$\to q_N$
joints, a measured convention (r297 $\sigma\le-0.516$; r300
$N_2$-need $0.908$; r324 atom-need $0.888$), not a pointwise
theorem.  What is proved: every exact link as a finite inequality
with explicit constants; the van der Corput bound at
$H=\max(2,\lceil\sqrt{m}\rceil)$ as a constant-free theorem;
the participation and R\'enyi identities; the kernel envelope;
the two-branch reduction; and the log-complete $m_0$-solve
(the r361 body dropped $2\log\log m$).  What is \emph{not}
proved: a uniform bound on the $\ell^1$ mass $L_1$ of the
block field, a uniform Fej\'er/diagonal ratio $R_0$, a uniform
edge scalar $|Z_{\loc}|$, and a family-uniform T1 constant
$C_K$.  The hydra-legal reduction is the pointwise implication
\[
  M_3\le\phi(m)\ \wedge\ L_1\le\Lambda\ \wedge\ S_F\le R_0\,D
  \ \wedge\ |Z_{\loc}|\le Z_0<M
  \implies q_N<1
\]
for $m\ge m_0$, with $\phi$, $\Lambda$, $R_0$, $Z_0$, $M$ and
$m_0$ named below.  Hypothesis H5 (the pair-margin law) is a
\emph{sister} sufficient condition, not an extra hypothesis of
the vdC route.  No RH claim.
\end{abstract}

\section{The object}\label{sec:obj}

Fix a canonical window of constructive level-2 block count
$m\ge 2$.  Write $M=\sqrt{5/7}$ for the r243 floor, $Z$ for the
terminal drive readout, $Z_{\loc}$ for the edge-local piece of
the r269 $F=0.20$ split, and $P=(P_j)_{j<m}$ for a real block
sequence (either the r287 level-2 border blocks, or the r298
difference field $P_\Delta$).  The Fej\'er energy at window
$H$ is
\[
  S_F
  =\frac1H\sum_t W_t^2
  =\sum_{|h|<H}\Bigl(1-\frac{|h|}{H}\Bigr)A(h),
\]
with $W_t$ the sliding $H$-window sums and $A$ the
autocorrelation of $P$.  The frozen rule is
$H=\max\bigl(2,\lceil\sqrt{m}\rceil\bigr)$.  Write
$\pref=(m+H-1)/H$, $D=\sum_j P_j^2$, $L_1=\sum_j|P_j|$,
$N_2=L_1^2/D$ (participation), and $M_3=\sum_j q_j^3$ with
$q_j=|P_j|/L_1$ (the r324 convention; equivalently
$S_3$ of r306).  The R\'enyi numbers are $N_3=M_3^{-1/2}$ and
$N_2=1/\sum q_j^2$.  The terminal cross-ratio is
$q_N=\rho_{N-1}/(5/7)$.  The r263 dictionary, measured on the
$42$-rung census and typed in Lean as
\texttt{pair\_terminal\_dictionary} (a \texttt{sorry},
transcription-blocked), reads $Z^2=(5/7)\,q_N$.  Hypothesis H5
is the named Prop \texttt{PairMarginLaw}:
$|Z_{\loc}|+\mathrm{pairBound}<M$.

Claim boundary: finite identities and a named reduction.  Not
evidence for or against the Riemann hypothesis.

\section{Each link, typed}\label{sec:links}

\subsection*{Exact finite theorems}

\begin{lemma}[Fej\'er window-sum and van der Corput]\label{lem:vdc}
For every real sequence $P$ of length $m$ and every
$H\in\{1,\ldots,m\}$,
\[
  \Bigl|\sum_j P_j\Bigr|^2
  \le \frac{m+H-1}{H}\,S_F,
\]
with $S_F=(1/H)\sum_t W_t^2\ge 0$.  Equality at $H=1$.  The
prefactor $m/H$ is false (witness $P=(1,1,1,1)$, $H=2$:
$14<16$).
\end{lemma}

\noindent
This is v964-S0 (covering identity $\sum_t W_t=H\sum_j P_j$ plus
Cauchy--Schwarz over $m+H-1$ positions), re-proved in Fractions
in the companion script.

\begin{lemma}[The $H$-rule prefactor]\label{lem:pref}
At $H=\max(2,\lceil\sqrt{m}\rceil)$ one has
$\pref\le\sqrt{m}+1$ for every $m\ge 1$.
\end{lemma}

\begin{lemma}[Canonical split]\label{lem:split}
$|Z|\le|Z_{\loc}|+|\sum P|$ whenever $\sum P$ is the bulk
remainder of the canonical extraction.  This is
\texttt{RH.PairBound.canonical\_split} (no \texttt{sorry}).
\end{lemma}

\begin{lemma}[Participation and R\'enyi]\label{lem:renyi}
$D\cdot N_2=L_1^2$ identically.  For a probability vector $q$,
the Lagrange identities
$2(S_2^2-S_3)=\sum_{i,j}q_i q_j(q_i-q_j)^2\ge 0$ and
$2(S_3^2-S_2 S_4)=\sum_{i,j}q_i^2 q_j^2(q_i-q_j)^2\ge 0$
give $N_2\ge N_3\ge N_4$, with equality exactly on the uniform
vector.  The r324 interpolation $M_3\le q_{\max} M_2$ and
$M_2\le q_{\max}$ is the same algebra.
\end{lemma}

\begin{lemma}[Kernel envelope]\label{lem:env}
The Fej\'er kernel
\[
  F_H(\theta)
  =\Bigl|\sum_{j<H}e^{-ij\theta}\Bigr|^2/H
\]
satisfies
$F_H(\theta)\le\min\bigl(H,\,1/(H\sin^2(\theta/2))\bigr)$.
Consequently $S_F\le R_{\mathrm{env}} D$ with $R_{\mathrm{env}}$
the $F_H$-weighted mean of $|\widehat P|^2$.  The trivial bound
$R_{\mathrm{env}}\le H$ is a theorem and is too coarse for
composition (it demands $N_2\gg m$).
\end{lemma}

\begin{lemma}[T1-floor algebra]\label{lem:t1alg}
If $g_i\ge 3/8$ and $F_i=(m q_i/\log m)\,g_i\le C_K$ for every
block, then $q_i\le(8/3)C_K(\log m)/m$ and
\[
  M_3
  \le\Bigl(\frac83 C_K\frac{\log m}{m}\Bigr)^2
  =\frac{64}{9}\,C_K^2\,\frac{(\log m)^2}{m^2}.
\]
\end{lemma}

\noindent
The constant $8/3$ is the r361 MED-CAP cap (saturated at
$\chi_4$ kz$53$).  The lemma is a theorem of the floor
\emph{and} a $C_K$-cap; both remain named remainders of other
rounds ($V_3'$ and T1).

\section{The pointwise composition}\label{sec:compose}

Combine Lemmas~\ref{lem:vdc}--\ref{lem:split}:
\[
  |Z|
  \le |Z_{\loc}|+\sqrt{\pref\cdot S_F}.
\]
If the right-hand side is strictly less than $M$, then $|Z|<M$.
Under the r263 dictionary this is $q_N<1$.  No slope enters.

Write $S_F\le R_0 D=R_0 L_1^2/N_2$ (Lemma~\ref{lem:env} supplies
the shape; $R_0$ is the remaining bound) and $N_2\ge N_3=M_3^{-1/2}$
(Lemma~\ref{lem:renyi}).  Then
\[
  |Z_{\loc}|+\sqrt{\pref\,R_0}\,L_1\,M_3^{1/4}<M
\]
is sufficient.  Rearranging with $|Z_{\loc}|\le Z_0<M$ and
$L_1\le\Lambda$ yields the $M_3$-cap
\begin{equation}\label{eq:phi}
  M_3
  \le\phi(m)
  :=
  \frac{(M-Z_0)^4}{\pref(m)^2\,R_0^2\,\Lambda^4},
  \qquad
  \pref(m)\le\sqrt{m}+1.
\end{equation}
Asymptotically $\phi(m)\sim c_\phi\, m^{-1}$ with
$c_\phi=(M-Z_0)^4/(R_0^2\Lambda^4)$, i.e.\ the H5-sufficient
exponent is $\beta=1/2$ in the form
$M_3\le c_1\,(\log m)^{-a}\,m^{-2\beta}$
($a=0$, $c_1=c_\phi$, $2\beta=1$), \emph{provided} $\Lambda$ and
$R_0$ are $O(1)$.

If $L_1\le C\,m^\gamma$, the same algebra gives $2\beta=1+4\gamma$,
so $\beta=1/2+2\gamma$.  The measured slope $\gamma\approx 0.196$
produces $\beta\approx 0.892$, which is the origin of the
$0.888$/$0.908$ convention: those numbers are slope-proxies for
$1/2+2\gamma$, not independent targets.  T1-floor
(Lemma~\ref{lem:t1alg}) supplies $\beta=1$, which beats
$1/2+2\gamma$ for every $\gamma\le 1/4$.

\begin{prop}[Compose, reduced]\label{prop:compose}
Let $M=\sqrt{5/7}$, $H=\max(2,\lceil\sqrt{m}\rceil)$,
$\pref=(m+H-1)/H$.  On a canonical window of block count $m$,
assume
\begin{enumerate}
\item[\rm(R)] $S_F\le R_0\,D$ for a frozen $R_0<\infty$,
\item[\rm(L)] $L_1\le\Lambda$ for a frozen $\Lambda<\infty$,
\item[\rm(Z)] $|Z_{\loc}|\le Z_0<M$,
\item[\rm(M3)] $M_3\le\phi(m)$ with $\phi$ as in~\eqref{eq:phi}.
\end{enumerate}
Then $|Z|<M$.  If the r263 dictionary holds, $q_N<1$.
If in place of~\rm(L) one has $L_1\le C m^\gamma$ with
$\gamma<1/4$ and in place of~\rm(M3) the T1-floor bound of
Lemma~\ref{lem:t1alg} with a frozen $C_K$, the same conclusion
holds for all $m\ge m_0$ of Proposition~\ref{prop:m0}.
\end{prop}

Hypothesis H5 is not an extra premise: the pair-margin law is a
sister sufficient condition ($|Z_{\loc}|+\mathrm{pairBound}<M$).
The vdC route replaces \texttt{pairBound} by
$\sqrt{\pref\,S_F}$.  On the $42$-rung head the two-branch
certificate already closes $q_N<1$ without compose (cheap
triangle $35/42$; six F2/phase certificates; kz$15$
exact-finite).  Compose is the cofinal statement.

\section{Log accounting and $m_0$}\label{sec:m0}

The r361 direct chain used $\mathrm{CRIT}=0.224=2(1-0.888)$ as
the budget for $N_2\ge m^{0.888}$:
$(8/3)^2 C_K^2(\log m)^2\le m^{0.224}$, and solved
$0.224\log m\ge 2\log((8/3)C_K)$, dropping $2\log\log m$.
The published thresholds $10^{10.0}$ ($C_K=4.91$) and
$10^{16.1}$ ($C_K=23.70$) are therefore the log-free
approximants.  The complete inequality is
\begin{equation}\label{eq:m0}
  0.224\,\log m
  \ge
  2\log\Bigl(\tfrac83 C_K\Bigr)+2\log\log m.
\end{equation}

\begin{prop}[Log-complete \texorpdfstring{$m_0$}{m0}]\label{prop:m0}
The unique positive solution of~\eqref{eq:m0} (equivalently
of $0.224\,t=2\log((8/3)C_K)+2\log t$, $t=\log m$) is
\[
  \log_{10} m_0
  \approx
  \begin{cases}
    25.81 & \text{at the rule-conditional freeze }C_K=4.91,\\
    32.85 & \text{at the measured census ceiling }C_K=23.70
  \end{cases}
\]
(companion script: $25.81$ / $32.85$ by bisection; the Sol
diagnosis of DCCXXXVIII).
The r361 figures $10^{10.0}$ and $10^{16.1}$ are the
log-free undershoots (complete-inequality ratios $0.4517$ and
$0.5351$, both $<1$: those $m$ do \emph{not} meet~\eqref{eq:m0};
Sol's $0.0437$/$0.0271$ compared a different left-hand side).  The
asymptotic claim ``every polylog $C_K$ suffices against a
fixed power $m^{0.888}$'' remains correct; the finite head
threshold was optimistic.
\end{prop}

\noindent
Against the \emph{pointwise} H5-need $\beta=1/2$ (bounded
$L_1,R_0,Z_0$), T1-floor with any frozen $C_K$ is far stronger
than necessary, and $m_0$ is modest: the inequality
$(64/9)C_K^2(\log m)^2/m^2\le c_\phi/m$ reduces to
$m/(\log m)^2\ge K$ with $K=(64/9)C_K^2/c_\phi$.  The large
published $m_0$ were an artefact of composing T1 against the
slope-proxy $m^{0.888}$ rather than against~\eqref{eq:phi}.

\section{The remaining lemma set}\label{sec:rest}

The compose implication of Proposition~\ref{prop:compose} is a
theorem of finite algebra.  The original COMPOSE task is
therefore \emph{reduced} to the following named remainders,
each strictly smaller than a slope law.

pointwise H5-need with bounded $L_1$ is $\beta=1/2$.  None of
the three is a theorem.  Compose replaces all three by
\eqref{eq:phi}.
