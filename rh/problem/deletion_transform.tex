\begin{abstract}\noindent
Round 390 identified occupied Fej\'er as node deletion from a
discrete Bernstein--Szeg\H{o} measure (full cosine grid,
$\gamma_k=\tfrac14$).
Round 389 placed the remaining cancellation load in the $\mu$-OP
Chebyshev Gram.
This note derives the closed Uvarov formula.
What is proved: if $\sigma$ is finite atomic and $\Xi$ a subset of
its support with masses $W$, writing
$\tau_n=\det(I-K_n[\Xi]W)$ for the Christoffel--Darboux Gram of
$\sigma$ at the deleted nodes ($\tau_0=1$), the occupied monic
coefficients are
\[
\gamma_n(\mu)
=
\gamma_n(\sigma)
\cdot
\frac{\tau_{n+1}\,\tau_{n-1}}{\tau_n^2}.
\]
One-point and block forms agree over $\Q$; iterated
Sherman--Morrison is the stable evaluation.
On FRAME-A $w=9$ the identity holds at $9\cdot10^{-15}$ against
occupied Fej\'er.
Bernstein--Szeg\H{o} specialises to
$\gamma_n(\mu)=\tfrac14\cdot\tau_{n+1}\tau_{n-1}/\tau_n^2$
up to the r390 residual $1.85\cdot10^{-6}$.
Assist $0.0399$ and $\mathrm{d}\Delta_{12}=0.04946$ are exact
Uvarov readouts of $K^\mu$.
What is \emph{not} proved: $F_\eps$ from $\rho_{\mathrm{AP}}<1/5$.
Two-period AP deletion ($\rho_{\mathrm{AP}}=1$) stays at
$\gamma=\tfrac14$; a consecutive $\nu$-run of length $3$ leaves
the box; the kz~$55$ jump $0.3942$ against $\tfrac25$ is census.
Cancellation of Assist lives in the \emph{signs} of $K^\mu[\Xi]$,
not in the positive $\tau$-quotients of $\gamma$.
No RH claim.
\end{abstract}

\section{Recollection}\label{sec:rec}

Occupied Fej\'er of~\cite{r390} is the restriction of the full
cosine-grid measure
$\sigma=\sum_k(1-x_k)\,\delta_{x_k}$,
$x_k=\cos(2\pi k/L)$, $L=4h-2$, to the $\mu$-nodes.
The complement $\Xi$ is the $\nu$-support.
The full grid is discrete Bernstein--Szeg\H{o}:
$\max_{k\ge 2}\lvert\gamma_k(\sigma)-\tfrac14\rvert=1.84\cdot10^{-6}$
on FRAME-A $w=9$.
$G_\varepsilon^\mu$ reduced to the pair $(F_\eps,W_\eps)$:
Fej\'er-on-occupied, and the r342 digamma/tent multiplier.
This note attacks $F_\eps$ as exact transformation theory.

\section{The Uvarov ratio}\label{sec:uvarov}

Let $\sigma=\sum_i w_i\delta_{x_i}$ be finite, positive, and atomic,
with monic orthogonal polynomials $P_n$, Hankel determinants
$\Delta_{n-1}=\det(m_{i+j})_{i,j=0}^{n-1}$, and
$\gamma_n=\Delta_n\Delta_{n-2}/\Delta_{n-1}^2$.
Let $K_n(x,y)=\sum_{j<n}\pi_j(x)\pi_j(y)$ be the orthonormal
Christoffel--Darboux kernel.
A Uvarov perturbation $\tilde\sigma=\sigma+\sum_{a\in\Xi}M_a\delta_{\xi_a}$
updates the Hankel by a rank-$m$ Gram:
$\tilde H_n=H_n+VMV^\top$.

\begin{lemma}[Hankel determinant]\label{lem:hankel}
Writing $\tau_n=\det(I+K_n[\Xi]M)$ with $\tau_0=1$,
\[
\frac{\tilde\Delta_{n-1}}{\Delta_{n-1}}=\tau_n.
\]
Consequently
\[
\tilde\gamma_n
=
\gamma_n\cdot\frac{\tau_{n+1}\,\tau_{n-1}}{\tau_n^2},
\qquad n\ge 1.
\]
Exact over $\Q$ on a five-atom toy (delete last:
$\tilde\gamma=\tfrac5{48},\tfrac8{75},\tfrac5{81}$;
two-point indices $(1,3)$:
$\tilde\gamma=\tfrac8{45},\tfrac4{15}$).
The block determinant and the iterated one-point form agree.
\end{lemma}

\begin{proof}
The matrix determinant lemma gives
$\det(\tilde H_n)=\det(H_n)\det(I+MV^\top H_n^{-1}V)$.
The monomial reproducing kernel is
$K_n(\xi_a,\xi_b)=(V^\top H_n^{-1}V)_{ab}$, so the second factor is
$\tau_n$.
The empty sum $K_0=0$ forces $\tau_0=1$.
The $\gamma$-identity is the three-term ratio of Hankels.
Iterating the $m=1$ case multiplies the scalar $\tau$-ratios;
the block determinant is the same product (sequential Schur).
\end{proof}

Deletion of existing masses is the case $M=-W$.
Then $\tau_n=\det(I-K_n[\Xi]W)$.
The numerically stable evaluation on MAIN is iterated
Sherman--Morrison: after deleting $\xi$ with mass $w$,
\[
\tau_n=1-w\,K_n(\xi,\xi),
\qquad
\tilde K_n(x,y)=K_n(x,y)+\frac{w}{\tau_n}K_n(x,\xi)K_n(\xi,y),
\]
then proceed to the next node.
On a clustered block of deletions the floating-point SM can
lose digits; the Fractions block form remains exact.

\begin{lemma}[Bernstein--Szeg\H{o} specialisation]\label{lem:bs}
If $\gamma_n(\sigma)=\tfrac14$ for $n\ge 1$, then
\[
\gamma_n(\mu)=\frac14\cdot\frac{\tau_{n+1}\,\tau_{n-1}}{\tau_n^2}.
\]
On the cosine-plus-Fej\'er grid this holds up to the r390 residual
(FRAME-A $w=9$: $1.85\cdot10^{-6}$).
The exact identity uses $\gamma_n(\sigma)$, not $\tfrac14$.
\end{lemma}

\begin{lemma}[FRAME-A identity]\label{lem:w9}
On FRAME-A $w=9$ ($S=367$, $n_\nu=104$, depth $182$), iterated
Uvarov of full-grid Fej\'er deleting the $\nu$-nodes reproduces
occupied Fej\'er at $\max_k\lvert\gamma_k^{\mathrm{Uvarov}}-\gamma_k^{\mathrm{occ}}\rvert=9.0\cdot10^{-15}$.
Last twelve: $\lvert\gamma-\tfrac14\rvert=0.03596$, jump $0.2379$,
inside~\cite{v3p}.
\end{lemma}

\section{What the formula says about \texorpdfstring{$F_\eps$}{Feps}}\label{sec:feps}

Write $r_n=\tau_n/\tau_{n-1}$.  Then
$\gamma_n(\mu)=\gamma_n(\sigma)\,r_{n+1}/r_n$, and
\[
\log\frac{\gamma_{n+1}}{\gamma_n}
=
\Delta^2\log\tau
+\log\frac{\gamma_{n+1}(\sigma)}{\gamma_n(\sigma)}.
\]
The box $\lvert\gamma-\tfrac14\rvert\le\tfrac1{16}$ is
$\lvert e^{\Delta\log\tau}-1\rvert\le\tfrac14$ in the
Bernstein--Szeg\H{o} specialisation;
the jump is a second difference of $\log\tau$.

\begin{lemma}[jump not implied]\label{lem:jump}
$\log\tfrac53>\tfrac25$.  The jump half of $G_\varepsilon$ is not
a corollary of the $\lvert\gamma-\tfrac14\rvert$ box
(r390, restated).
\end{lemma}

Two-period AP deletion --- every other cosine node, hence
$\rho_{\mathrm{AP}}=1$ --- remains Bernstein--Szeg\H{o}:
on $h=40$, last twelve $\lvert\gamma-\tfrac14\rvert=0$ and jump $0$.
The hoped resonance kill of $F_\eps$ does \emph{not} fire.
A consecutive deletion run of length $3$ on the same grid
leaves the box (jump $0.462$).
Run length $2$ stays in.
Scramble occupation seed $3$ on FRAME-A leaves by jump $0.444$.

$V_3'$ of~\cite{v3p} fails first at a coherent last-twelve
$\gamma$-scale of $2.0$; scale $1.5$ stays in $\mathcal{A}_{15}$.
The $\lvert\gamma-\tfrac14\rvert\le\tfrac1{16}$ half is the WKB
input.
The constant $\tfrac25$ is extra regularity, not forced by
$\mathcal{A}_{15}$.
Raising it toward $\log\tfrac53\approx 0.511$ would make the jump
redundant with the box and is legal for $V_3'$.
It is not moved in this round: the kz~$55$ margin $0.0058$ is
census, not a theorem of the $\tau$-quotient.

\section{Assist and \texorpdfstring{$\mathrm{d}\Delta$}{dDelta} as Uvarov readouts}\label{sec:assist}

The IIKS Gram is $E_{ij}=\sqrt{v_i v_j}\,K_n^\mu(y_i,y_j)$.
Uvarov of $\mu+\nu$ deleting the $\nu$-masses \emph{is} $\mu$, so
$K^\mu$ is the updated kernel of $K^{\mu\cup\nu}$.
On FRAME-A $w=9$, $k=183$:
$\lambda_{\max}=0.999830$, $\mathrm{maxdiag}=0.9614$,
$\mathrm{assist}=0.0399$, Gershgorin assist $13.32$,
cancellation $0.9970$.
The kernel matches the $B$-matrix route at $1.6\cdot10^{-7}$.

Occupied-Fej\'er $K^{\mu_F}$ dressed with the actual $\nu$-weights
gives $\lambda_{\max}=10.12$: node deletion from Bernstein--Szeg\H{o}
does \emph{not} carry Assist.
The $99.7\%$ cancellation is the sign pattern of $K^\mu[\Xi]$, not
the positive $\tau$-quotients that control $\gamma$.

The r388 split $Q=Q^T+\Delta$ is the same readout:
$Q_k=\sum_i v_i\pi_k^\mu(y_i)^2$ is the diagonal of the kernel
increment $K_{k+1}^\mu-K_k^\mu$.
FRAME-A last twelve $\lvert\mathrm{d}\Delta\rvert=0.04946$,
$\lVert\Delta\rVert_\infty=0.3347$, reproducing~\cite{r388}.

\section{Census and adversaries}\label{sec:census}

\end{prop}

Dropping $\tau_{n-1}$ from the ratio fails ($0.122$).
Clustered run of $3$ leaves the box.
Scramble occupation seed $3$ leaves by jump.
AP deletion does \emph{not} leave --- it is a named non-kill of
$F_\eps$, and a kill of Assist/$L^*$ only (r387).

\section{What remains}\label{sec:rest}

\begin{prop}[reduction]\label{prop:red}
The deletion algebra is SATZ.
$F_\eps$ reduces to a bound on $\Delta^2\log\tau$ for
$F_1$-bounded $\nu$-runs (max run $\le 2$, r382, MAIN $85/85$)
together with half-filling; $\rho_{\mathrm{AP}}<1/5$ does not
imply the box (AP deletion is the regular case $\gamma=\tfrac14$).
The jump half at $0.3942$ is census.
Assist and $\mathrm{d}\Delta$ reduce to the \emph{Sign-Schur}
lemma: a sign bound on the off-diagonal of the Uvarov image
$K^\mu[\Xi]$, equivalently the signed $\mu$-CD remainder of~\cite{r387}
now with an exact kernel formula.
$W_\eps$ (the r342 multiplier) is untouched and remains the
weight sibling of $F_\eps$.
The chain
\[
G_\varepsilon^\mu
\;\Longleftarrow\;
F_\eps\wedge W_\eps
\;\Longleftarrow\;
\bigl(\Delta^2\log\tau\text{ under }F_1\bigr)
\wedge W_\eps
\]
stands on the positive side;
Assist stays on the signed side as Sign-Schur.
No $m_4$.  No RH claim.
\end{prop}
