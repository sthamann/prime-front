\begin{abstract}\noindent
Round 417 found $|\tau_{\mathrm{un}}|\to 0$ (slope $-0.47$)
with $|\sch|$ flat, so $\sch\to\phibb$ and $\phibb$ would be
the asymptotic sign carrier.
Round 418 found six vacuous core windows with $\phibb>0$
that nevertheless live, via
$\sch=\phibb-(\tilde a^{2}+\tilde b^{2})$.
Those two cannot both be cofinal.
Three hypotheses are tested.
$H3$ (the vacuous family empties) is \textbf{refuted}.
A razor-pole account of the overflow
($(C-I)^{-1}$ explosion at $C_{\min}-1\sim 10^{-7}$)
is \textbf{refuted}.
$H2$ saves the six \emph{now} ($\tau^{2}>\phibb$) but cannot
be cofinal (vacuous $\tau_{\mathrm{un}}$ is the one that
vanishes).  $H1$ is a large-$N$ census, not a law on the six.
No RH claim.
\end{abstract}

\section{The tension}\label{sec:ten}

On the vacuous chart $J_{2}=I_{2}$,
\begin{equation}\label{eq:vac}
\sch
\;=\;
\phibb-\tau^{2},
\qquad
\tau^{2}=\tilde a^{2}+\tilde b^{2},
\qquad
\phibb=c_{J}+\Sigma,
\end{equation}
with $c_{J}=\mathrm{den}-2$ and
$\Sigma=2\,s^{T}(C+I)(C-I)^{-1}s$ for $C\succ I$.
Living overflow: $\phibb>0$ and $\tau^{2}>\phibb$.
Dead vacuous: $\phibb>0$ and $\tau^{2}<\phibb$.

\begin{lemma}[vacuous overflow]\label{lem:vac}
On the vacuous branch, $\sch<0$ cofinally --- either via
$\phibb<0$ cofinally ($H1$/$H3$) or via a source-pure
lower bound on the couplings ($H2$: $\tau^{2}>\phibb$
wherever $\phibb>0$).
\end{lemma}

\noindent
\textbf{Verdict: REDUZIERT.}
$H3$ is false.
$H2$ is true at finite $N$ for the named six and false
as a cofinal floor.
$H1$ does not hold as a trend on the six; it holds as a
census on EXT/selected vacuous windows.
No $k_{0}$.

\section{Resolution of $H1$, $H2$, $H3$}\label{sec:H}

\begin{prop}[three hypotheses]\label{prop:H}
On the named census:
\begin{itemize}
\item $H1$ (the six trend through zero).
The six stay $\phibb>0$ on $N\in[151,540]$;
$\mathrm{d}\phibb/\mathrm{d}\log N=-0.003$ (flat).
$k_{z}=49$ at $N=540$ is still $+0.0197$.
Healthy vacuous windows already at $N=344$ are negative:
the $N$-ranges overlap.  Not a small-$N$ artefact.
EXT vacuous ($k_{z}=70,42,72$) and selected vacuous
($k_{z}=26,43$) have $\phibb<0$ --- a large-$N$ census,
not a law on the six.
\item $H2$ (vacuous $\tau$ does not vanish; r417 was
P1-dominated).
The opposite: vacuous $|\tau|$ slope $-1.24$,
$|\tau_{\mathrm{un}}|$ slope $-0.99$;
P1 $|\tau|$ slope $+0.05$, $|\tau_{\mathrm{un}}|$ slope
$+0.03$ (flat).
The r417 fall is \emph{vacuous-driven}.
The six live now by $\tau^{2}>\phibb$
(margins $0.0015$--$0.046$).
A cofinal lower bound on $\tau$ fails because
vacuous $\tau_{\mathrm{un}}$ is the vanishing one.
\item $H3$ (vacuous windows migrate to P1).
Core vacuous $14/42$.  Terciles by $N$:
$3/14$, $4/14$, $7/14$ --- the fraction \emph{grows}.
EXT $3/6$ vacuous.  Selected mixed.
\textbf{Refuted.}
\end{itemize}
\end{prop}

\section{What distinguishes the six}\label{sec:char}

The overflow set is
$k_{z}\in\{12,13,16,19,39,49\}$.
$k_{z}=12$ is the r377 LATE prototype.

\begin{prop}[not a nearer pole]\label{prop:razor}
Every vacuous core window sits at
$C_{\min}-1\in[1.3\cdot 10^{-7},\,7.5\cdot 10^{-5}]$.
Healthy windows are \emph{closer} to the razor than
overflow windows.
The overlap of the Sherman--Morrison border $s$ with the
$C_{\min}$-mode is $\sim 10^{-10}$; the razor share of
$\Sigma$ is at most $0.022$ ($k_{z}=39$), else $\sim 0$.
Overflow is not an explosion of $(C-I)^{-1}$.
The r377 product $h_{N}h_{N+1}=+1$ on every vacuous
window (overflow and healthy) --- not a separator.
$c_{J}\in[-0.520,-0.348]$ and $\Sigma\in[0.369,0.650]$
on the six; $\phibb=\Sigma+c_{J}$ is a small positive
$O(0.01)$ (except $k_{z}=39$, $\phibb=+0.129$).
\end{prop}

The six are distinguished by $\Sigma>|c_{J}|$ together
with a \emph{larger} $|\tau_{\mathrm{un}}|$
($0.068$--$0.399$ against healthy $0.020$--$0.102$),
not by $C_{\min}$, not by r377, not by a razor pole.

\section{Life versus death}\label{sec:life}

\begin{prop}[the separator]\label{prop:life}
Living overflow: $\tau^{2}>\phibb$ (all six).
Dead vacuous $\chi$ ($\chi_{3}$-$23$, $\chi_{3}$-$33$,
$\chi_{4}$-$20$): $\phibb>0$ and $\tau^{2}<\phibb$,
so $\sch>0$.
That is the life/death mechanism of the overflow class.
Dead vacuous $C_{\min}-1$ is larger
($10^{-4}$--$3\cdot 10^{-3}$) --- further from the razor.
Dropping $\tau$ on a living overflow sends $\sch$ to
$\phibb>0$ (the mutant dies).
Over $\mathbb{Q}$,
$\phibb=\tfrac1{10}$, $\tau^{2}=\tfrac14$ gives
$\sch=-\tfrac3{20}$; the drop-$\tau$ mutant is
$+\tfrac1{10}$.
\end{prop}

\section{Kills}\label{sec:kills}

Drop-border: no $\sch$ scalar ($\indm K_{2}=\indm A_{0}$).
Scramble $\mathrm{nneg}(A_{0})=21$; two-period $S=21$
has $\mathrm{nneg}=4$.
The drop-$\tau$ mutant kills every living overflow.
A false $c_{J}$ is inherited from r418 (kills the
$\phibb$ split, not the $\tau$-save).

\section{What is a theorem, what is a census}\label{sec:honest}

Equation~\eqref{eq:vac} is a theorem (r417 chart).
Proposition~\ref{prop:life} is a finite identity plus a
named separator.
Lemma~\ref{lem:vac} is not proved:
$H3$ is refuted, a cofinal $\tau$-floor is unavailable,
and the six themselves do not cross $\phibb=0$.
Cofinal vacuous $\sch<0$ reduces to $\phibb<0$ on the
large-$N$ vacuous windows that the EXT/selected census
exhibits --- a named open, no $k_{0}$.
The six are a named finite class, not a cofinal
phenomenon.
No RH claim.
