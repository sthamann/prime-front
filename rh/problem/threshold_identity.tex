\begin{abstract}\noindent
Round 409 printed $\|\Tfrak_0\|=1.00000$ on the
positive-definite windows $k_z=42,130$.
That looked like a combinatorial identity.
It is not.
The exact lemma is the dictionary
\[
  \Tfrak_0^*\Tfrak_0 \;=\; C^{-1},
  \qquad
  \|\Tfrak_0\| \;=\; 1/\sqrt{\lambda_{\min}(C)},
\]
and the energy split on the quotient
$\mathcal{P}_{<d_0}/\mathcal{K}_0$:
$\|\Tfrak_0\|\le 1$ if and only if every min-norm interpolant
carries at least half its dual-weighted energy on $Y$.
On PD windows, $C_{\min}=1+\varepsilon$ with
$\varepsilon\in[10^{-8},10^{-4}]$, so $\|\Tfrak_0\|$ sits
$O(\varepsilon)$ below $1$.
No saturating polynomial with exact half-half split exists
on the tested family.
The excess direction of a P1 window is $C$'s lowest
eigenvector (which \emph{is} $\Tfrak_0$'s top singular vector),
not the Fourier hole-Nyquist face of r410.
No RH claim.
\end{abstract}

\section{The quotient}

Write $d_0=N_w-3$ and $\mathcal{K}_0=\ker(\mathrm{ev}_Y)$
inside $\mathcal{P}_{<d_0}$, so
$\mathcal{K}_0=P_Y\cdot\mathcal{P}_{<d_0-|Y|}$.
$\Tfrak_0$ is the min-norm interpolant $Y\to X$ in the
dual weights $u^\vee$ (r409).
By construction,
$\Tfrak_0(W_Y^{1/2}q|_Y)=W_X^{1/2}q^*|_X$ where $q^*$ is
the $X$-orthogonal projection of $q$ onto $\mathcal{K}_0^\perp$.
On $\mathcal{K}_0$ the $Y$-energy vanishes and the split
does not apply.

\begin{lemma}[Dictionary and projection, over $\mathbb{Q}$]
\label{lem:dict}
On the five-node toy of r409, $\Tfrak_0^*\Tfrak_0=C^{-1}$
to $4.6\cdot 10^{-17}$ and
$\|\Tfrak_0\|=1/\sqrt{\lambda_{\min}(C)}$.
$P_Y$ has $Y$-energy $0$.
After subtracting the $\mathcal{K}_0$-component in the
$X$-inner product, $\Tfrak_0(s_Y q|_Y)=s_X q_{\min}|_X$
to $1.3\cdot 10^{-15}$.
The mutants $u^\vee=1/u$ and $d_0-1$ break the identity.
\end{lemma}

\begin{lemma}[Energy split]\label{lem:split}
For $q\in\mathcal{K}_0^\perp$ write
$v=W_Y^{1/2}q|_Y$.
Then
\[
  \frac{\|q\|_{X,u^\vee}^2}{\|q\|_{Y,u^\vee}^2}
  \;=\;
  \frac{v^T C^{-1}v}{v^T v}.
\]
Hence
$\|\Tfrak_0\|\le 1\iff C\succeq I\iff$ every such $q$
satisfies $\|q\|_X\le\|q\|_Y$.
A raw Chebyshev $T_k$ is not in the quotient:
on FRAME-A, $T_1$ has energy ratio $2.73>\|\Tfrak_0\|^2=1.17$.
\end{lemma}

\begin{lemma}[Machine dictionary]\label{lem:machine}
On FRAME-A,
$\|\Tfrak_0^*\Tfrak_0-C^{-1}\|_F/\|C^{-1}\|_F=6.64\cdot 10^{-14}$
and $\|\Tfrak_0\|=1.080138437324=1/\sqrt{C_{\min}}$
to $10^{-14}$, $C_{\min}=0.857119$.
The top right singular vector of $\Tfrak_0$ equals $C$'s
lowest eigenvector (correlation $1$).
The Fourier hole-Nyquist face $v_F$ of r410 has
correlation $0.386$ and $\|\Tfrak_0 v_F\|=0.656538<1.080$.
The $\mu$-OP profile $B^T v$ peaks at index $165/181$
(deep, not a low-degree $T_k$).
\end{lemma}

\section{Saturation is not exact}

\begin{prop}[PD gap, not an identity]\label{prop:sat}
On $k_z=42$, $C_{\min}-1=+4.565\cdot 10^{-8}$ and
$\|\Tfrak_0\|-1=-2.283\cdot 10^{-8}$.
On $k_z=130$, $C_{\min}-1=+4.90\cdot 10^{-9}$.
On core-$42$ the $14$ vacuous windows have
$C_{\min}-1\in[1.33\cdot 10^{-7},7.54\cdot 10^{-5}]$,
none equal to $0$.
There is no tested combinatorial $q$ with exact
$\|q\|_X=\|q\|_Y$ on the quotient.
The printed $1.00000$ of r409 is $1/\sqrt{1+\varepsilon}$.
\end{prop}

\begin{prop}[$k=37$ is an eigenvalue crossing]\label{prop:k37}
Along the $\theta$-ordered prefix of FRAME-A holes,
$k=36$ has $C_{\min}-1=+6.82\cdot 10^{-5}$ ($n_C=0$) and
$k=37$ has $C_{\min}-1=-9.23\cdot 10^{-5}$ ($n_C=1$,
$\|\Tfrak_0\|=1.000046$).
The last-added Christoffel weight is $K=6.94$, not a
collapse.
Leave-one-out from the $37$-prefix restores $n_C=0$ for
several holes, so the birth is not a single named atom.
Permute (same $Y$) already exceeds at prefix $k=10$;
its $K_{\min}=0.148$ appears only at full $Y$ (r410).
\end{prop}

\section{Kills}

(Christoffel collapse of the $\mu$-chain; the dictionary
still holds).
Scramble: $n_C=21$, $\|\Tfrak_0\|=16.66$, $C_{\min}=0.0036$.
Periodic $1010$ at $S=20$: $\|\Tfrak_0\|=165$, $n_C=3$.
Living $\chi_3$-$9$: $\|\Tfrak_0\|=0.99956<1$;
dead $\chi_3$-$15$: $\|\Tfrak_0\|=1.00092>1$
(the split holds; death is edge, r401).

\section{What this reduces, and what it does not}

P1 is exactly one excess direction on the quotient:
exactly one $q\in\mathcal{K}_0^\perp$ with $\|q\|_X>\|q\|_Y$,
equivalently $\#\{\lambda(C)<1\}=1$, equivalently
exactly one singular value of $\Tfrak_0$ above $1$.
That is r407 in energy language.
It is not an exact half-filling identity, not a
combinatorial saturator, and not $L^*$.
The mincut does not move.
