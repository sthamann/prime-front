\section{The scalar}\label{sec:obj}

Write $A_{0}=R_{N-3}-\tfrac12 I$ on the hole nodes $Y$
and $s=u_B$ the Sherman--Morrison border of r362/r401,
with border scale $\mathrm{den}>0$.
In the r369 Haynsworth matrix
$\Phi_N=J^{-1}+U^{T}A_{0}^{-1}U$ the border-border entry is
\begin{equation}\label{eq:split}
\phibb
\;=\;
[J^{-1}]_{bb}+s^{T}A_{0}^{-1}s
\;=\;
c_J+\Sigma,
\qquad
c_J=\mathrm{den}-2,
\end{equation}
where $[J^{-1}]_{bb}=\mathrm{den}$ and the $-2$ is the last-slot
inverse $(A^{\dagger})_{\mathrm{last}}^{-1}=(-1/2)^{-1}$.
The r407 dictionary
$R=C(I+C)^{-1}$ gives
$A_{0}=\tfrac12(C-I)(C+I)^{-1}$ and therefore
\begin{equation}\label{eq:Cres}
A_{0}^{-1}
\;=\;
2(C+I)(C-I)^{-1},
\qquad
\Sigma
\;=\;
s^{T}A_{0}^{-1}s.
\end{equation}
Spectrally $\Sigma=\sum_i|\langle s,\psi_i\rangle|^{2}/\lambda_i(A_{0})$.
On living P1 there is one negative bulk eigenvalue
(the window-emergent defect).  Split
$D=|\langle s,\psi_{-}\rangle|^{2}/\lambda_{-}$

positive rest $R_{+}$.

\section{The lemma}\label{sec:lem}

\begin{lemma}[$\varphi_{bb}$ sign]\label{lem:phibb}
$\phibb<0$ for every window of the selected sequence,
source-pure: the border self-energy through the bulk
propagator undershoots the explicit $J$-bound.
\end{lemma}

\noindent
\textbf{Verdict: REFUTED} as a world-blind law on the
named living census (six vacuous core windows have
$\phibb>0$).  On the computed selected anchors
$a_k=2^k$ ($k=4,5,6,7,9$) the inequality holds as a
census, not a theorem (no $k_{0}$).
The identity \eqref{eq:split}--\eqref{eq:Cres} is not
the lemma; it is the exact decomposition against which
the sign was tested.

\begin{prop}[split and $C$-resolvent]\label{prop:split}
Over $\mathbb{Q}$,
$\phibb=c_J+\Sigma$ with $\Sigma=s^{T}A_{0}^{-1}s$,
and $A_{0}^{-1}=2(C+I)(C-I)^{-1}$.
P1 toy $A_{0}=\operatorname{diag}(-\tfrac12,2,3)$,
$s=e_{0}$, $\mathrm{den}=1$:
$c_J=-1$, $D=-2$, $R_{+}=0$, $\phibb=-3$.
Vacuous toy $A_{0}=\operatorname{diag}(2,3,4)$,
$s=e_{0}$, $\mathrm{den}=\tfrac12$:
$c_J=-\tfrac32$, $\Sigma=\tfrac12$, $\phibb=-1$.
The $2\times 2$ diagonal $C$-toy matches
\eqref{eq:Cres} exactly.
A pole-save toy $A_{0}=\operatorname{diag}(-\tfrac1{10},\tfrac15)$,
$s=(1,1)$ has $\phibb=-6$ and $\phibb|_{D=0}=+4$
(the mutant construction is live).
\end{prop}
