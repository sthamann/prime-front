\section{The lemma}\label{sec:lem}

\begin{lemma}[$c_J$ vs $\Sigma$]\label{lem:cj}
On the vacuous branch at large $N$,
$c_J<-\Sigma$ (equivalently $\phibb<0$, equivalently
$\mathrm{sch}<0$ cofinally via the r419 asymptotics),
source-pure.
\end{lemma}

\section{The $\mathrm{den}$ formula}\label{sec:den}

\begin{prop}[border normalisation]\label{prop:den}
Write $B_w=S_{N-2}+\tfrac57$ for the r362/r397 budget
($S=\sum\rho_k$, $\rho_k=F_k^{2}/h_k$ the von~Mangoldt
border-chain masses) and $b$ the $\mu$-orthonormal
border vector.
Let $v=B_{\mu}b/\sqrt{B_w}$, $\gamma=\|b\|^{2}/B_w$,
$v_{t}=\varepsilon_{Y}\odot v$ and $s=R_{N-3}v_{t}$.
Then
\begin{equation}\label{eq:den}
\mathrm{den}
\;=\;
1+\gamma-v_{t}\cdot s,
\qquad
c_J=\mathrm{den}-2.
\end{equation}
Over $\mathbb{Q}$, $\gamma=\tfrac23$, $v_{t}\cdot s=\tfrac16$
gives $\mathrm{den}=\tfrac32$, $c_J=-\tfrac12$.
At $k_z=9$,
$\mathrm{den}=1.60111$, $\gamma=0.6778$, $B_w=8.382$.
On the named census every window has
$\mathrm{den}\in[1.460,1.652]<2$, so the gap to $2$ is
$O(1)$ in $[0.348,0.540]$.
Vacuous $\mathrm{den}\sim N$ slope $-0.03$ (flat).
\end{prop}

Thus $\mathrm{den}<2$ is an explicit budget statement
with an $O(1)$ margin.
It is \emph{not} the balance $c_J+\Sigma<0$.

\section{Where $s$ sits; the $\Sigma$ bound}\label{sec:sig}

\begin{prop}[occupation and the naive bound]\label{prop:sig}
On vacuous windows $s$ occupies moderate $C$-modes
(mass-weighted $\lambda\sim 3.2$--$4.6$).
The $C_{\min}$-share is $\le 3\cdot 10^{-7}$.
$\|s\|^{2}\sim 0.06$ is flat.
The bound
$\Sigma\le 2\|s\|^{2}\max_{\mathrm{occ}}(\lambda+1)/(\lambda-1)$
over modes with relative mass $>10^{-6}$ is
$200$--$24000$ times $\Sigma$: near-razor modes with
tiny occupation blow the maximum.
No tight source-pure upper bound of the form
$\Sigma\le C_{\mathrm{expl}}\|s\|^{2}$ with $C_{\mathrm{expl}}=O(1)$
is obtained.
$\Sigma$ itself is $O(0.40)$ and flat.
\end{prop}

\section{The reserve; the six; selected}\label{sec:res}

\section{One balance}\label{sec:one}

\section{What is a theorem, what is a census}\label{sec:honest}
