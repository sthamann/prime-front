\begin{abstract}\noindent
Rounds 381 and 385 left $C_\varepsilon$ as the live first-order pairing
$\lvert\mathrm{FO}_k[-\nu]\rvert\le\tfrac1{32}$ on the last twelve, written
as $\gamma_k\lvert Q_k-Q_{k-1}\rvert$ with
$Q_k=Q_k^T+\Delta_k$.
This note asks whether a Szeg\H{o}/Geronimus bound on the consecutive
deformation $\Delta_k-\Delta_{k-1}$, together with second-order domination
along $\mu\to\mu-\nu$, proves that box.
What is proved: the split
$\mathrm{FO}_k=\gamma_k\bigl((Q_k^T-Q_{k-1}^T)+(\Delta_k-\Delta_{k-1})\bigr)$;
homogeneity of monic Jacobi coefficients under a global mass scale;
and that a Fej\'er-pure two-period mesh has $\varepsilon=0$ and
$\mathrm{FO}=0$.
What is \emph{not} proved: $C_\varepsilon$.
The candidate $\lvert\Delta_k-\Delta_{k-1}\rvert\le C\cdot\mathrm{osc}(\varepsilon\text{ on }\nu)$
holds on MAIN with $C_*=0.0056$ and is useless (the majorant
$g\,(\mathrm{d}Q^T+C_*\,\mathrm{osc})$ reaches $0.191>\tfrac1{32}$);
consecutive $Q$ is uncorrelated with $\lVert\varepsilon\rVert_\infty$
($\rho=0.025$) and with range-oscillation ($\rho=-0.287$).
A scramble of the same $d_{\mathrm{arm}}$ channel keeps
$\mathrm{osc}(\varepsilon)=3.54<8.76$ and sends $\lvert\mathrm{d}\Delta\rvert$
to $10^6$.
Along the $\mu$-reference the midpoint ratio is $0.861$ at degree $40$
and $0.640$ at full depth, both $>\tfrac14$;
$\lvert R_2\rvert/\lvert\mathrm{FO}\rvert=4.19$ on the last twelve.
Neumann domination is false, so $R_2$ is not absorbed.
The hydra-legal remainder is the r381 triple
$G_\varepsilon^\mu\wedge C_\varepsilon\wedge R_2$, with $C_\varepsilon$
a cancelled last-twelve pairing (not an $\varepsilon$-size bound) and
$G_\varepsilon^\mu$ the Jacobi box of Fej\'er$\times d_{\mathrm{arm}}$
(the r342 dictionary, $v_{\mathrm{pred}}<10^{-4}$, is the right language
for that sibling).
No RH claim.
\end{abstract}

\section{The object}\label{sec:obj}

Write $\pi_k$ for the $\mu$-orthonormal polynomials and
\begin{equation}\label{eq:Q}
Q_k:=\langle\pi_k^2\rangle_\nu,\qquad
Q_k^T:=\frac{\sum_\nu v(1+\cos 2k\theta)}{\sum_\mu w(1+\cos 2k\theta)},
\qquad
\Delta_k:=Q_k-Q_k^T.
\end{equation}
Round 385: $\mathrm{FO}_k[-\nu]=\gamma_k(Q_k-Q_{k-1})$ as an identity,
and $Q^T$ stays at the mass ratio $\alpha$ under the trivial Weyl bound.
Condition $C_\varepsilon$ of~\cite{r381} is
$\lvert\mathrm{FO}_k[-\nu]\rvert\le\tfrac1{32}$ for the last twelve
degrees of the $v_2$-chain.
The mesh $x=\cos(2\pi j/L)$, $L=4h-2$, and the factorisation
$w\propto(1-x)\,d_{\mathrm{arm}}$ are construction identities
of~\cite{v3p}.
Write $\varepsilon_j=(w_j/\bar w_j)-1$ for the Fej\'er-stripped
relative arm-length, $\bar w\propto(1-x)$, mean-zero by construction.

The candidate of this round: a Geronimus bound
$\lvert\Delta_k-\Delta_{k-1}\rvert\le C\cdot\mathrm{osc}(\varepsilon\text{ on }\nu)$
with an explicit $C$ small enough that, together with the Chebyshev
consecutive $\mathrm{d}Q^T$ and a dominated $R_2$, one has $C_\varepsilon$.

\section{What is exact}\label{sec:exact}

\begin{lemma}[split]\label{lem:split}
On a quasi-definite atomic $\mu$,
\[
\mathrm{FO}_k[-\nu]
=
\gamma_k\Bigl(
(Q_k^T-Q_{k-1}^T)+(\Delta_k-\Delta_{k-1})
\Bigr).
\]
This is~\cite{r385}'s pairing plus~\eqref{eq:Q}.
FRAME-A $w=9$, last twelve, floating-point residual $2\cdot10^{-15}$.
\end{lemma}

\begin{lemma}[homogeneity]\label{lem:hom}
If every weight of a quasi-definite atomic measure is scaled by a
positive $\lambda\in\Q$, the monic orthogonal polynomials are unchanged
and $\gamma_k=h_k/h_{k-1}$ is invariant.
Exact over $\Q$ on a five-atom toy at $\lambda=3$, depth $3$.
\end{lemma}

A global mass scale is therefore invisible to $\gamma$.
The crude wall $\lVert\varepsilon\rVert_\infty=5.166$ of~\cite{r381}
already quotients out that scale (Fej\'er-stripped, mean zero).
What remains is variation of $d_{\mathrm{arm}}$.

\begin{lemma}[Fej\'er-pure]\label{lem:pure}
On the two-period Fej\'er mesh of~\cite{r382} (equal occupancy,
weights $\propto(1-x)$, no $d_{\mathrm{arm}}$), one has $\varepsilon=0$,
$Q_k$ independent of $k$, and $\mathrm{FO}_k=0$.
Checked at $S=21$ and $S=81$, amplitude $2/3$.
\end{lemma}

This is the control deformation: when $\varepsilon$ vanishes, consecutive
$\Delta$ vanishes.  The live field is $d_{\mathrm{arm}}$.

\section{Oscillation does not close $C_\varepsilon$}\label{sec:osc}

Write $\mathrm{osc}_\nu(\varepsilon)=\max\varepsilon|_\nu-\min\varepsilon|_\nu$
and $C_{\mathrm{emp}}=\lvert\mathrm{d}\Delta\rvert_{12}/\mathrm{osc}_\nu(\varepsilon)$.

\begin{prop}[census]\label{prop:census}
On the sealed constructors:
\begin{enumerate}
\item FRAME-A $w=9$: last-twelve
  $\lvert\mathrm{FO}\rvert=0.00978<\tfrac1{32}$,
  $\mathrm{d}Q=0.0387$, $\mathrm{d}Q^T=0.0197$,
  $\mathrm{d}\Delta=0.0495$, $\lVert\varepsilon_\mu\rVert_\infty=5.166$,
  $\mathrm{osc}_\nu=8.764$, $C_{\mathrm{emp}}=0.0056$,
  $\gamma_{12}=0.276$.
\item Core-$42$: $\mathrm{corr}(\mathrm{d}Q,\lVert\varepsilon\rVert_\infty)=0.025$,
  $\mathrm{corr}(\mathrm{d}\Delta,\mathrm{osc}_\nu)=-0.287$.
  Amplitude grows from $5$ to $123$ while consecutive $Q$ stays $O(10^{-2})$
  to $O(10^{-1})$.  The $L^\infty$ wall of~\cite{r381} does not enter the
  consecutive difference --- and neither does range-oscillation of
  $\varepsilon$ (both quantities track the same $d_{\mathrm{arm}}$
  dynamic range).
\item The osc-majorant
  $\gamma(\mathrm{d}Q^T+C_*\,\mathrm{osc}_\nu)$ with $C_*=0.0056$
  reaches $0.191>\tfrac1{32}$ (closes on $5/42$).
  Doubling $C_*$ already overshoots $\tfrac1{32}$ on $w=9$ itself
  (majorant $0.0328$).
\item Eleven small-$N$ windows ($N\le 220$): last-twelve
  $\lvert\mathrm{FO}\rvert\le 0.01182$ on MAIN and $\le 0.01541$ on
  $\chi_3$, both $<\tfrac1{32}$.
  (The $181$-pack builder was not rerun; MAIN-$85$ at depth $80$ is a
  fixed-depth smoothness census, not last-twelve of $v_2$ for large $N$,
  disclosed.)
\end{enumerate}
\end{prop}

The bound $\lvert\mathrm{d}\Delta\rvert\le C_*\,\mathrm{osc}_\nu$ is true
on the source and does not prove $C_\varepsilon$: range-oscillation of
$\varepsilon$ is still $O(\lVert\varepsilon\rVert_\infty)$, and the
last-twelve pairing cancels that amplitude.
A single Fourier mode $a\cos(2m\theta)$ on an equal-weight cosine toy
does move last-twelve $\mathrm{FO}$ proportionally to $a$; the source
$d_{\mathrm{arm}}$ is not that mode.
The Geronimus picture that last-twelve $\delta\gamma$ tracks Fourier
mode $q\sim M$ of $d_{\mathrm{arm}}$ remains the natural attack on
$C_\varepsilon$ and is not carried here --- it is $C_\varepsilon$ itself.

\section{The $\mu$-reference does not dominate $R_2$}\label{sec:r2}

Round 381 measured $\lvert\mathrm{quad}\rvert/\lvert\mathrm{lin}\rvert=1.013$
along Fej\'er-constant $\to$ actual at degree $40$.
The reduction $G_\varepsilon\Longleftarrow G_\varepsilon^\mu\wedge C_\varepsilon\wedge R_2$
uses the other path: $\mu$ as reference, $-\nu$ as the finite step.
A ratio $<\tfrac14$ would have made $R_2$ a Neumann remainder of the
Christoffel--Darboux perturbation of size $\mathrm{mass}(\nu)/\mathrm{mass}(\mu)=0.061$.

\begin{prop}[$R_2$ not dominated]\label{prop:r2}
On FRAME-A $w=9$, writing $g(t)$ for last-twelve $\gamma$ of $\mu-t\nu$:
\begin{enumerate}
\item Degree $40$: $\lvert\mathrm{quad}\rvert/\lvert\mathrm{lin}\rvert=0.861$
  (Fej\'er-path $1.013$ reproduced).
\item Full depth $n=182$: ratio $0.640$;
  exact $\lvert R_{2,k}\rvert/\lvert\mathrm{FO}_k\rvert=4.19$ on the last
  twelve, with
  $R_{2,k}=\gamma_k(\mu-\nu)-\gamma_k^\mu-\mathrm{FO}_k[-\nu]$.
\item Eleven small-$N$ windows at degree $40$: ratios in $[0.559,1.126]$,
  all $>\tfrac14$.
\end{enumerate}
The mass ratio $0.06$ is not the small parameter of the last twelve:
the Christoffel sampling $Q_k$ is $O(1)$ (last-twelve maximum $0.389$).
First-order perturbation theory about $\mu$ is a finite $O(1)$ step.
$R_2$ is not absorbed.
\end{prop}

\section{Adversaries}\label{sec:adv}

\paragraph{Scramble.}
FRAME-A $w=9$, seed $1$: $\mathrm{osc}_\nu(\varepsilon)=3.54<8.76$
(smaller than MAIN) and $\lvert\mathrm{d}\Delta\rvert_{12}=2.16\cdot10^6$.
The osc-bound with any $C$ that works on MAIN is false on the scramble.
Scramble does not inflate the $d_{\mathrm{arm}}$ field; it breaks the
$\mu$-OP sampling of $\nu$ (the r381/r385 named kill, reproduced as a
kill of Geronimus-by-oscillation).

\paragraph{Mutant $C\times 2$.}
Twice the measured $C_*$ already fails to prove $C_\varepsilon$ on $w=9$
(majorant $0.0328>\tfrac1{32}$).  Halving $C_*$ fails on $6/42$ core
windows.  The measured constant is not a closing constant.

\paragraph{Wrong reference.}
Replacing $\mathrm{osc}_\nu$ by $\lVert\varepsilon\rVert_\infty$ makes
the majorant worse (core maximum $0.353$).  Amplitude is the wrong
norm for a consecutive difference.

\paragraph{Two-period, $c=2/3$.}
Lemma~\ref{lem:pure}: $\varepsilon=0$ and $\mathrm{FO}=0$.
$C_\varepsilon$ holds trivially and $\lambda_{\max}(E_{22})>1$ still
(the r385 witness that $Q<1$ is not the floor plate).
A control, not a kill of $C_\varepsilon$.

\section{The reduction}\label{sec:red}

\begin{prop}[reduction]\label{prop:red}
The candidate ``oscillation of $\varepsilon$ plus Neumann $R_2$ proves
$C_\varepsilon$'' is strictly larger than the r381 triple:
\begin{enumerate}
\item \emph{Split and homogeneity.}  SATZ
  (Lemmas~\ref{lem:split}--\ref{lem:pure}).
  Consecutive $\Delta$ is a $k$-th layer pairing; a global mass scale
  does not enter.
\item \emph{Osc-Geronimus as a closing bound.}  False.
  The inequality with $C_*=0.0056$ is a MAIN census, not a theorem of
  $\mathrm{osc}(\varepsilon)$, and the implied majorant does not yield
  $\tfrac1{32}$.  Scramble is the named kill.
\item \emph{Neumann $R_2$.}  False on the $\mu$-reference
  (Proposition~\ref{prop:r2}).  The reduction does not collapse to
  $G_\varepsilon^\mu\wedge C_\varepsilon$.
\item \emph{What remains.}
  $C_\varepsilon$: cancelled last-twelve pairing of $d_{\mathrm{arm}}$
  against $\pi_k^2-\pi_{k-1}^2$ of $\mu$ (Geronimus/Fourier mode
  $q\sim M$, not $\mathrm{osc}(\varepsilon)$).
  $R_2$: Taylor remainder of the finite step $\mu\to\mu-\nu$.
  $G_\varepsilon^\mu$: the Jacobi box of the positive measure
  $w\propto(1-x)\,d_{\mathrm{arm}}$.  The r342 dictionary
  $F_A(\xi)=-\log\pi+\Re\psi(\tfrac14+i\xi/2)$ plus the two-cell tent,
  with measured $v_{\mathrm{pred}}<10^{-4}$, is the right language for
  that sibling --- an explicit Jacobi perturbation of Jacobi-$(0,1)$,
  not an oscillation bound.
\end{enumerate}
The chain of~\cite{v3p,r381} stands:
$V_3'\Longleftarrow G_\varepsilon\Longleftarrow
G_\varepsilon^\mu\wedge C_\varepsilon\wedge R_2$.
\end{prop}
