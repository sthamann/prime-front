\begin{abstract}\noindent
Rounds 404 and 405 stopped at the same named remainder:
Euler/Digamma Grams are exact on the weight layer, while
$A_0=R_{N-3}-\half I$ lives behind the dual resolvent.
This note writes the exact intertwiner.
Let $u^\vee$ be the Borodin dual of $|u|$ and let $C$ be the
Christoffel--Darboux Gram of the $\mu$-only dual chain,
evaluated on the hole nodes $Y$ and dressed by
$\sqrt{u^\vee|_Y}$.
Then
\[
  R_{N-3}\;=\;C(I+C)^{-1},
\]
with $\operatorname{eig}(R)=\lambda(C)/(1+\lambda(C))$, hence
$\indm A_0=\#\{\lambda(C)<1\}$ and P1 $\iff\lambda_2(C)\ge 1$.
The $\nu$-Hankel after compression has rank $N_\nu$, not one.
The Loewner sandwich of the inverse upper-bounds $R$ and
cannot force $R\succeq\half I$.
The Euler source Gram is class SOURCE\_GRAM\_EXACT on the
moment layer of $u$, which is \emph{not} the Gram that $R$
inverts.
No RH claim.
\end{abstract}

\section{The formula}\label{sec:formula}

Write $u$ for the folded signed union weight on the cosine
grid $X$, $Y=\operatorname{supp}(u_-)$ the hole nodes, and
$u^\vee_j=(1-x_j)^2/(S|u_j|\kappa_j^2)$ the Borodin dual
(r356).
Let $n=N_w-3$.
In the Chebyshev basis $(T_j)_{j<n}$ let $G_+$ be the Hankel
Gram of $u^\vee$ on the $\mu$-nodes, $\Phi$ the
Chebyshev--Vandermonde on $Y$, $D=\operatorname{diag}(u^\vee|_Y)$,
and $A=D^{1/2}\Phi$.
Then $G_\nu=A^TA$ and $G=G_++G_\nu$, and the dual CD Gram of
depth $n$ on $Y$ is the congruence
\begin{equation}\label{eq:fG}
  R_{N-3}
  \;=\;
  A G^{-1} A^T
  \;=\;
  C(I+C)^{-1},
  \qquad
  C\;:=\;A G_+^{-1} A^T.
\end{equation}
The second form is Woodbury:
$A(G_++A^TA)^{-1}A^T=C(I+C)^{-1}$.
The three-term recurrence of $u^\vee|_\mu$ \emph{is} the
Cholesky of $G_+$ in the OP basis, so $C=BB^T$ with $B$ the
$\mu$-OP matrix evaluated on $Y$ and dressed by
$\sqrt{u^\vee|_Y}$ --- the numerically stable form of
\eqref{eq:fG} at working depth (Hankel $\kappa(G)\sim 10^{11}$
on a deep razor window mis-reads $\lambda_2(C)-1$).

\begin{lemma}[Hankel linearity]\label{lem:hankel}
Over $\mathbb{Q}$, the monomial Hankel satisfies
$G(\mu-\nu)=G(\mu)-G(\nu)$ and $G(\mu)=G(\mu_{p_2})+G(\mu_{p_3})$
on a five-atom toy.
\end{lemma}

\begin{lemma}[Woodbury and $R=C(I+C)^{-1}$]\label{lem:woodbury}
Over $\mathbb{Q}$: the $2\times 2$ Woodbury identity with
$\eta=\tfrac56$, $\det=11$; and for $C=\operatorname{diag}(1,2)$,
\[
  C(I+C)^{-1}
  \;=\;
  I-(I+C)^{-1}
  \;=\;
  \operatorname{diag}\bigl(\tfrac12,\tfrac23\bigr).
\]
\end{lemma}

\begin{lemma}[Inverse is Loewner-antitone]\label{lem:antitone}
If $A=\operatorname{diag}(2,4)$ and $B=\operatorname{diag}(1,3)$
then $A\succeq B\succ 0$ and $A^{-1}\preceq B^{-1}$ over
$\mathbb{Q}$.
\end{lemma}

\begin{lemma}[Fractional-linear dictionary]\label{lem:fl}
If $C=\operatorname{diag}(2,\tfrac12)$ then
$R=C(I+C)^{-1}=\operatorname{diag}(\tfrac23,\tfrac13)$,
so $\#\{\lambda(C)<1\}=1=\indm(R-\half I)$.
In exact arithmetic, $\lambda\mapsto\lambda/(1+\lambda)$
sends $\{\lambda<1\}$ onto $\{r<\half\}$ on $[0,\infty)$.
\end{lemma}

\begin{lemma}[Machine formula]\label{lem:formula}
On FRAME-A, $\|C(I+C)^{-1}-R_{\mathrm{chain}}\|_F/\|R\|_F
=1.84\cdot 10^{-14}$ at depth $n=181$, and the Chebyshev--Hankel
form of \eqref{eq:fG} matches the chain at depth $20$ to
$3.1\cdot 10^{-15}$.
The sorted fractional-linear image of $\operatorname{eig}(C)$
matches $\operatorname{eig}(R)$ to $1.5\cdot 10^{-13}$.
$C_{\min}=0.85712$, $\lambda_2(C)=1.00018$,
$C_{\min}/(1+C_{\min})=0.46153=r_{\min}(R)$, $\indm A_0=1$.
\end{lemma}

\begin{lemma}[Signed consumption]\label{lem:sign}
Passing the signed $u$ to the dual weight produces
$104$ negative dual masses on FRAME-A ($=|Y|$); the
construction uses $|u|$.
The source Chebyshev--Hankel of $u$ and the dual Hankel of
$u^\vee$ differ by relative residual $141$ at degree $12$
--- $R$ inverts the dual Gram, not the Euler-source Gram.
\end{lemma}

\section{Moment-layer Euler Gram}\label{sec:euler}

\begin{prop}[SOURCE\_GRAM\_EXACT on the moment layer]
\label{prop:euler}
The Chebyshev--Hankel of the folded prime-positive weights
equals $FF^T$ with features $F_{\alpha j}=T_j(x_\alpha)\sqrt{w_\alpha}$,
residual $1.55\cdot 10^{-16}$, $\lambda_{\min}>0$.
This is the source-exact Gram sought by r404, on the layer
where Fold/Tent is linear (lag-residual $3.54\cdot 10^{-16}$).
A permutation of those weights yields relative residual $0.54$.
Node jitter of $G(\nu)$ at $10^{-3}$ yields residual $0.014$.
\end{prop}

The class is tautological as a Gram of monomials, and
\emph{not} a Cholesky of $A_0$.
It does not transport through $u\mapsto 1/|u|$.

\section{Rank, sandwich, census}\label{sec:rank}

\begin{prop}[$\nu$-rank is $N_\nu$, not one]\label{prop:rank}
On FRAME-A, $\operatorname{rank} G_\nu=104=|Y|$ and
$\operatorname{rank}(R-C)=104$ ($r_{\mathrm{eff}}=1.18$).
The dual CD tail $R_{N_w-1}-R_{N_w-3}$ is rank two (r363)
and does not isolate the $A_0$-defect.
The one defect of MAIN is spectral:
exactly one eigenvalue of $C$ lies below $1$.
\end{prop}

\begin{prop}[Sandwich is the wrong direction]\label{prop:sand}
$C-R=C(I+C)^{-1}C\succeq 0$, so $C\succeq R$.
On FRAME-A, $\lambda_{\min}(C)=0.857>\half$ while
$\lambda_{\min}(R)=0.461<\half$.
Operator monotonicity of $G\mapsto G^{-1}$ therefore
upper-bounds $R$ and cannot force $R\succeq\half I$.
\end{prop}

\begin{prop}[Census and kills]\label{prop:census}
On core-$42$, $\indm A_0=\#\{\lambda(C)<1\}$ on $42/42$
($28$ P1 and $14$ vacuous).
Living $\chi_3$: $37/37$; dead $\chi_3$: $5/5$ (the dead
fulfill the transport; they die at $\mathrm{sch}>0$, r401).
Scramble: $21=21$.
Weight-permutation: $20=20$ (kills P1; the dictionary still
holds).
Wrong depth $n+1$: relative residual $0.031$, rank one.
\end{prop}

\section{What this reduces, and what it does not}\label{sec:red}

P1 on a canonical window is exactly $\lambda_2(C)\ge 1$,
with $C$ the $\mu$-only dual CD kernel on $Y$.
That is a threshold form on the \emph{right} layer (the dual
moment inverse), which r400 did not have.
It is not a source-explicit factorisation of $A_0$ by Euler
features: those live on $G(u)$, and $R=f(G(u^\vee))$.
The Aronszajn parallel-sum of reproducing kernels is the
coefficient-layer identity
$(G_++G_\nu)^{-1}=G_+^{-1}:G_\nu^{-1}$ when both are
invertible; $G_\nu$ has rank $N_\nu$, so the update is
rank-$N_\nu$, not rank one.
