\begin{abstract}\noindent
The compact-window Weil form at fixed $L$ is a finite program:
high-mode dominance (classical) plus a finite Schur block.
The frequency pairing is
$A=(2\pi)^{-1}\int\sigma_A(t)|\hat h(t)|^2\,dt$ with
$\sigma_A(t)=\mathrm{Re}\,\psi(\tfrac14+it/2)-\log\pi$,
identical to the r475/r476 $u$-space symbol (interval match at
six test frequencies).  Prime nodes run to $e^{2L}$, not $e^L$.
A crude high-mode threshold $t_0(L)$ is sealed at $L=0.3,0.8,1.0$.
The crude coupling bound does not close the Schur complement;
the direct $2\times2$ of (constant, first Dirichlet) is positive
definite at all three calibration points.  Yoshida/Bombieri
$L=0.3$ ($P\equiv0$) passes.  Not $\lambda_*(L)\ge0$.
Verdict \texttt{ARCHITECTURE\_STANDS($0.3,0.8,1.0$)}.
No RH claim.
\end{abstract}

\section{Frequency form}

After r461/r471, $\mathrm{supp}\,h\subset[-L,L]$,
$g=h*\widetilde h$, $\mathrm{supp}\,g\subset[-2L,2L]$,
$\QW=A-P+\Pi$ with
$P(g)=\sum_{n\le e^{2L}}2\Lambda(n)n^{-1/2}g(\log n)$
(scramble-sensitive) and $\Pi$ of rank at most two.
Plancherel $\hat h(t)=\int h(x)e^{itx}\,dx$,
$\|h\|_2^2=(2\pi)^{-1}\int|\hat h|^2$:
\[
 A(g)=\frac1{2\pi}\int_{\mathbb R}\sigma_A(t)\,|\hat h(t)|^2\,dt,
\]
\begin{lemma}[Symbol identity]\label{lem:sig}
$\sigma_A(t)=\mathrm{Re}\,\psi(\tfrac14+it/2)-\log\pi$,
equal to the r476 Bose series
$-(\gamma+\log\pi)+\sum_k 2\bigl[(2k+2)^{-1}-\alpha_k/(\alpha_k^2+t^2)\bigr]$,
$\alpha_k=2k+\tfrac12$.  Sealed overlap at
$t=0,1,2,5,10,20$.  In particular
$\sigma_A(0)=-(\gamma+\log\pi)-(\pi/2+3\log2)\in[-5.37219,-5.37218]$
and $\sigma_A$ is even and increasing on $[0,\infty)$.
\end{lemma}

\section{High-mode lemma}

On the ideal subspace $\{\hat h=0\text{ on }|t|<t_0\}$,
\[
 \frac{\QW(h)}{\|h\|_2^2}
 \ge\sigma_A(t_0)-2S_{\mathrm{eff}}(L)-4\sinh L,
\]
with $S_{\mathrm{eff}}=\sum_{n\le e^{2L}}\Lambda(n)n^{-1/2}$
exact and $|\Pi|\le4\sinh L\,\|h\|_2^2$ from
$\hat h(\pm i/2)=\langle h,e^{\mp x/2}\rangle$.

\begin{lemma}[$t_0$ and $c$]\label{lem:t0}
Fix $c_\star=0.01$.  If $\sigma_A(t_0)\ge2S_{\mathrm{eff}}+4\sinh L+c_\star$
then $\QW\ge c_\star\|h\|_2^2$ on that ideal subspace.
Sealed $t_0$ (upper ends):
$t_0(0.3)=22$ (Bose, $\sigma_{\mathrm{lo}}=1.2517\ge1.2281$),
$t_0(0.8)=4300$ (log-bound $\log|z|-\log\pi-1/t$, calibrated at $t=10,20,40$),
$t_0(1.0)=2.44\cdot10^5$ (same bound).
At $L=0.3$ one has $S_{\mathrm{eff}}=0$.
\end{lemma}

\section{Schur reduction}

Split frequency at $t_0$:
$\QW=[Q_{\ell\ell},Q_{\ell h};Q_{h\ell},Q_{hh}]$.
$A$ is diagonal in frequency; $P$ and $\Pi$ couple the blocks.
If $Q_{hh}\succeq c_\star$ then
$\QW\succeq0\Longleftarrow Q_{\ell\ell}-Q_{\ell h}Q_{hh}^{-1}Q_{h\ell}\succeq0$.
The crude operator bound $\|Q_{\ell h}\|\le4\sinh L+2S_{\mathrm{eff}}$
makes the Schur correction larger than $Q_{\ell\ell}$ at every
calibration point (YB crude Schur $\approx-125$).

\begin{lemma}[Direct $2\times2$]\label{lem:22}
On $\mathrm{span}\{1,\cos(\pi x/(2L))\}$ the normalized Gram is
positive definite:
YB Schur $\in[4.673\cdot10^{-2},4.930\cdot10^{-2}]$,
$L=0.8$ Schur $\in[7.414\cdot10^{-2},7.507\cdot10^{-2}]$,
$L=1.0$ Schur $\in[3.553\cdot10^{-2},3.804\cdot10^{-2}]$.
Constant-mode Rayleigh
$r(1_{[ -L,L]})=0.1584,\,0.08027,\,0.05334$ at the three $L$.
\end{lemma}

\section{Scaling forecast}

$t_0\sim2\pi\exp\bigl(2S_{\mathrm{eff}}(L)+4\sinh L\bigr)$
and $N\sim 2Lt_0/\pi$.  $S_{\mathrm{eff}}$ grows like a truncated
prime-power Mellin sum up to $e^{2L}$.  At $L_{16}=2.773$,
$S_{\mathrm{eff}}\approx29.14$, $4\sinh L\approx31.88$,
threshold $\approx90$, hence $t_0\sim10^{39}$.  With the crude
$P$-bound the full block is not a practical computation.
Sharpening $|g(\log n)|$ on high modes (oscillation, not
$\|h\|_2^2$) turns $L_{16}$ back into a computing question.
The architecture itself does not grow with $k$ or $L\to\infty$.

\section{What this does not do}

It does not prove $\lambda_*(L)\ge0$ at any $L$.  The ideal
high-mode subspace meets compact support only at $h=0$.
The $2\times2$ is not the full Paley--Wiener block of width $t_0$.
No cofinal quantifier.  No RH claim.
