\begin{abstract}\noindent
Round 391 tests the suspicion that the two energy premises of
COMPOSE$^-$ --- $(R)$: $B(\omega+\beta,\omega+\beta)\le K D$ and
$(L)$: $L_1\le C m^\gamma$ with $\gamma<1/4$ --- are theorems of
the fold construction, not of the arithmetic of the weights.
Scramble (r383/r386) breaks neither.  What is proved: the
Euclidean split
$\lVert\Sigma\rVert^2+\lVert\Delta\rVert^2=2(\lVert P_\beta\rVert^2+\lVert P_\omega\rVert^2)$
and the correlation identity
$\lVert\Sigma\rVert^2/\lVert\Delta\rVert^2=(1+\eta)/(1-\eta)$;
Cauchy--Schwarz counting $L_1^2\le m D$; Fej\'er Gram; the DC
Rayleigh $R=3.6875$ on the length-$16$ all-ones toy.
What is \emph{not} proved: a cofinal $K$ via block-Gershgorin
(fully diagonally dominant on only $4/42$ of the core ladder;
atom-Gershgorin is $\lambda\le H$, already too coarse), or
$\gamma<1/4$ from counting (the $L_1^2\le m D$ bridge is
tautological with $n_{\mathrm{eff}}\sim m$).
Weight randomisation at frozen fold geometry respects $K=4$
except under positive alignment $P_\omega\approx P_\beta$.
The geometry kill is DC/sign-flatten, \emph{not} merged blocks.
No RH claim.
\end{abstract}

\section{Recollection}\label{sec:rec}

Write $\Sigma=\omega+\beta=P_\omega+P_\beta$,
$\Delta=\beta-\omega=P_\beta-P_\omega$, $D=\lVert\Delta\rVert^2$,
$L_1=\lVert\Delta\rVert_1$, and $B$ the positional Fej\'er form
of~\cite{r378,r386}.  The named remainder of r386 is
$B(\Sigma,\Sigma)\le K D$.  COMPOSE$^-$ of~\cite{r378} still
asks $(R)$, $(L)$, living $(Z')$~\cite{r386}, $M_3\le\phi(m)$,
and the r263 dictionary.  Claim boundary: finite identities
and named reductions.  Not evidence for or against the
Riemann hypothesis.

\section{Shared machinery}\label{sec:shared}

\begin{lemma}[Euclidean split]\label{lem:split}
$\lVert\Sigma\rVert^2+\lVert\Delta\rVert^2
=2(\lVert P_\beta\rVert^2+\lVert P_\omega\rVert^2)$.
Writing $\eta=2\langle P_\beta,P_\omega\rangle/
(\lVert P_\beta\rVert^2+\lVert P_\omega\rVert^2)$ one has
$\lVert\Sigma\rVert^2/\lVert\Delta\rVert^2=(1+\eta)/(1-\eta)$
whenever $\lvert\eta\rvert<1$.  Consequently
$B(\Sigma,\Sigma)/D=\mathrm{ray}(\Sigma)\cdot(1+\eta)/(1-\eta)$
with $\mathrm{ray}(\Sigma)=B(\Sigma,\Sigma)/\lVert\Sigma\rVert^2$.
\end{lemma}

\begin{lemma}[Fej\'er Gram]\label{lem:fejer}
$B$ is a Gram form.  Atom-Gershgorin of the Fej\'er kernel
gives $\lambda_{\max}(K)\le H$, hence $B(u,u)\le H\lVert u\rVert^2$,
already recorded in~\cite{r383} and too coarse for an absolute $K$.
\end{lemma}

\begin{lemma}[Cauchy--Schwarz counting]\label{lem:cs}
$L_1^2\le m D$ for every real block field of length $m$.
\end{lemma}

\begin{lemma}[Atom-Gershgorin is $H$]\label{lem:gersh}
The Fej\'er Toeplitz matrix has diagonal $1$ and off-diagonal
row-sum $H-1$.  Gershgorin disks give $\lambda\le H$.  This is
Lemma~\ref{lem:fejer}.
\end{lemma}

\noindent
On a length-$16$ all-ones vector with $H=4$ one has
$R=3.6875>C_{\mathrm{MAIN}}=3/10$: a field-independent bound
$C_{\mathrm{MAIN}}\le 1$ remains refuted.  Exact period-$2$
gives $R=1/16$ (anti-alignment kills the energy).

\begin{lemma}[White-block class, census]\label{lem:white}
On the $42$-rung core, $\mathrm{med}\,\lVert\Sigma\rVert^2/D=0.999$
and $\mathrm{med}\,\cos(P_\beta,P_\omega)=-0.0006$;
$\mathrm{ray}(\Sigma)\le 3.099<4$.  Block length has median $4$
and maximum $13$.  Superblock Gershgorin of $\Sigma$ is fully
diagonally dominant on only $4/42$ (minimum fraction $0.667$).
\end{lemma}

\section{Premise $(R)$}\label{sec:R}

\begin{prop}[$(R)$, reduced]\label{prop:R}
$B(\Sigma,\Sigma)\le 4 D$ on the construction class: CORE-$42$
($0/42$ violations, maximum $2.8196$ at kz$19$), $\chi_3$-$42$
($0/42$, maximum $2.746$ at kz$67$), and the $R$-star kz$37$
($B_\Sigma/D=2.975<4$).  Independent weight randomisation at
frozen fold geometry (permute, Gauss, Rademacher, exponential,
Dirichlet) stays below $K=4$.  Positive alignment
$P_\omega=a P_\beta$ with $a=0.3$ already exceeds $4$ and
$a\to 1$ is unbounded --- there is no field-independent $K$.
DC-flatten and sign-kill break $K=4$; merged folds and shuffled
blocks do \emph{not}.  Superblock Gershgorin is refuted as a
SATZ.  The mutant $K/2=2$ fails on $9/42$ and at kz$37$.
\end{prop}

\noindent
Outcome: \textbf{REDUCED} to the white-block class
($\eta\approx 0$, $\mathrm{ray}=O(1)$), with $K=4$ a census
envelope.  Remainder: prove
$\mathrm{ray}(\Sigma)\cdot(1+\eta)/(1-\eta)\le K$ from fold
pairing, not Gershgorin.

\section{Premise $(L)$}\label{sec:L}

The von~Mangoldt triangle on positions still has slope
$+0.307\not<1/4$~\cite{r383}.  The new counting is at block
level.  Lemma~\ref{lem:cs} is a theorem, and on the core it is
nearly sharp ($L_1/\sqrt{m D}$ median $0.702$) because
$n_{\mathrm{eff}}\sim m$ is the r301 identity $n_{\mathrm{act}}=m$.
The translation $\gamma_{\mathrm{CS}}=\tfrac12+\tfrac12\mathrm{sl}_D$
equals $+0.214<1/4$ on CORE-$42$ and $+0.192$ on $\chi_3$, but
this is the $D$-decay of the L2 cascade, not an independent
construction bound.  Measured $\mathrm{slope}(L_1)=+0.202$
(CORE-$42$) and $+0.182$ ($\chi_3$) stays below $1/4$ under
weight randomisation; the DC mutant $L_1=m$ has slope $1$ and
kills every field-independent $\gamma<1/4$.

\begin{prop}[$(L)$, reduced]\label{prop:L}
$L_1$ is construction-class: scramble and independent
weight-rand leave $L_1\le 3$ on the pins; the CORE-$42$
log-log slope is $+0.202<1/4$.  Cauchy--Schwarz counting is a
SATZ and is tautological with $n_{\mathrm{eff}}\sim m$.  The
triangle does not close.  The geometry kill is the DC field
$L_1=m$.  A cofinal $\gamma<1/4$ is not proved.
\end{prop}

\noindent
Outcome: \textbf{REDUCED}.  Remainder: a construction bound on
the bulk defect mass (equivalently, cofinal $\gamma<1/4$).

\section{Consequence}\label{sec:C}

\begin{prop}[COMPOSE$^-$ after r391]\label{prop:C}
$(R)$ and $(L)$ are construction-class censuses, not cofinal
SATZ.  COMPOSE$^-$ still requires
\[
  (Z')\ \text{living ladder}\ \wedge\
  M_3\le\phi(m)\ \wedge\
  \mathrm{Dict}\ \wedge\
  \text{the two envelopes $K=4$, $\Lambda=3$ (or the named remainders).}
\]
The r389 Weyl energy owns the \emph{signed} cancellation
objects (assist, $C_\varepsilon$, $Z_{\mathrm{loc}}$), not the
positive quantities $S_F$, $D$, $L_1$.
\end{prop}
