\section{Object}\label{sec:object}

\begin{prop}[$n_{\mathrm{stab}}$-compression]\label{prop:object}
Let $n_{\mathrm{stab}}$ be the r449 Chebyshev
Fenster-Vergleich (skip $m_0,m_1$; first neighbour
break; $\varepsilon$-robust).
The prefix object is the r362/r369 construction
with chain depth $n_{\star}=n_{\mathrm{stab}}$
instead of $N_w-3$:
\[
B_w=\Bigl(\sum_{j=1}^{n_{\star}-1}\rho_j\Bigr)+\frac57,
\]
then $q^{\dagger}$ / $\delta=1-q^{\dagger}$ from
\texttt{solve\_qdag} on that truncated chain plus
the same border augmentation.
Prefix-$80$ was a flat cut; this is the
stabilized depth.
\end{prop}

\section{Living consistency}\label{sec:cons}

\begin{prop}[object split on living windows]\label{prop:split}
On every measured living window the full-window
$\delta$ and the $n_{\mathrm{stab}}$-$\delta$
disagree at relative gap $\ge 0.97$
($k_z5,9,17,26,43,69,116,136$).
The prefix sits higher ($\delta\in[0.103,0.214]$)
than the full living chart
($\delta\in[0.028,0.109]$).
The Chebyshev cut is a moment-agreement depth,
not the ABD-living depth: living windows keep
accumulating $C$-resolvent past $n_{\mathrm{stab}}$,
so $q^{\dagger}$ rises and $\delta$ falls.
This validates that the objects are distinct;
it does \emph{not} validate numerical closeness.
\end{prop}

\section{Floor fits}\label{sec:fit}

\begin{prop}[three charts, M2 on the prefix]\label{prop:fit}
Three transparent fits (r421 \texttt{diagnose\_seq}):
\begin{enumerate}
\item \textbf{Slice only} (full-window $k=5..9$):
winner M1, $\delta_{\infty}=+0.02741$,
$\mathrm{AIC}_1=-56.06 < \mathrm{AIC}_2=-50.95$.
The r445 floor stands on its own chart.
\item \textbf{Mixed} (full-slice $+$ $n_{\mathrm{stab}}$-deep,

winner M2, M1-grid hits $-0.02$,
$\mathrm{AIC}_2=-74.82 < \mathrm{AIC}_1=-72.92$.

\item \textbf{Object-pure} (every $n_{\mathrm{stab}}$-$\delta$,
abscissa $a$):
winner M2, $\mathrm{AIC}_2=-101.17 < \mathrm{AIC}_1=-99.18$.

drop $k_z500$ still M2; deep-only prefers M3.

The prefix-$80$ band $0.09$--$0.13$ was not a

than $0.027$ but still decaying (M2).

\section{$n_{\mathrm{stab}}$ scaling}\label{sec:scale}

\begin{prop}[$n_{\mathrm{stab}}$ grows]\label{prop:scale}
On the deep set $k_z136,137,170,197,230,500$,
\[
n_{\mathrm{stab}}\;\sim\;0.0417\,a^{1.273}.
\]
Absolute depth grows ($175\to 1525$ from
$k_z137\to 500$).
The ratio $n_{\mathrm{stab}}/N$ falls in
$[0.021,0.11]$: the cut is not a fixed fraction
of the window.
The extraction quantifier needs
$n_{\mathrm{stab}}\to\infty$ so that every
finite $\mathrm{meshExp}(f)$ is eventually
inside the prefix after onset $a_0(f)$.
Empirical growth supports that; it is not a
theorem.  Onset $a_0(f)$ remains finite per $f$.
\end{prop}

\section{Dead $\chi$}\label{sec:chi}

\begin{prop}[prefix does not classify $\chi$]\label{prop:chi}
CHI$_3$-$15$ on the \emph{full} window stays
edge-dead: $q^{\dagger}=1.023$, $n_{\mathrm{flip}}=0$.
Its Chebyshev $n_{\mathrm{stab}}$ against
neighbours $14/16$ is $37$.
At $n=40$ (and still at $n=200$) the same
r362/r369 pack is living ($q^{\dagger}=0.863$

Death occurs in the last three grades
($200\to 203$).
The prefix object is the selected-window
extraction cone, not a $\chi$-classifier.
Edge death remains a full-window statement.
\end{prop}
