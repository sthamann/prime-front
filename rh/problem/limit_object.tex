\begin{abstract}\noindent
Round 453 collapsed the cofinal prefix mincut
to two eventual conditions at each fixed $n$:
prefix $h$-positivity through $n$, and
$\rho_n<5/7$.
Both are properties of a \emph{limit object}
once the folded Chebyshev moments $m_k[j]$
converge degreewise (r452 rate $a^{-1.9}$).
This note identifies that object.
For every fixed $j\ge 2$ the MAIN ladder
equals the ARCH fold (von~Mangoldt comb
deleted, same geometry) through $n_{\mathrm{stab}}$,
and has already arrived at a universal
$m_\infty[j]$ on the deep windows.
A PNT-smoothed comb is a nearby classical
proxy (relative $10^{-4}$ to $10^{-7}$);
ARCH agrees at $10^{-15}$.
The limit chain stays $h$-positive with
$\rho_j(\infty)=0\pm 10^{-4}$ through
$j=200$.
No RH claim.  No anti-RH claim.
\end{abstract}

\section{The limit of the moments}\label{sec:A}

Write $m_k[j]=\int T_j\,d(\tilde\mu_k-\tilde\nu_k)$
for the folded signed Chebyshev moments of
window $k$ (r449).
$m_0$ and $m_1$ track the archimedean mass
and have \emph{no} finite limit
($m_0(\mathrm{kz}\,17)=2.599$,
$m_0(\mathrm{kz}\,500)=6.976$).
For $j\ge 2$ the ladder is already universal.

\begin{lemma}[degreewise identification]\label{lem:ident}
On the measured ladder, for every fixed
$j\in\{2,\dots,32\}$ and every window with
$n_{\mathrm{stab}}>j$,
\[
\bigl|m_k[j]-m_{500}[j]\bigr|
\le C_j\,a_k^{-1.9}
\]
with $C_j$ small enough that the deep
windows ($k_z\ge 69$) already sit at
$m_{500}[j]$ to relative $10^{-11}$.
The values
\begin{align*}
m_\infty[2]&=0.1698990367924736,\\
m_\infty[4]&=0.0089525326636440,\\
m_\infty[8]&=0.0010081970691365,\\
m_\infty[16]&=0.0001230324491490,\\
m_\infty[24]&=0.0000362950580291,\\
m_\infty[32]&=0.0000152886541084
\end{align*}
are the kz$500$ pins (live re-gate).
Richardson extrapolation against $a$ is
degenerate: the fit hits the grid edge
because the sequence is already constant.
Dropping kz$500$ (and the two deepest)
leaves $m_2$ of kz$137$ at the same pin
to $10^{-10}$.
\end{lemma}

\begin{lemma}[ARCH kernel]\label{lem:arch}
Let $m_k^{\mathrm{ARCH}}$ be the same fold
applied to the archimedean lags alone
(prime comb deleted).
On kz$17,116,136$ the first relative
$10^{-4}$-break between MAIN and ARCH is
exactly $n_{\mathrm{stab}}$.
Deeper windows agree through the computed
prefix ($k_z=197$: through at least $220$).
Hence, for each fixed $j\ge 2$,
\[
m_\infty[j]
=m^{\mathrm{ARCH}}[j].
\]
The von~Mangoldt comb is in the kernel of
every latched moment.
\end{lemma}

A PNT world --- same tent, same
$(2/L)(1-\cos\theta)$ fold, comb replaced
by $\mathrm{d}\psi(x)\sim e^{u}\,\mathrm{d}u$
--- agrees with MAIN on $j=2,\dots,15$ at
relative $7.7\cdot10^{-6}$ (kz$17$) and
$5.9\cdot10^{-11}$ (kz$137$).
SCRAMBLE is a different test (it breaks
the fold).
The honest identification is ARCH, not
the PNT proxy: ARCH is exact at machine
precision, PNT is a nearby classical
density.
Arithmetic lives in the
\emph{contamination degree} $n_{\mathrm{stab}}$
and in the $a^{-1.9}$ rate, not in
$m_\infty[j]$.

\section{The limit chain}\label{sec:B}

The Stieltjes--Lanczos chain of the
\emph{signed} moment problem cannot be
taken at a single $(m_0,m_1,m_{\ge 2})$
because $m_0,m_1$ have no finite limit.
The load-bearing chain is the $\mu$-chain
plus the border $\rho$-chain of the live
deep windows (already at $m_\infty[j]$
for each fixed $j$).

\begin{lemma}[limit-chain census]\label{lem:chain}
On kz$197$ and kz$500$, for every
$j\in\{20,40,80,160,200\}$,
\[
|\rho_j|+10^{-4}<5/7.
\]
The measured values are
\begin{center}
\begin{tabular}{@{}lrr@{}}
\toprule
$j$ & $\rho_j$ (kz$197$) & $\rho_j$ (kz$500$) \\
\midrule
$20$  & $7.2\cdot10^{-8}$ & $5\cdot10^{-31}$ \\
$40$  & $5.1\cdot10^{-9}$ & $8\cdot10^{-12}$ \\
$80$  & $1.3\cdot10^{-8}$ & $3\cdot10^{-11}$ \\
$160$ & $7.3\cdot10^{-6}$ & $4\cdot10^{-10}$ \\
$200$ & $5.6\cdot10^{-6}$ & $1.2\cdot10^{-6}$ \\
\bottomrule
\end{tabular}
\end{center}
Taking $\rho_j(\infty)=0$ with a sealed
error bar $10^{-4}$ (larger than every
measured $|\rho_j|$), the $5/7$ test
cannot be lifted through $j=200$.
The $\mu$-chain couplings stay positive
through depth $200$ on kz$500$
($h_0>0$, $\min b_i>0$).
f64 already places the deepest $\rho$
at $10^{-6}$; further dps cannot push
a value across $5/7$.
\end{lemma}

The sequence $\rho_j$ on the plateau is
consistent with $0$ (the r452
constancy identity), not with the
on-window $10^{-6}$ noise as a
structural floor.
Trend: no growth toward $5/7$ on the
computed prefix.

\section{The reduced proof program}\label{sec:C}

\begin{lemma}[{[L1] degreewise convergence}]\label{lem:L1}
\textbf{MEASURED.}
$m_k[j]\to m_\infty[j]$ for each fixed
$j\ge 2$, at neighbour rate $a^{-1.9}$
(r452) and with the identification
$m_\infty=m^{\mathrm{ARCH}}$
(Lemma~\ref{lem:arch}).
\end{lemma}

\begin{lemma}[{[L2] continuity of the finite chain}]\label{lem:L2}
\textbf{MEASURED} (numerical Lipschitz).
On a neighbourhood of the kz$17$ moments
in which the $\mu$-chain stays
$h$-positive, the pack map
$m\mapsto(q_n^\dagger,\rho_n)$ is Lipschitz:
a $T_2$-mass perturbation of relative
size $10^{-6},10^{-5},10^{-4}$ gives
\[
\Bigl|\frac{\Delta q}{\Delta m_2}\Bigr|
\in\{0.096,\,0.091,\,0.061\}.
\]
The elementary algebra is the r453
identity $q_n=S_{n-1}/B_w$ together
with the holomorphic dependence of a
finite Jacobi chain on its moment
vector (standard; the Lipschitz
census is the machine check).
\end{lemma}

\begin{lemma}[{[L3] limit-chain property}]\label{lem:L3}
\textbf{MEASURED} for $j\le 200$;
\textbf{OPEN} as a theorem for all $j$.
$h_j(\infty)>0$ and
$\rho_j(\infty)<5/7$ on the computed
prefix (Lemma~\ref{lem:chain}).
\end{lemma}

\begin{thm}[eventual prefix mincut]\label{thm:eventual}
Assume [L1], [L2], and [L3] for a fixed
depth $n$.
Then there exists $K$ such that every
window $k\ge K$ has $n<n_{\mathrm{stab}}(k)$
and $\rho_n(k)<5/7$.
By the r453 identity this is the prefix
mincut at $n$, \emph{eventually}
(stronger than frequently).
The Lean \texttt{Iff.rfl} identification
then gives the full mincut at that $n$.
\end{thm}

Status of the implication: the algebra
from prefix mincut to the two
conditions is \textbf{PROVED} (r453).
The passage ``[L3] for all $j$'' is
\textbf{OPEN}.
The finite-$n$ statement
``eventually, for this $n$'' is
\textbf{REDUCED} to [L1]+[L2]+the
$j=n$ case of [L3], and that case is
\textbf{MEASURED} for every
$n\le 200$.
