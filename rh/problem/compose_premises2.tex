\begin{abstract}\noindent
Round 386 attacks two named remainders of r383.
$(Z')$: $\lvert Z_{\loc}\rvert\le Z_0'<M$ on the
\emph{living} ladder (FRAME-A $89$ $+$ FRAME-B $8$ $+$
$\chi$ with $q_N<1$).  The six r383 $\chi$-violators are
exactly the terminal-dead sprouts $q_N>1$.  The death
channel $\lvert Z_{\loc}\rvert\ge M\iff q_N>1$ holds on
$181/181$ as a biconditional census.  The living supremum is
$0.833870$ at $\chi_4$ kz$46$, which exceeds r383's
$Z_0=4/5$ and forces $Z_0'=21/25=0.84<M$.
Naive finite-head Chebyshev remains refuted
($n_{\mathrm{edge}}$ $167$--$4689$; the edge triangle grows).
Signed cancellation is real (median $0.055$) but
$\mathrm{cancel}_{\max}\cdot\mathrm{tri}_{\max}=4.08>M$, so a
uniform-cancel times triangle does not close.
$(R)$: $B(\omega,\omega)/D\le 3/10$ on $181/181$ (maximum
$0.130$, log-log slope $-1.16$).  Cauchy--Schwarz on the
$r298$ remainder is a theorem ($181/181$), tight at kz$37$.
FAB $m q_{\max}\le 14.93\log m$ is not family-uniform
(FRAME-B FAB $18.07$).  CS$+$FAB therefore does not prove an
$O(\mathrm{polylog})$ bound on $S_F/D$.  The named remainder
is $B(\omega+\beta,\omega+\beta)\le K D$.  $R_0=4$ stays a
census.  No RH claim.
\end{abstract}

\section{Living $(Z')$}\label{sec:Z}

Write $M=\sqrt{5/7}$, $Z=Z_{\loc}+t_{\mathrm{bulk}}$
(r383, Lean \texttt{canonical\_split}), and
$q_N=\rho_{N-1}/(5/7)$.  Under the r263 dictionary
$Z^2=(5/7)q_N$ one has $q_N<1\iff\lvert Z\rvert<M$.

\begin{lemma}[Death triangle]\label{lem:death}
$\lvert Z\rvert\ge\lvert Z_{\loc}\rvert-\lvert t_{\mathrm{bulk}}\rvert$.
If $\lvert Z_{\loc}\rvert\ge M$ and
$\lvert t_{\mathrm{bulk}}\rvert\le\lvert Z_{\loc}\rvert-M$, then
$\lvert Z\rvert\ge M$.
\end{lemma}

\begin{lemma}[Dictionary at rho]\label{lem:dict}
On FRAME-A $w=9$, $\rho_{N-1}=Z^2$ and
$q_N=\rho_{N-1}/(5/7)=(7/5)Z^2$ to relative $10^{-15}$.
\end{lemma}

\begin{prop}[$(Z')$, reduced]\label{prop:Zp}
On the living ladder of $175$ windows,
$\lvert Z_{\loc}\rvert\le 21/25<M$.  The measured supremum is
$0.833870$ at $\chi_4$ kz$46$.  The r383 constant $4/5$
fails on that living $\chi$ window.  The six dead windows
are exactly $\chi_3\{15,19,23,33,39\}$ and $\chi_4\{20\}$,
and on the $181$-surface
\[
  \lvert Z_{\loc}\rvert\ge M
  \iff
  \lvert Z\rvert\ge M
  \iff
  q_N>1.
\]
This biconditional is a census, not a theorem: $t_{\mathrm{bulk}}$
can reduce $\lvert Z\rvert$ relative to $\lvert Z_{\loc}\rvert$
(living $\chi_4$--$46$: $0.834\to 0.713$) but does not pull any
of the six dead rows below $M$.
A PNT-free finite-head Chebyshev bound is refuted
($n_{\mathrm{edge}}$ grows; the unsigned edge triangle has
log-log slope $+0.25$ and range $2.86$--$14.51$).
Signed cancellation is essential (median factor $0.055$) and
is not a uniform SATZ
($\mathrm{cancel}_{\max}\cdot\mathrm{tri}_{\max}>M$).
A cofinal $m_2$ is not proved.  Scramble seed~$1$ at $w=9$
does not break $(Z')$.
\end{prop}

\noindent
Outcome: \textbf{REDUCED} to $Z_0'=21/25$ on the living
ladder, with the death channel a census iff and the source
bound a named remainder (cancellation law, not a triangle).

\section{Cofinal $R_0$}\label{sec:R}

\begin{lemma}[Fej\'er Cauchy--Schwarz]\label{lem:cs}
$B(\Delta,\omega+\beta)^2\le B(\Delta,\Delta)\,B(\omega+\beta,\omega+\beta)$.
On $181/181$ rows the inequality holds; at the $R$-star
kz$37$ it is tight
($T_{\Delta}/D=2.9116$ versus $\sqrt{B_{\Delta\Delta}B_{\Sigma}}/D\approx 2.921$).
\end{lemma}

\begin{prop}[Cofinal $R_0$, reduced]\label{prop:R}
$S_F\le 4\,D$ remains a $181$-census (maximum $2.916788$ at
FRAME-A kz$37$).  The weaker window bound
$B(\omega,\omega)\le(3/10)\,D$ holds on $181/181$ (maximum
$0.129976$, median $0.010$, log-log slope $-1.16$).
A field-independent $C_{\mathrm{MAIN}}\le 1$ is refuted by the
DC toy ($R=\mathrm{MAIN}/D=3.6875$).
CS$+$FAB does not yield $O((\log m)^B)$ on $S_F/D$: the A-law
$m q_{\max}\le 14.93\log m$ fails on FRAME-B (FAB $18.07$),
and FAB controls $P_\Delta$ not $P_\omega$.
The smallest named remainder is
\[
  B(\omega+\beta,\omega+\beta)\le K D
\]
for an absolute $K$ (at the $R$-star one has $B_\Sigma/D=2.975<4$
and $B_{\Delta\Delta}/D=2.869<4$; CS then gives
$\lvert T_\Delta\rvert/D\le 4$, and $S_F/D\le 3/10+4$).
The log-log slope of $R$ on the surface is $-0.328$ (census,
not a theorem).  Scramble seed~$1$ does not break $R_0=4$.
\end{prop}

\noindent
Outcome: \textbf{REDUCED}.  Remaining: the Gram bound on
$B(\omega+\beta,\omega+\beta)/D$ (or an equivalent weight-sum
inequality on the window$+$border sum field).
