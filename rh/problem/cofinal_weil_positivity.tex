the Riemann hypothesis, never a lower-strength statement.  It asks
whether an explicit bordered dual resolvent is at least one half of
the identity for infinitely many indices.  The matrix is built from
the von~Mangoldt comb, the archimedean Weil kernel, one fixed smooth
comparison comb, and the reserve $5/7$.  Round 469 binds the strength
reading: because the compact-window floor $\lambda_*(L)$ is
nonincreasing, positivity on any unbounded cofinal sequence is the
full Weil criterion after the two internal identifications and the
three separately named external bridges.  No such proof is claimed.
What is supplied is an exact finite formulation, a certified finite
census, the r467/r468 anatomy of its correlation wall, and a
scramble filter that rules out whole classes of attacks.  A distinct
fixed-$L$ prize is stated separately; unlike the cofinal problem it
is not RH-equivalent.

\section{The named sequence}

For $k\ge1$ set
\begin{equation}\label{eq:sequence}
 a_k=2^k,\qquad r_k=\lfloor\sqrt{k}\rfloor,\qquad
 m_k=k2^{r_k}-1,\qquad M_k=m_k+1,\qquad
 L_k^{\rm grid}=2M_k-2,
\end{equation}
and
\begin{equation}\label{eq:delta}
 \Delta_k=\frac{\log a_k}{M_k}=2^{-r_k}\log2,\qquad
 N_k=\frac{M_k}{2}=k2^{r_k-1}.
\end{equation}
Here $M_k$ is even for the range of interest.  The folded grid has
$m_k=L_k^{\rm grid}/2$ points
\[
 \theta_{k,j}=\frac{2\pi j}{L_k^{\rm grid}},\qquad
 x_{k,j}=\cos\theta_{k,j},\qquad 1\le j\le m_k .
\]
The last point is its own mirror and receives half the generic fold
weight.

\section{The faithful fold}

Write
\[
 T_\Delta(t)=\max\{0,1-|t|/\Delta\}
\]
and let $\Lambda(n)=\log p$ if $n=p^\ell$ is a prime power, and
$0$ otherwise.  The prime lags are the finite sums
\begin{equation}\label{eq:prime-lag}
 c^P_{k,i}
 =-\sum_{2\le n\le a_k^2}\frac{\Lambda(n)}{\sqrt n}
 \left[
 T_{\Delta_k}(i\Delta_k-\log n)
 +T_{\Delta_k}(i\Delta_k+\log n)
 \right],
 \quad 0\le i<M_k .
\end{equation}
The reflected term is nonzero only when $\log n<\Delta_k$.

For completeness, define the archimedean lag kernel without a
numerical quadrature convention.  Put $\gamma_{\!E}$ for Euler's
constant.  For $s\ge\Delta>0$ let
\begin{equation}\label{eq:arch-far}
 A_\Delta(s)=-
 \int_{s-\Delta}^{s+\Delta}
 T_\Delta(s-w)\frac{e^{-w/2}}{1-e^{-2w}}\,dw .
\end{equation}
For $0\le s<\Delta$, put $t=T_\Delta(s)$,
$S_s(w)=\frac12(T_\Delta(s-w)+T_\Delta(s+w))$, and define
\begin{align}
A_\Delta(s)={}&-(\gamma_{\!E}+\log\pi)t
+2\int_0^{s+\Delta}
\frac{t e^{-2w}-S_s(w)e^{-w/2}}{1-e^{-2w}}\,dw \notag\\
&{}-t\log(1-e^{-2(s+\Delta)}).
\label{eq:arch-near}
\end{align}
The apparent singularity at $w=0$ is removable.  Set
$c^A_{k,i}=A_{\Delta_k}(i\Delta_k)$ and
$c_{k,i}=c^A_{k,i}+c^P_{k,i}$.

The circulant spectral density and signed folded weights are
\begin{align}
d_{k,j}
 &=
 c_{k,0}+2\sum_{i=1}^{M_k-2}c_{k,i}\cos(i\theta_{k,j})
 +c_{k,M_k-1}\cos((M_k-1)\theta_{k,j}),\label{eq:density}\\
w_{k,j}
 &=
\frac{2}{L_k^{\rm grid}}(1-\cos\theta_{k,j})d_{k,j}
\times
\begin{cases}
1,&j<m_k,\\[1mm]
\frac12,&j=m_k.
\end{cases}\label{eq:fold-weight}
\end{align}
Define positive measures on $[-1,1]$ by the sign split
\begin{equation}\label{eq:munu}
 \mu_k=\sum_j\max(w_{k,j},0)\,\delta_{x_{k,j}},
 \qquad
 \nu_k=\sum_j\max(-w_{k,j},0)\,\delta_{x_{k,j}} .
\end{equation}
Equations \eqref{eq:sequence}--\eqref{eq:munu} are the faithful
production order: tents, total lags, circulant density, fold, sign
split.  No atom-table cutoff occurs; all prime powers through
$a_k^2$ are included.

\section{The border, budget, and \texorpdfstring{$R^\dagger$}{R-dagger}}

The border is fixed by the following smooth comparison comb.
Let $\eta=10^{-2}$,
\[
 u_\ell=\ell\eta,\qquad
 m_\ell=2e^{u_\ell/2}\eta,\qquad
 0<u_\ell<2\log a_k .
\]
Replace the prime-power positions and masses in
\eqref{eq:prime-lag} by $(u_\ell,m_\ell)$, equivalently use
\[
 c^{\sigma}_{k,i}
 =-\frac12\sum_\ell m_\ell
 \left[T_{\Delta_k}(i\Delta_k-u_\ell)
      +T_{\Delta_k}(i\Delta_k+u_\ell)\right],
\]
add the same $c^A_{k,i}$, and apply
\eqref{eq:density}--\eqref{eq:fold-weight}.  Retaining the signs
gives a finite signed measure $\sigma_k$.

Assume the moment Gram of $\mu_k$ is positive through degree
$N_k-1$, and let $p_{k,0},\ldots,p_{k,N_k-1}$ be its real
orthonormal polynomials.  Define
\begin{equation}\label{eq:border-budget}
 b_{k,i}=\int p_{k,i}(x)\,d\sigma_k(x),\qquad
 B_k^{\rm bud}=\sum_{i=0}^{N_k-2}b_{k,i}^{\,2}+\frac57,\qquad
 \beta_k=\frac{b_k}{\sqrt{B_k^{\rm bud}}}.
\end{equation}
Thus both numerical conventions in the border are explicit:
$\eta=0.01$ and the reserve is exactly $5/7$.

Here is an intrinsic definition of the dual block.  Let
$X_k=\operatorname{supp}(\mu_k+\nu_k)$,
$P_{X_k}(x)=\prod_{\xi\in X_k}(x-\xi)$, and, for
$\xi\in X_k$, let
\[
 u^\vee_{k,\xi}
 =\frac{1}{(\mu_k+\nu_k)(\{\xi\})\,P'_{X_k}(\xi)^2}.
\]
Let $\Pi^\vee_{k,N_k-1}$ be the dressed Christoffel--Darboux
projection of rank $N_k-1$ for this reciprocal measure, and let
$R_k$ be its restriction to $Y_k=\operatorname{supp}\nu_k$.
For $y\in Y_k$, set
\[
 (v_k)_y=\sqrt{\nu_k(\{y\})}
 \sum_{i=0}^{N_k-1}(\beta_k)_i p_{k,i}(y),\qquad
 (D_k)_{yy}=\operatorname{sgn}P'_{X_k}(y),
 \quad\widetilde v_k=D_kv_k,
\]
and $\gamma_k=\|\beta_k\|^2$.  The bordered dual resolvent is
\begin{equation}\label{eq:rdagger}
 \boxed{\displaystyle
 \Rd(W_k)=
 \begin{pmatrix}
 R_k^{-1}&\widetilde v_k\\
 \widetilde v_k^T&1+\gamma_k
 \end{pmatrix}^{-1}.}
\end{equation}
The definition applies whenever the displayed Gram and inverse
conditions hold.  Their failure at infinitely many indices would
itself refute the requested statement.

\section{The problem in three coordinates}

\begin{problem}[Finite-matrix form]\label{prob:main}
\Ograde{} Prove
\[
\boxed{\qquad
\Rd(W_k)\succeq\frac12 I
\quad\text{for infinitely many }k
\text{ in the sequence \eqref{eq:sequence}.}\qquad}
\]
\end{problem}

This semidefinite statement, rather than a strict asymptotic margin,
is the requested finite-matrix form.  It has two useful equivalent
coordinates.

\paragraph{Classical coordinate.}
For a compactly supported $h\in L^2(\R)$ put
$g=h*\widetilde h$ and $Q_{\rm W}(h)=W(g)$, the additive-log Weil
quadratic form.  Define
\[
 \lambda_*(L)=
 \inf_{\substack{0\ne h\in L^2(\R)\\
 \operatorname{supp}h\subset[-L,L]}}
 \frac{Q_{\rm W}(h)}{\|h\|_2^2}.
\]
The selected support scales are
\begin{equation}\label{eq:support}
 U_k=N_k\Delta_k=\frac{k\log2}{2},\qquad
 L_k=\frac{U_k}{2}=\frac{k\log2}{4}.
\end{equation}
After the internal fold/read and archimedean identifications,
Problem~\ref{prob:main} reads
\[
 \lambda_*(L_k)\ge0
 \quad\text{for an unbounded subsequence.}
\]
Because the support spaces are nested, $\lambda_*(L)$ is
nonincreasing; hence this is cofinal compact-window Weil positivity.

\paragraph{Spectral coordinate.}
Let
\[
 (\mathcal B_k)_{y i}=\sqrt{\nu_k(\{y\})}\,p_{k,i}(y),\qquad
 C_k=\mathcal B_k^T\mathcal B_k,\qquad u_k=\beta_k .
\]
If $C_ke_{k,j}=\lambda_{k,j}e_{k,j}$ and $C_k\prec I$, then
\begin{equation}\label{eq:spectral}
 q_k^\dagger
 :=u_k^T(I-C_k)^{-1}u_k
 =\sum_j\frac{|\langle u_k,e_{k,j}\rangle|^2}
                 {1-\lambda_{k,j}}.
\end{equation}
Finite duality and Schur complementation give
\begin{equation}\label{eq:three-way}
 \Rd(W_k)\succeq\half I
 \quad\Longleftrightarrow\quad
 \begin{pmatrix}I-C_k&u_k\\u_k^T&1\end{pmatrix}\succeq0
 \quad\Longleftrightarrow\quad q_k^\dagger\le1
\end{equation}
under the stated nondegeneracy assumptions.  Thus the sharp internal
task is to bound the spectral measure of the explicit border vector
near the saturating edge $\lambda=1$.
