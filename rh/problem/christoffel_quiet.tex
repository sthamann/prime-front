\title{\bfseries Christoffel quietness, reduced\\[2mm]
\large Chebyshev sampling at trivial Weyl, the Fej\'er
deformation, and why $Q_k<1$ is not the floor plate}
\author{Stefan Hamann}

\begin{abstract}\noindent
Rounds 381 and 382, on the two programme sides, named the same
sampling: the $\mu$-orthogonal polynomial squares against $\nu$.
This note asks whether that quietness
$Q_k:=\langle\pi_k^2\rangle_\nu/h_k(\mu)\le\bar q<1$
is a theorem of the fold geometry (the $3/8$-floor, F1 run-lengths,
and the cosine grid), not a measurement.
What is proved: Chebyshev--Gauss exactness on the equal-weight
cosine grid; the polynomial identity $T_k^2=(1+T_{2k})/2$;
the first-order pairing
$\mathrm{FO}_k[-\nu]=\gamma_k(Q_k-Q_{k-1})$;
the one-Rayleigh identity
$\langle p_k,p_k\rangle_{\mu-\nu}=h_k(\mu)\,(1-Q_k)$;
and that the Chebyshev sampling of $\nu$ stays at the mass
ratio $\alpha$ under the \emph{trivial} Weyl bound
$\lvert S_m\rvert\le\mathrm{mass}(\nu)$.
What is \emph{not} proved: the deduction from that geometry to
the $\mu$-OP profile $Q_k$.  On FRAME-A $w=9$, the Chebyshev
predictor stays at $\alpha=0.061$ while the true $Q_k$ grows to
$0.389$; the Koksma--Hlawka majorant $kD^*$ is $33$ at last-twelve
and does not close.  The two-period adversary of amplitude $2/3$
has $Q_k\equiv 0.683<1$ and still crosses $\lambda_{\max}(E_{22})>1$.
The hydra-legal remainder is the pair $(\Delta, C)$: the Fej\'er$/d_{\mathrm{arm}}$
deformation $\Delta_k=Q_k-Q_k^T$ (the live term of $C_\varepsilon$),
and the coherence factor $\lambda_{\max}=\max Q\cdot(1+\mathrm{assist})$
(the $L^2$ remainder of~\cite{r382}).
The Weyl scale of the original candidate is trivial, not PNT,
not subconvex, and not RH-equivalent: the Chebyshev--Weyl route
never reaches a zeta-quality demand.
No RH claim.
\end{abstract}

\section{The object}\label{sec:obj}

Write $\pi_k$ for the $\mu$-orthonormal polynomials,
$h_k(\mu)=\int\pi_k^2\,d\mu=1$, and
\begin{equation}\label{eq:Qk}
Q_k
:=
\langle\pi_k^2\rangle_\nu
=
\sum_j v_j\,\pi_k(y_j)^2.
\end{equation}
The same ratio with monic polynomials is identical
(machine check: FRAME-A $w=9$, maxdev $5\cdot10^{-15}$).
Round 381 records last-twelve $Q\in(0.35,0.47)$ on MAIN and
$Q>10^3$ on a scramble; the pairing $C_\varepsilon$ is
$\lvert\mathrm{FO}_k[-\nu]\rvert\le\tfrac1{32}$ on the last twelve.
Round 382 records the Christoffel comparison
$h_k(w)\ge(1-\lambda_{\max}(E_{k+1}))h_k(\mu)$ and names the
$L^2$ tail on the flank blocks as the remainder for $n_0=N-1$.
The mesh $x=\cos(2\pi j/L)$, $L=4h-2$, is a construction identity
of~\cite{v3p}.

The candidate lemma of this round: a geometric bound
$Q_k\le\bar q<1$, with $\bar q$ read from the $3/8$-floor and
the cosine-grid equidistribution of $\nu$-angles, would feed both
consequences (a) a one-Rayleigh floor plate and (b) $C_\varepsilon$.

\section{What the grid sees}\label{sec:grid}

\begin{lemma}[$T_k^2$]\label{lem:t2}
The Chebyshev polynomials satisfy $T_k^2=(1+T_{2k})/2$ as an
identity in $\Q[x]$.  In angular coordinates
$T_k(\cos\theta)=\cos(k\theta)$, so $\pi_k^2$ of the Chebyshev
reference is a two-mode trigonometric polynomial (frequencies
$0$ and $2k$).
\end{lemma}

\begin{lemma}[first-order pairing]\label{lem:fo}
On a quasi-definite atomic $\mu$, writing $\gamma_k=h_k/h_{k-1}$
for the monic Jacobi coefficients,
\[
\mathrm{FO}_k[-\nu]
=
\gamma_k\bigl(Q_k-Q_{k-1}\bigr).
\]
Exact over $\Q$ on a finite node set; FRAME-A $w=9$, first twelve,
floating-point residual $2\cdot10^{-17}$.
\end{lemma}

\begin{lemma}[one Rayleigh]\label{lem:ray}
For the $\mu$-orthogonal polynomial of degree $k$,
\[
\langle p_k,p_k\rangle_{\mu-\nu}
=
h_k(\mu)\,(1-Q_k).
\]
This is \emph{not} $h_k(\mu-\nu)$ and is \emph{not}
$\lambda_{\max}(E_{k+1})<1$.  It is the quadratic form of one
vector.  On the three-atom $(3,-2,3)$: $Q_0=\tfrac12$ and
$h_0(1-Q_0)=2$.
\end{lemma}

\begin{lemma}[Chebyshev--Gauss]\label{lem:cheb}
On the equal-weight cosine grid $x_j=\cos(2\pi j/L)$ of length
$n-1$ with $L=2n$, the $\mu$-orthonormal sampling of the
\emph{full} grid is $Q_k\equiv 1$ for every $k<n/2$
(maxdev $4\cdot10^{-16}$ at $n=32$).  Consequently the
Chebyshev partial sum of a subset is a mass-weighted Fourier
coefficient of occupancy against the two-mode density of
Lemma~\ref{lem:t2}.
\end{lemma}

\begin{lemma}[total variation]\label{lem:tv}
$\mathrm{TV}(\cos(2k\theta),[0,\pi])=4k$.  The Koksma--Hlawka /
Erd\H{o}s--Tur\'an majorant of the Chebyshev sampling error is
therefore $O(k D^*)$ on the circle.
\end{lemma}

\section{The geometric candidate, measured}\label{sec:geom}

Write $\alpha=\mathrm{mass}(\nu)/\mathrm{mass}(\mu)$ and
$Q_k^T$ for the Chebyshev predictor
$\sum_\nu v(1+\cos 2k\theta)/\sum_\mu w(1+\cos 2k\theta)$.
Write $D^*$ for the star discrepancy of the $\nu$-angles in
$[0,\pi]$ (mapped to $[0,1]$), and
$S_m=\sum_j v_j e^{im\theta_j}$.

\begin{prop}[census]\label{prop:census}
On the sealed constructors:
\begin{enumerate}
\item FRAME-A $w=9$: $\alpha=0.0611$, node-fraction $0.2834$,
  $D^*=0.1832$, last-twelve $Q=0.3891$, last-twelve
  $\lvert\mathrm{FO}\rvert=0.00978<\tfrac1{32}$,
  $\max\lvert Q-Q^T\rvert=0.335$, $kD^*=33.4$ at $k=182$,
  $\lvert Q-\alpha\rvert/(kD^*)=0.0089$.
\item Chebyshev predictor $Q^T$ stays in $(0.055,0.079)$
  through the last twelve --- at $\alpha$, not at $Q$.
\item Core-$42$: $Q_{\max}\in[0.164,0.483]<1$ (depth $N-1$ for
  $N\le 220$, else $80$); eleven small-$N$ windows have
  last-twelve $\lvert\mathrm{FO}\rvert\le 0.01182<\tfrac1{32}$.
\item MAIN-$85$ star discrepancy $D^*\in[0.105,0.212]$;
  $\chi_3$-$42$ $D^*\in[0.125,0.248]$.
  (The $181$-pack builder of~\cite{v3p} was not rerun: it is too
  expensive for a discrepancy-only census, disclosed.  The two
  programme surfaces MAIN and $\chi_3$ already show $D^*=\Theta(1)$.)
\end{enumerate}
The Koksma candidate $Q_k\le\alpha+O(k D^*)$ does not close:
the majorant exceeds $1$ already at $k\sim 10$.  The effective
ratio $\lvert Q-\alpha\rvert/(kD^*)$ is $o(1)$, so the coupling
is better than generic Koksma, but the error $Q-\alpha$ itself
still grows to $0.30$.  That growth is the $\mu$-OP deformation
off Chebyshev, not a discrepancy of $\nu$-angles.
\end{prop}

The node-fraction $0.28$ versus mass-ratio $0.06$ is the Fej\'er
factor $1-x=2\sin^2(\theta/2)$: $\nu$ occupies a quarter of the
nodes but a thinner share of the $(1-x)$-weighted mass.  The
$3/8$-floor of~\cite{r361} controls gaps, not OP squares.

\section{Weyl scale}\label{sec:weyl}

\begin{prop}[scale]\label{prop:weyl}
Let $S_m$ be the $\nu$-weighted exponential sum on the cosine
angles.
\begin{enumerate}
\item \emph{Trivial.}  $\lvert S_m\rvert\le\mathrm{mass}(\nu)$.
  For the Chebyshev predictor this yields
  $Q_k^T\le 2\alpha\sim 0.12<1$ uniformly in $k$.
  Measured $\lvert S_{2k}\rvert/\mathrm{mass}(\nu)\in(0.04,0.33)$
  at $k\ge 1$ on $w=9$: near-trivial, no extra cancellation.
\item \emph{PNT / subconvex / RH.}  Not demanded.  The
  Chebyshev--Weyl route is already closed by~(i) and still
  does not yield the $\mu$-OP lemma, because $Q_k\neq Q_k^T$.
\item \emph{What would close $C_\varepsilon$.}
  Lemma~\ref{lem:fo} needs $\lvert\gamma\Delta Q\rvert\le\tfrac1{32}$,
  i.e.\ consecutive differences of $Q$, not a uniform bound on
  $S_{2k}$.  On $w=9$ the last-twelve $\lvert\mathrm{FO}\rvert=0.0098$
  is a smoothness fact of the discrete $\mu$-chain, census-grade.
\item \emph{What would close $n_0=N-1$.}
  Even the true $Q_k\le 0.48$ (core census) is insufficient:
  $\lambda_{\max}(E_k)=\max Q\cdot(1+\mathrm{assist})$ with
  assist $1.5$--$4$ on $w=9$ (at $k=183$,
  $\lambda_{\max}=0.9998$, $\max Q=0.393$, assist $1.54$).
  The floor plate is the Gram $\lambda_{\max}<1$ of~\cite{r382},
  not the diagonal $Q_k<1$.
\end{enumerate}
The circularity threat --- that a needed Weyl bound might be
RH-equivalent --- does not arise.  The honest position of the
lemma on the scale trivial${}\leftrightarrow{}$RH is
\textbf{trivial}.
\end{prop}

The better-than-Koksma coupling of Leg~B is the two-mode
Fourier of Lemma~\ref{lem:t2} (error $=O(\lvert S_{2k}\rvert)$,
not a sum of $k$ frequencies) together with the fact that the
\emph{live} error is not that Fourier term at all.  No
explicit-formula input is consumed; the firewall of the probe
forbids zero and prime oracles.

\section{Adversaries}\label{sec:adv}

\paragraph{Scramble.}
FRAME-A $w=9$, seed $1$: $Q_k$ crosses $1$ at $k=70$; last-twelve
$Q=4.8\cdot10^6$.  This is the r381 named break, reproduced.
Low-degree discrepancy is comparable to MAIN ($D^*\sim 0.24$
versus $0.18$ at depth $40$).  Scramble does not break the
angle-discrepancy first; it breaks the $\mu$-OP sampling at
high degree.

\paragraph{Two-period, $c=2/3$.}
On $S=81$, $Q_k\equiv\alpha=0.683<1$ for every $k$ (Chebyshev
even/odd: Lemma~\ref{lem:cheb} plus equal occupancy).
$\lambda_{\max}(E_{22})=1.029>1$.  Amplitude $c=1$ has $Q_0=1.025>1$
and kills at $h_0$.  The $Q$-chain does not see the binding
half-filling adversary of~\cite{r382}.  Consequence~(a) of the
candidate lemma is false as a sufficient condition.

\paragraph{EXT-heavy seven.}
Windows $\{69,96,97,99,107,117,129\}$ of~\cite{r382}, where
$r_{\max}$ reaches $0.879$, have $Q_{\max}\in[0.127,0.221]$ at
depth $200$ (all $<0.25$).  Quietness of $Q$ holds at that
depth; the flank-mass ratio is a different object.

\section{The reduction}\label{sec:red}

\begin{prop}[reduction]\label{prop:red}
The candidate ``$Q_k\le\bar q<1$ from fold geometry'' is
strictly larger than the pair $(\Delta,C)$:
\begin{enumerate}
\item \emph{Chebyshev sampling of $\nu$.}  SATZ at trivial Weyl
  (Lemmas~\ref{lem:t2}--\ref{lem:cheb} and
  Proposition~\ref{prop:weyl}(i)).  $Q_k^T\le 2\alpha<1$.
\item \emph{Deformation}
  $\Delta_k:=Q_k-Q_k^T$
  of the $\mu$-chain off Chebyshev (Fej\'er Jacobi-$(0,1)$ plus
  $d_{\mathrm{arm}}$).  This is the live term of $C_\varepsilon$
  via Lemma~\ref{lem:fo}.  Census: last-twelve
  $\lvert\mathrm{FO}\rvert\le 0.01182$ on eleven small-$N$
  core windows; not a cofinal law.
\item \emph{Coherence}
  $\lambda_{\max}(E_k)=\max_j Q_j\cdot(1+\mathrm{assist})$.
  Assist is the off-diagonal of the $\nu$-Gram in the $\mu$-OP
  basis --- the r285/r382 remainder.  Two-period $c=2/3$ is the
  named witness that $Q<1$ does not control $\lambda_{\max}$.
\end{enumerate}
The original lemma is therefore reduced, not proved and not
vacuously refuted: (i) is a theorem, (ii) and (iii) are strictly
smaller named remainders, one on each programme side
($C_\varepsilon$ and $n_0=N-1$).
\end{prop}

The chain of~\cite{v3p,r381,r382} stands with this split in
place of a single geometric $Q$-bound.  T1 / COMPOSE unchanged.
No RH claim.

\section*{Claim boundary}

Finite identities for Chebyshev sampling and Jacobi pairing on
a fixed node set, a named reduction of the geometric
quietness candidate to $(\Delta,C)$, a scramble break, and a
two-period witness that $Q<1$ is not the floor plate.
No statement about zeros of $\zeta(s)$, in either direction.
The Weyl-scale adjudication is that \emph{no} zeta-quality
bound is consumed or required.
No RH claim.
