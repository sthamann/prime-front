\begin{abstract}\noindent
The mixed form of round 369 writes
$M^{\dagger}=R^{\dagger}-\half I=A^{\dagger}+UJU^{T}$
with an explicit $3\times 3$ signature $J=\operatorname{diag}(1,1,1/\mathrm{den})$
and $U$ the last two dual Christoffel--Darboux columns together
with the Sherman--Morrison border.
Haynsworth (proved in \texttt{RH/Haynsworth.lean}) converts
positivity of $M^{\dagger}$ into a comparison of
$\indm\Phi$ with $\indm A^{\dagger}$,
$\Phi=J^{-1}+U^{T}(A^{\dagger})^{-1}U$.
The $3\times 3$ $\Phi$ is not a restatement of that identity:
after an explicit Sylvester map $D_{k}$ on the $K_{2}$ block
and a scale $d_{3}=1/\sqrt{|\mathrm{sch}|}$ it equals a
two-parameter family $\Phi_{\mathrm{edge}}(a,b)$ whose
inertia is constant on each chart, whose spectral gap to $0$
is uniformly positive on any bounded set of $(a,b)$, and
whose reconstruction error is identically zero.
On the named census the couplings $(a,b)$ stay in a compact
set with $g^{*}\ge 0.08$; dead $\chi$ leave the living chart
by $\mathrm{sch}>0$.
No RH claim.
\end{abstract}

\section{The $3\times 3$ object}\label{sec:obj}

Write $A_{0}=R_{N-3}-\half I$ on the hole nodes $Y$,
$U_{\mathrm{CD}}$ the last two dual CD columns, and
$A^{\dagger}=\operatorname{blkdiag}(A_{0},-\half)$.
Sherman--Morrison on the bordered dual resolvent supplies a
third column $w=(s;-1)$ and $\mathrm{den}=1+\gamma-v^{T}Rs$,
with $J=\operatorname{diag}(1,1,1/\mathrm{den})$.
Then~\cite{r369}
\[
\Phi
\;=\;
\begin{pmatrix} K_{2} & c_{0}\\ c_{0}^{T} & \varphi_{bb}\end{pmatrix},
\qquad
K_{2}=I+U_{\mathrm{CD}}^{T}A_{0}^{-1}U_{\mathrm{CD}},
\]
and $K_{2}=(I+R_{+})-ww^{T}/\lambda_{-}$ on the $\mathrm{nneg}(A_{0})=1$
branch, with
$\det K_{2}=\det(I+R_{+})\,(1-\gamma/\lambda_{-})$~\cite{r375}.
The Schur complement of $K_{2}$ in $\Phi$ is the scalar
\[
\mathrm{sch}
\;=\;
\varphi_{bb}-c_{0}^{T}K_{2}^{-1}c_{0}.
\]
Haynsworth on $\Phi$ itself reads
$\indm\Phi=\indm K_{2}+\indm(\mathrm{sch})$.
On the named construction $\indm K_{2}=\indm A_{0}$, so
\begin{equation}\label{eq:shift}
\indm\Phi
\;=\;
1+\indm A_{0}
\qquad\Longleftrightarrow\qquad
\mathrm{sch}<0.
\end{equation}
The $+1$ is the border mode.  It is not the macroscopic
$\tfrac12$-cluster of $R_{N-3}$ (round 398): that cluster sits
in $A_{0}$ from above and is bulk.

\section{Explicit $D$ and the model family}\label{sec:D}

Let $K_{2}=V\operatorname{diag}(\mu)V^{T}$ be the $2\times 2$
spectral theorem and $D_{2}=V\operatorname{diag}(1/\sqrt{|\mu|})$.
Set $d_{3}=1/\sqrt{|\mathrm{sch}|}$ on $\mathrm{sch}\neq 0$ and
$D=\operatorname{blkdiag}(D_{2},d_{3})$.
Then $D^{T}K_{2}D$ in the leading $2\times 2$ is
$J_{2}=\operatorname{diag}(\operatorname{sign}\mu)$, and the
off-block of $D^{T}\Phi D$ is a pair $(a,b)\in\R^{2}$.

\begin{lemma}[reconstruction]\label{lem:recon}
Let $J_{2}$ be diagonal with entries $\pm 1$, let
$\tau=(a,b)^{T}$, and let $s=\operatorname{sign}(\mathrm{sch})$.
The unique symmetric $3\times 3$ with leading block $J_{2}$,
off-block $\tau$ and Schur complement $s$ is
\begin{equation}\label{eq:edge}
\Phi_{\mathrm{edge}}(\tau)
\;=\;
\begin{pmatrix}
J_{2} & \tau\\
\tau^{T} & s+\tau^{T}J_{2}^{-1}\tau
\end{pmatrix}.
\end{equation}
Hence $D^{T}\Phi D=\Phi_{\mathrm{edge}}(a,b)$ identically
(the error operator is zero).
\end{lemma}

This is not a fit and not a fixed matrix $M_{0}$.
The family wanders with $\tau=(a,b)$.

\begin{lemma}[model lemma]\label{lem:model}
On the P1 chart $J_{2}=\operatorname{diag}(-1,1)$, $s=-1$,
\[
\indm\Phi_{\mathrm{edge}}(a,b)
\;=\;
2
\;=\;
1+\indm A_{\mathrm{edge}}
\]
for every $(a,b)\in\R^{2}$, where $A_{\mathrm{edge}}=(-1)$
records the single negative of $K_{2}$.
On the vacuous chart $J_{2}=I_{2}$, $s=-1$,
$\indm\Phi_{\mathrm{edge}}=1=1+0$.
In both charts $\det\Phi_{\mathrm{edge}}=\det(J_{2})\,s=\pm 1\neq 0$.
\end{lemma}

\begin{proof}
Haynsworth additivity on~\eqref{eq:edge}:
$\indm\Phi_{\mathrm{edge}}=\indm J_{2}+\indm(s)$.
The P1 chart contributes $(1)+(1)$; the vacuous chart $(0)+(1)$.
The determinant identity is the Schur formula.
On the rational point $(a,b)=(0,0)$ one has
$\Phi_{\mathrm{edge}}=\operatorname{diag}(-1,1,-1)$ with inertia
$(1,2,0)$, and $\Phi_{\mathrm{edge}}(1,0)$ has determinant $1$.
\end{proof}

The gap $g(\tau)=\mathrm{dist}(0,\sigma(\Phi_{\mathrm{edge}}))$
is positive on any bounded $K\subset\R^{2}$.
Along the $a$-axis,
$g(a,0)=(2+a^{2}-|a|\sqrt{a^{2}+4})/2\sim 1/a^{2}$.
A $64$-point check on the circle of radius $3.2$ gives
$g^{*}\ge 0.082$ and inertia $2$ at every sample.

\section{The coordinate $\tau$}\label{sec:tau}

Three candidates were tested, in the order asked.

\textit{\textup{(i)} The r375 pair.}
$\rho=\gamma/\lambda_{-}$ and $\delta=\det(I+R_{+})$,
projectively $\xi=(\rho-1)/(\rho+1)$, $\eta=\delta/(1+\delta)$.
On the $\mathrm{nneg}=1$ branch of core-$42$,
$\xi\in[0.121,0.622]$, $\eta\in[0.768,0.991]$,
$\mathrm{den}\in[1.46,1.63]$, halves-medians trend-free.
An isotropic reconstruction of $\Phi$ from $(\rho,\delta)$
alone has $\|E\|/g=80.7$ at FRAME-A: the pair does not
determine the $3\times 3$.
Spearman of $|(a,b)|$ against $\xi$ is $0.12$.

\textit{\textup{(ii)} The r321 sliding coordinate.}
A QMAX proxy of the CD columns has Spearman $0.26$ against
$|(a,b)|$ and Spearman $-0.96$ against $N$ (it shrinks).
It is not the $3\times 3$ modulus.

\textit{\textup{(iii)} The two signature couplings.}
$\tau=(a,b)$ as in \S\ref{sec:D}.
On core-$42$ (P1), $|(a,b)|\le 3.144$; the disk of radius $3.2$
is a compact $K$ with $g^{*}\ge 0.08$.
Large-$N$ EXT windows have \emph{smaller} $|(a,b)|$
($0.075$ at $N=1119$, $0.203$ at $N=3022$).
The set $K$ is stable in $N$ (r360-conform: $\tau$ may wander
inside a fixed compact).

\section{Census and two-sidedness}\label{sec:census}

\begin{prop}[named census]\label{prop:census}
On core-$42$, a six-row EXT sample, Twin $kz\in\{9,18,52\}$,
living $\chi_{3}$ ($37$) and living $\chi_{4}$ ($41$):
$J\succ 0$, Haynsworth balance,~\eqref{eq:shift}, and
$\|E\|\le 10^{-10}$.
Dead $\chi_{3}$ ($5/5$) and dead $\chi_{4}$-$20$ have
$\mathrm{sch}>0$ and violate~\eqref{eq:shift}.
Scramble has $\indm A_{0}=21$ and the border pack dies at step $37$.
Two-period $S=21$ has $\indm A_{0}=4$.
Dropping the border column yields shift $0$ already on the
rational $4$-node model.
\end{prop}

The reconstruction error is a SATZ (Lemma~\ref{lem:recon}),
not a shrinking remainder.
Cofinality of $K$ is not proved.

\section{Named kills}\label{sec:kills}

Dead $\chi$: $\mathrm{sch}>0$, so the border mode of $\Phi$ is
positive --- the shift fails, or $\indm\Phi=0$ (dead $\chi_{4}$-$20$).
Several dead rows have $|(a,b)|$ up to $6.7$, outside living $K$.
Scramble: $\indm A_{0}=21$ leaves the edge chart $\{0,1\}$;
the SM pack is not positive-prefix.
Two-period: $\indm A_{0}=4$.
Mutants: omitting the border column kills the $+1$;
the wrong-sign $J_{33}$ is already caught in Fractions by
round 369 (residual $867/115$).

\section*{Claim boundary}

Finite identities for the mixed form, for Haynsworth, for the
$3\times 3$ reconstruction and for the model lemma on both charts;
a named census that the living worlds land in a compact $K$
with $g^{*}>0$ and that dead $\chi$ leave by $\mathrm{sch}>0$.
No statement about zeros of $\zeta(s)$, in either direction.
No RH claim.
