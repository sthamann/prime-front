\begin{abstract}\noindent
Round 480 reduced $\lambda_*(0.3)\ge c$ to positivity of the
even $3\times3$ matrix Schur
$S=B-CC^\top/c_{\mathrm{tail}}$ on
$\mathrm{span}\{\psi_0,\psi_2,\psi_4\}$
(numeric $\lambda_{\min}(S)\approx5.69\cdot10^{-3}$)
and an easier odd partner.
This round tries to enclose $S$.
Route~(b) with the integer Fourier ONB is killed:
$Q(u_0,u_n)\sim1/n$, and the infinite tail of $C$ eats the
midpoint Schur.
The identity
$\|Au\|_{L^2(\mathbb R)}^2=(2\pi)^{-1}\int\sigma_A^2|\hat u|^2$
is a false formula for $\|Au\|_{L^2[-L,L]}$ and overestimates
$CC^*$ by an order of magnitude.
Even Legendre polynomials have an explicit spherical Bessel
Fourier transform and a tiny $\varepsilon$, but the same
completeness leak as $M$ grows.
The working basis is the Nystr\"om--Slepian interpolant.
The kernel is entire: Gauss--Legendre $n=48$ interpolation
remainder is $<10^{-80}$, discrete residuals are $10^{-16}$,
so $\varepsilon=\lambda_6(K_8)\le1.46\cdot10^{-8}$ and
$c_{\mathrm{tail}}\ge0.2408$ lift from the discrete matrix
(PROVED).
The remaining integrals are
$\int_0^T\sigma\hat\psi_i\hat\psi_j$
(oscillatory, working window $T\lesssim200$ at $n=64$)
and the $[T,\infty)$ integration-by-parts tail.
The crude envelope $|\hat\psi|\le2|\psi(L)|/t$ produces an
$A$-tail $\sim0.76$ on mode~$4$ at $T=150$, larger than the
Schur margin, so it is not a pad.
True windows through $T=320$ stay PD
($\lambda_{\min}(S)\approx7.1\cdot10^{-3}$)
but are not an interval hull.
Verdict \texttt{REDUCED(interval-FT+IBP-tail)}.
Fourier--HS is not reused.
Yoshida--Bombieri zone.  No RH claim.
\end{abstract}

\section{Convention}

$\mathrm{supp}\,h\subset[-L,L]$, $g=h*\widetilde h$,
$\QW=A-P+\Pi$, prime nodes $n\le e^{2L}$.
At $L=0.3$ one has $P\equiv0$.
Plancherel $\hat h(t)=\int h(x)e^{itx}\,dx$,
$\|h\|_2^2=(2\pi)^{-1}\int|\hat h|^2$.
The r479/r480 contract is in force:
$t_*=8$, $c_{\mathrm{tail}}$, $\kappa_{\mathrm{low}}$,
and the named operator-norm bound is already false.

\section{Basis choice}

The tail floor
$A\succeq\sigma_A(t_*)I+(\sigma_A(0)-\sigma_A(t_*))K_{t_*}$
is operator-wide.
On $\mathrm{ran}(1-P)$ it gives
$c_{\mathrm{tail}}=\sigma(t_*)(1-\varepsilon)+\sigma(0)\varepsilon$
with $\varepsilon=\|(1-P)K(1-P)\|$, a single Rayleigh number
for whatever $3$-dimensional $P$ one chooses.

\paragraph{Fourier ONB (killed).}
$V=\mathrm{span}\{1,\cos(\pi x/L),\dots,\cos(4\pi x/L)\}$
has $\varepsilon\approx1.3\cdot10^{-4}$ and a certified
$5\times5$ Gram (r478, Higham $\mu\ge4.06\cdot10^{-3}$).
The off-block
$Q(u_0,u_n)$ decays like $1/n$
($0.321,0.161,0.111,0.078,0.066$ for $n=1..5$).
At truncation $M=10$ one has $\lambda_{\min}(S)\approx1.3\cdot10^{-4}$;
the leftover Hilbert--Schmidt mass for $n\ge10$ is $\sim0.08$
and kills the Schur.
This is the r478 obstruction in refined clothing.

\paragraph{$\sigma^2$-Plancherel (killed).}
$\|M_\sigma u\|_{L^2(\mathbb R)}^2=(2\pi)^{-1}\int\sigma^2|\hat u|^2$
majorizes $\|Au\|_{L^2[-L,L]}^2$ but is not equal to it
($M_\sigma u$ is not time-limited).
Used as an exact $CC^*$ it produces
$\lambda_{\min}(S)\approx-10$.

\paragraph{Even Legendre (not closed).}
$P_0,P_2,P_4$ have Fourier transforms
$2Li^n j_n(tL)$ (spherical Bessel).
$\varepsilon\approx2\cdot10^{-6}$.
As the complementary degree $M$ grows,
$\|C\|$ creeps ($0.31\to0.55$ from $M=6$ to $M=20$)
and $\lambda_{\min}(S)$ hovers near $4\cdot10^{-3}$
without a certified infinite-$C$ bound.

\paragraph{Nystr\"om--Slepian interpolant (working).}
The first three even prolates are the unique small space
that is almost invariant under $K_8$, so $C$ lives in one
or two neighbouring modes ($\psi_6,\psi_8$) and the
$M$-truncation of r480 is honest.
The interpolant through the Gauss nodes, extended by
$\widetilde\psi(x)=\lambda^{-1}\sum_j w_j K(x,x_j)\psi(x_j)$,
is an explicit combination of sincs.

\section{What is proved: the lift of $\varepsilon$}

\begin{lemma}[Kernel interpolation]
The sinc kernel $K(x,y)=\sin(t_*(x-y))/(\pi(x-y))$
is entire in each variable.
The Gauss--Legendre remainder prefactor on $[-L,L]$ at $n=48$
is $7.59\cdot10^{-230}$.
Times $t_*^{96}$ this is $<10^{-80}$.
\end{lemma}

\begin{lemma}[Discrete residuals]
The weighted Nystr\"om eigenproblem at $n=48$ has
node-wise $L^2$ residuals $<10^{-15}$ on modes $0..6$.
Weyl plus the interpolation remainder therefore lifts
\[
 \varepsilon=\lambda_6(K_8)\le1.46\cdot10^{-8},
 \qquad
 c_{\mathrm{tail}}\ge0.2408.
\]
Endpoint jets $\psi(\pm L)$ of modes $0..4$ agree at
$n=48$ versus $n=64$ to $10^{-12}$.
\end{lemma}

\section{What is not proved: the tail integral}

Double integration-by-parts gives
\[
 \hat\psi(t)
 =\frac{2\psi(L)\sin(tL)}{t}
 -\frac{2\psi'(L)\cos(tL)}{t^2}
 +O(t^{-3}),
\]
because $\psi(\pm L)\ne0$.
The envelope $|\hat\psi|\le2|\psi(L)|/t$ produces
\[
 \int_{150}^\infty\frac{(1+\log t)\,|\widehat{\psi}_4|^2}{\pi}\,dt
 \;\sim\;0.76,
\]
larger than $\lambda_{\min}(S)/3$.
It cannot be used as a pad.

True increments at $n=96$ (not a hull):
$A[80,160]$ adds $0.163$ to $A_{44}$ and $0.187$ to $A_{46}$;
windows through $T=320$ keep
$\lambda_{\min}(S)\approx7.1\cdot10^{-3}$.
Gauss-in-$x$ dies past $T\sim200$ at $n=64$
(the ratio $|\hat\psi|/(2|\psi(L)|/t)$ jumps from $0.9$ to $5$).

The remaining object is the oscillatory integral
$\int_T^\infty\sigma_A(t)\hat\psi_i(t)\hat\psi_j(t)\,dt$
with a Filon or cosine-integral remainder small enough
to leave a third of the margin.
Four even block modes instead of three does not remove it
($\lambda_{\min}(S)$ stays $\approx6.8\cdot10^{-3}$ at $T=240$).
