\begin{abstract}\noindent
Rounds 404 and 405 compared Euler/Digamma Grams to
$A_0=R_{N-3}-\half I$ without a dual quotient
(residual of order $209$).
Round 407 wrote the dictionary $R=C(I+C)^{-1}$ with
$C=\Tfrak^*\Tfrak$ in the dual orthogonal-polynomial basis
(FRAME-A residual $1.8\cdot 10^{-14}$).
This note lifts that dictionary to a Jacobi-free source
formula for the operator $\Tfrak_0$ itself: the min-norm
interpolant $Y\to X$ in $\mathcal{P}_{<d_0}$, $d_0=N_w-3$.
Then $R_{\mathrm{graph}}=(I+\Tfrak_0^*\Tfrak_0)^{-1}$ equals
the r356 dual resolvent, and
$\indm(I-\Tfrak_0^*\Tfrak_0)=\indm A_0=\indm\cD_0$.
The r369 Haynsworth matrix $\Phi_N=J^{-1}+U^TA_0^{-1}U$
\emph{is} the Woodbury precision template (r406 Lean
\texttt{woodbury\_inv}); it is \emph{not} the pair-plus-border
Schur of the augmented Gram, so the candidate
$\Theta_{\mathrm{terminal}}=C^T\Phi_N C$ is
\textbf{refuted} as a literal matrix identity.
Sequential Uvarov innovations satisfy $n_{\mathrm{neg}}^{\mathrm{inn}}\le 1$
on core-$42$ (census, $42/42$), not a theorem, and do not
identically equal $\indm A_0$ ($32/42$).
The unaugmented operator on six terminal-dead $\chi$
windows stays near contractive; the r362 graph factor
$\Tfrak^\dagger$ of the genuine border functional flips all
six ($\|\Tfrak^\dagger\|>1$).
That last identity uses the r362 Gram: it is a dictionary,
not a new theorem of $R^\dagger$.
No RH claim.
\end{abstract}

\section{The source formula}\label{sec:formula}

Write $u$ for the folded signed union weight on the cosine
grid, $Y=\operatorname{supp}(u_-)$ the hole nodes, $X$ the
$\mu$-nodes, and $u^\vee_j=1/(u_j P_{\Xi'}^2(x_j))$ the
Borodin dual (r356; $P_{\Xi'}$ the node polynomial of the
grid).
Let $d_0=N_w-3$, $W_X=\operatorname{diag}(u^\vee|_X)$,
$W_Y=\operatorname{diag}(u^\vee|_Y)$.
Let $L_{ij}=\ell_j(x_i)$ be Lagrange on $Y$ evaluated on $X$,
and let $K_{ia}=(P_Y\cdot\varphi_a)(x_i)$ for a basis
$(\varphi_a)_{a<d_0-|Y|}$ of $\mathcal{P}_{<d_0-|Y|}$.
The weighted kernel projector is
\begin{equation}\label{eq:PiK}
  \Pi_K
  \;=\;
  W_X^{1/2}K\bigl(K^TW_XK\bigr)^{-1}K^TW_X^{1/2}.
\end{equation}
The min-norm interpolant in $\mathcal{P}_{<d_0}$ is
\begin{equation}\label{eq:T0}
  \Tfrak_0
  \;=\;
  (I-\Pi_K)\,W_X^{1/2}\,L\,W_Y^{-1/2}.
\end{equation}
Equivalently, the KKT of constrained least squares
$\min\|W_X^{1/2}\Phi_X c\|$ subject to $\Phi_Y c=v$.
Lagrange-plus-$\Pi_K$ is $\mathbb{Q}$-exact and catastrophically
ill-conditioned at working depth; the stable evaluation is
Chebyshev CLS.
No $R$, $A_0$, eigenvector, or target Cholesky enters the
constructor.

The Schur short-circuit Gram on $Y$ is
$S_0=L^TW_XL-L^TW_XK(K^TW_XK)^{-1}K^TW_XL$, and
$\cD_0=W_Y-S_0$ is congruent to $I-\Tfrak_0^*\Tfrak_0$
(Haynsworth I7).

\begin{lemma}[Three bases, kernel, graph over $\mathbb{Q}$]
\label{lem:basis}
On the five-node toy $\{ -1,-\half,0,\half,1\}$ with
positive square masses $\{4,9,1,16,25\}$, holes
$Y=\{0,\half\}$, and $d_0=3$, the dual masses
$u^\vee=1/(u P'^2)$ are rational squares.
Monomial, Newton, and Chebyshev bases produce identical
$S_0$; $\ker(C_Y)$ has dimension $d_0-|Y|=1$; and the
dressed resolvent
$W_Y^{1/2}(W_Y+S_0)^{-1}W_Y^{1/2}$ equals both the dual
Christoffel--Darboux Gram of depth $d_0$ and
$C(I+C)^{-1}$ with $C$ the $\mu$-only dual Gram on $Y$.
The mutant $u^\vee=1/u$ (no $P'^2$) and the shifts
$d_0\pm 1$ break the identity.
\end{lemma}

\begin{lemma}[Woodbury rank-one precision]\label{lem:woodbury}
Over $\mathbb{Q}$, for $H=\bigl(\begin{smallmatrix}3&1\\1&4\end{smallmatrix}\bigr)$,
$e=(1,-1)$, $\omega=2$,
\[
  \eta=e^TH^{-1}e=\tfrac{9}{11},\qquad
  \Theta=\bigl(\omega^{-1}+\eta\bigr)^{-1}=\tfrac{22}{29},
\]
and $S^+-S=\Theta\,ee^T$.
This is the scalar case of
$S^+-S=E^T(\Omega^{-1}+K_0A^{-1}K_0^T)^{-1}E$
with $E=L_0-K_0A^{-1}B$ (r406 \texttt{woodbury\_inv} /
\texttt{oneDefectDelta\_eq\_woodbury}, sorry-free).
\end{lemma}

\begin{lemma}[Machine graph identity]\label{lem:graph}
On FRAME-A, Chebyshev CLS at $d_0=N_w-3$ yields
$\|R_{\mathrm{graph}}-R_{\mathrm{chain}}\|_F/\|R\|_F
=2.16\cdot 10^{-14}$ against the independent r407 chain
builder, and
$\Tfrak_0^*\Tfrak_0=R^{-1}-I$ to $7.47\cdot 10^{-14}$.
$\|\Tfrak_0\|_2=1.08014$ (exactly one singular value $>1$).
$\indm(I-\Tfrak_0^*\Tfrak_0)=\indm A_0=\indm\cD_0=1$.
Named MAIN windows: $k_z=15$ residual $3.18\cdot 10^{-14}$,
$\|\Tfrak_0\|=1.107$; $k_z=18$ residual $2.82\cdot 10^{-14}$,
$\|\Tfrak_0\|=1.250$; $k_z=42$ residual $3.89\cdot 10^{-13}$,
$\|\Tfrak_0\|=1.000$; $k_z=130$ residual $1.03\cdot 10^{-12}$,
$\|\Tfrak_0\|=1.000$, $\indm=0$.
\end{lemma}

The identity $\Pi_K=0$ (depth $d_0=|Y|$) reproduces a residual
of order the r404 miss: $\|R_{\mathrm{graph}}-R\|=28.5$, and
Euler-on-$Y$ versus $A_0$ has relative residual $282$.

\begin{lemma}[Augmented graph factor]\label{lem:tdag}
Let $\Tfrak_0$ be CLS at depth $N_w-1$, $\mathrm{vt}$ the image
of the genuine border functional $\lambda_B$ (r362
\texttt{bvec\_chunked}, not a virtual node), and $k$ the
$Y$-vanishing extra column of CLS with $\ell_B$.
The min-norm lift $u_{\min}=\Tfrak_0(\Tfrak_0^*\Tfrak_0)^{-1}\mathrm{vt}$
plus the kernel column scaled to the r362 $\gamma$ produces
$\Tfrak^\dagger=[\Tfrak_0\mid u_{\min}+\alpha k]$ with
\[
  (\Tfrak^\dagger)^*\Tfrak^\dagger
  \;=\;
  (R^\dagger)^{-1}-I
\]
to $7.42\cdot 10^{-14}$ against the r362 Gram, and
$R^\dagger_{\mathrm{graph}}=R^\dagger_{\mathrm{r362}}$ to
$2.18\cdot 10^{-14}$.
On FRAME-A, $\|\Tfrak^\dagger\|_2=0.999917$
($\lambda_{\min}(R^\dagger)=0.50004146$).
A virtual star-node has spread $56.8$ (r405: Sherman--Morrison
border $\neq$ aggregated $z^{K+1}$).
\end{lemma}

This uses the r362 Gram as the comparison object.
It is a dictionary, not a new existence theorem for
$R^\dagger\succ\half I$.

\section{What $\Phi_N$ is, and what it is not}\label{sec:phi}

\begin{prop}[$\Theta_{\mathrm{terminal}}=C^T\Phi_N C$ refuted]
\label{prop:phi}
The r369 mixed-Haynsworth matrix
$\Phi_N=J^{-1}+U^TA_0^{-1}U$ is the Woodbury precision
template of Lemma~\ref{lem:woodbury}.
On FRAME-A its eigenvalues are approximately
$(-2.813,-0.0665,1.804)$.
The pair-plus-border Schur complement of the augmented Gram
$Z-I$ has eigenvalues approximately
$(0.289,0.294,0.999)$.
Relative mismatch $0.96$.
The last sequential Uvarov pivot on FRAME-A is $+0.633$, not
$-\mathrm{sch}$ and not the middle Haynsworth eigenvalue.
Literal $\Theta_{\mathrm{terminal}}=C^T\Phi_N C$ is therefore
\textbf{refuted}.
\end{prop}

\begin{prop}[Unfitted Krein transport refuted]\label{prop:krein}
Let $F$ be the r404 Euler feature matrix on $X$ and
$F^{\mathrm{tr}}=F\cdot\Tfrak_0$.
The positive Gram $F^{\mathrm{tr}}(F^{\mathrm{tr}})^T$ has
no negative eigenvalue and relative residual $5.62$ against
$\cD_0$.
The cos/sin Krein combination $F_+F_+^T-F_-F_-^T$ after
transport has alignment $3\cdot 10^{-4}$ and residual $3.37$.
No unfitted scalar recovers $\cD_0=\Tfrak_0$-transported Euler.
A pure positive Gram is excluded.
\end{prop}

\section{Census and kills}\label{sec:census}

Sequential Uvarov innovations are the last Schur pivots of
the growing-$Y$ family $\cD_m$, \emph{not} the LDL of the
terminal $\cD_0$ (that LDL is Sylvester-tautological with
$\indm\cD_0$).
Source order is the grid order of $Y$; $|u|$-descending order
kills the FRAME-A count.

\begin{prop}[Innovation census, not a theorem]\label{prop:census}
On core-$42$, $n_{\mathrm{neg}}^{\mathrm{inn}}\le 1$ on
$42/42$ ($0$ on $24$ windows, $1$ on $18$).
Equality with $\indm A_0$ holds on $32/42$ only: the ten
mismatches $k_z\in\{20,22,23,30,31,34,40,48,60,82\}$ are all
of type $0\neq 1$ (the sequential count undercounts the
inertia).
Living $\chi_3$: $37/37$ have $\indm(I-\Tfrak_0^*\Tfrak_0)\le 1$;
dead $\chi_3$: $5/5$; $\chi_4$-$20$: $0$.
Scramble: algebra residual $1.81\cdot 10^{-14}$,
$\indm=21$, $\|\Tfrak_0\|=16.66$, sequential count $12\neq 21$.
Weight-permutation: residual $3.34\cdot 10^{-12}$,
$\indm=20$ (kills P1; the graph identity still holds).
Gauss weight-jitter: residual $1.64\cdot 10^{-14}$,
$\indm$ flips $1\to 20$.
The six terminal-dead $\chi$ windows all satisfy
$\|\Tfrak^\dagger\|>1$ (operator flip), including
$\chi_3$-$23$ and $\chi_3$-$33$ whose unaugmented
$\Tfrak_0$ stayed contractive; the r405 ones-vector lift
stayed contractive on those windows.
\end{prop}

The algebra is world-blind.
The \emph{signature} of the innovations carries the arithmetic.

\section{What this reduces, and what it does not}\label{sec:red}

P1 on a canonical window is exactly one singular value of
$\Tfrak_0$ above $1$, equivalently $\indm(I-\Tfrak_0^*\Tfrak_0)=1$,
equivalently (r407) $\lambda_2(C)<1$ fails.
That is the same threshold, now with a Jacobi-free source
constructor.
It does not identify $\Phi_N$ with the terminal Uvarov
precision, does not transport Euler/Digamma without a fitted
Krein form, and does not prove $n_{\mathrm{neg}}^{\mathrm{inn}}=\indm A_0$.
The augmented identity of Lemma~\ref{lem:tdag} reuses the
r362 Gram: it does not close $R^\dagger$ as a new theorem
and does not move the mincut.
