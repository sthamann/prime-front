\begin{abstract}\noindent
Round 438 is a sealed \emph{search} round, not a lemma-first
grind: residual-$0$ finds become theorems, but nothing counts
without a machine-precision gate on every training window, a
hard train/holdout split, $\mathbb{Q}$-exact toys, and kill
tests.
The r395 truncation artefact is why holdout worlds
($\chi_3/\chi_4$, EXT, Selected $k\ge 7$) are never seen in any
fitness loop.
Three independent tracks share the r407/r431/r433 builders.
Ausgang \textbf{A:NOT\_FOUND / B:NOT\_FOUND / C:NOT\_FOUND}:
no exact source formula for $\psi_-$, $\lambda_2(C)-1$ or
$q^\dagger$; no source-pure SOS certificate for $-\det K_2$ or
$\indm(A_0)\le 1$; no canonical-cut window with
$\indm(A_0)\ge 2$.
Truncation $n=N_w-5$ \emph{does} inflate $\#\{\lambda(C)<1\}$
to $2$ --- typed r395, not a falsifier of the two-case
statement at the canonical cut $n=N_w-3$.
No RH claim.
\end{abstract}

\section{The split (r395)}

Fitness of every track sees only MAIN windows
$\mathrm{kz}\in\{9,18,20\}$ and, for Track~C, the sealed
canonical-adjacent MAIN grid
$\{5,9,12,16,18,20,22,36,38,39,52\}$ at the working depth
$n=N_w-3$.
The following are \textbf{never} in a fitness loop:
$\chi_3$ and $\chi_4$ (matched frame, parity $a=1$),
the EXT anchor $\mathrm{kz}=35$, and Selected $k\ge 7$
($\mathrm{kz}=43$).
They are evaluated after search.

Sealed budget: Track~A seed $43801$, population $24$,

\section{$\mathbb{Q}$ toys}

\begin{lemma}[machinery]\label{lem:q}
On the two-node toy $Y=\{0,1\}$, the grammar expression
$\mathbf{1}-Y$ recovers $e_0$ with cosine $1$ (residual $0$).
The r375 factorization $\det K_2=-7$, $\gamma=9/2$,
$\lambda_-=1$ is exact.
$A_0=\mathrm{diag}(-1,1,2,3)=\mathrm{diag}(1,1,2,3)-2\,e_0e_0^T$
has remainder $0$ and $\indm(A_0)=1$.
The detector flags $\mathrm{diag}(-1,-1/2,2,3)$ as
$\indm=2$.
\end{lemma}

\begin{lemma}[train/holdout ward]\label{lem:split}
The fitness tag set is exactly $\{\mathrm{MAIN}\text{-}9,
\mathrm{MAIN}\text{-}18,\mathrm{MAIN}\text{-}20\}$.
Permute of MAIN-$9$ has $n_C=21$; scramble has $n_C=21$.
The illegal hybrid $\chi_3$ with $a=0$ has $n_C=3$.
\end{lemma}

\section{Track A: grammar GP}

Hole samples of $\psi_-$ are the bottom eigenvector of the
r407 matrix $\Cmat$ (already on $Y$).
Grammar terminals: $y_i$, $u^\vee(y_i)$, $P_Y'(y_i)$,
$1/P_Y'$, $\sqrt{u^\vee}$, $\mathbf{1}$, Nyquist face,
$\theta_i=\arccos y_i$, $\mathrm{sign}(P_Y')$, hole gap,
distance to nearest $X$-node.

\begin{prop}[NOT\_FOUND]\label{prop:a}
On MAIN-$9$ the four canonical candidates have
$|\cos|=\mathrm{ones}\,0.0072$,
centred $0.0010$, Nyquist $0.3859$, $v_{\mathrm{top}}\,0.6766$.
Uniformly on $\{9,18,20\}$ the best seed is $1/P_Y'$
(min-cos $0.1010$); $v_{\mathrm{top}}$ drops to $0.0584$ at
$\mathrm{kz}=18$.
Sealed GP (seed $43801$, $24\times 10$) stays at
min-cos $0.1010$, expression \texttt{INVPY}.
Source NNLS for $\lambda_2(C)-1$ and $q^\dagger$ has
relative residual $\gg 1$ (predictions not even signed
correctly).
Holdout min-cos of \texttt{INVPY} on $n_C=1$ worlds is
not $0.999$.
\end{prop}

The target $>0.999$ uniform is missed by an order of
magnitude.
No \texttt{FOUND\_CANDIDATE} (that bar is $0.9$).

\section{Track B: SOS/LP}

\begin{prop}[NOT\_FOUND]\label{prop:b}
On the $n_-=1$ train branch, nonnegative least squares of
eight source-pure scalars onto $-\det K_2$ has relative
residual $13.2$ (coefficients cannot represent the
target).
The r375 identity $-\det K_2=\det(I+R_+)(\gamma/\lambda_--1)$
holds on MAIN-$9$ (fact $=-5.038869=\det K_2$,
$\gamma/\lambda_-=1.846$) but consumes $\psi_-$ and is
therefore not a source certificate.
The family $P=\theta_0 I+\theta_1\mathrm{diag}(u^\vee)$,
$v=v_{\mathrm{top}}$, remainder $A_0-P+vv^T$ has best
$\lambda_{\min}=-0.261$ (not PSD) on the sealed $6\times 6$
grid.
\end{prop}

\section{Track C: adversarial grid}

The two-case object is $A_0=R_{N_w-3}-I/2$, equivalently
$n_C=\#\{\lambda(\Cmat)<1\}$ at depth $n=N_w-3$.

\begin{prop}[NOT\_FOUND, truncation diagnostic]\label{prop:c}
On the canonical cut $dn=0$, all $11$ CORE windows have
$n_C\le 1$.
Closest $\lambda_2(C)-1=2.245\cdot 10^{-7}$ at
$\mathrm{kz}=52$.
Weight-scale GA ($|\varepsilon|\le 2\cdot 10^{-4}$, $72$
points on MAIN-$9$) keeps $n_C=1$.
Holdout: $\chi_3$-$9$ $n_C=0$, $\chi_3$-$15$ $n_C=1$,
$\chi_3$-$19$ $n_C=1$, $\chi_4$-$9$ $n_C=1$,
$\chi_4$-$20$ $n_C=0$, Selected $k=7$ $n_C=0$,
EXT-$35$ $n_C=1$ --- none has $n_C\ge 2$.
Truncation $dn=-2$ inflates $n_C=2$ on
$\mathrm{kz}\in\{5,18,22,38,52\}$ (and $dn=-1$ at
$\mathrm{kz}=52$).
That object is \emph{not} $A_0$; it is the r395
truncation class and is not a falsifier of the
two-case statement.
\end{prop}

\section{What this does not say}

The round does not prove $\indm(A_0)\le 1$, does not produce
a source formula for $\psi_-$ or $q^\dagger$, and does not
falsify the two-case candidate on the canonical cut.
It hardens the census: the closest canonical
$\lambda_2-1$ in the sealed grid is $2.245\cdot 10^{-7}$,
and every holdout world stays in $\{0,1\}$ negatives.
Search was allowed to be dirty; verification was not.
