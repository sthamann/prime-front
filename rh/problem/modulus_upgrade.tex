\section{Proved constants}

\begin{lemma}[Prime channel]\label{lem:P}
$|P(g)-P(g')|\le 2S_{\mathrm{eff}}(L)\,\|g-g'\|_\infty$, where
$S_{\mathrm{eff}}(L)=\sum_{n\le e^{2L}}\Lambda(n)/\sqrt n$
is a finite exact interval sum.  At $L=0.3$ one has $2L<\log2$,
hence $S_{\mathrm{eff}}=0$.
\end{lemma}

\begin{lemma}[Pole channel]\label{lem:Pi}
$|\Pi(g)-\Pi(g')|\le 8\sinh(L)\,\|g-g'\|_\infty$.
\end{lemma}

\begin{lemma}[Autocorrelation]\label{lem:g}
$\|g-g'\|_\infty\le(\|h\|_2+\|h'\|_2)\|h-h'\|_2$.
\end{lemma}

\begin{lemma}[Jackson]\label{lem:J}
On each mesh interval of length $\delta$, if $e=h-I_\delta h$
vanishes at the endpoints, the sine series gives
$\|e\|_2\le(\delta^2/\pi^2)\|h''\|_2$ and
$\|e'\|_2\le(\delta/\pi)\|h''\|_2$.
\end{lemma}

\begin{lemma}[Arch increment, enclosure]\label{lem:A}
Write $N(w)=e^{-w/2}/(1-e^{-2w})$ and
$B(w)=e^{-2w}/(1-e^{-2w})$.  Then
\[
 |A(g)-A(g')|
 \le(\gamma+\log\pi)\,|\Delta g(0)|
 +2\int_0^{2L}\min\bigl(2\|\Delta g\|_\infty,\,w\,\mathrm{Lip}(\Delta g)\bigr)
   N(w)\,dw
 +2|\Delta g(0)|\Bigl(\int_0^{2L}|B-N|
 +\int_{2L}^\infty B\Bigr),
\]
with $\int_0^\eta w N$ summed from the $w=0$ endpoint
$\sum_k\alpha_k^{-2}$ and $\int_a^b N$ by a geometric Bose series
with $a>0$.  The bound is sealed, not sharp.
\end{lemma}

The r471 certificate $Q_L\succeq\mu I$ in coefficient Euclidean
norm converts to an $L^2$-Rayleigh floor $R_{\mathrm{grid}}=\mu/\delta$
on the step family.  The combination implication (PROVED) is: if
$R_{\mathrm{grid}}>\omega_L(\delta,\sigma_2)$ then
$\QW(h)\ge(R_{\mathrm{grid}}-\omega)\|h\|_2^2$ for every $H^2$ function
with that $\sigma_2$ whose interpolant lies in the certified family.
