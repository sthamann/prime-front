\begin{abstract}\noindent
Round 383 attacks the three named remainders of the reduced
compose implication of~\cite{r378}: an absolute Fej\'er/diagonal
ratio $(R)$, an $\ell^1$ bound $(L)$, and an edge scalar $(Z)$.
What is proved: the Fej\'er identity and bilinear Cauchy--Schwarz;
the exact $r298$ split $S_F=B(\omega,\omega)+B(\Delta,\omega+\beta)$;
the exact $L^2$ identity
$M_3=(({\log m})^2/m^2)\,T_2\,E_\pi(F^2)$;
$\Lambda(n)\le\log n$ by trial division;
the $H$-rule $\pref\le\sqrt{m}+1$;
and the T1-floor algebra conditional on $g_i\ge 3/8$.
What is \emph{not} proved: a cofinal absolute $R_0$, a cofinal
$\gamma<1/4$ from the von~Mangoldt triangle, a family-uniform
$E_\pi(F^2)$ bound, or a PNT-free finite-head bound on
$\lvert Z_{\loc}\rvert$.
Exact period-$2$ fold pairing makes $B(\omega,\omega)$ of size
$O(D/m)$, but the live window field has pair-residual median
$0.512$ --- the emptiness of the window main term is a census,
not a pairing theorem.
On the $181$-window surface one has $S_F\le 4\,D$ and $L_1\le 3$
(measured maxima $2.917$ and $2.126$); $\lvert Z_{\loc}\rvert\le 4/5<M$
on every FRAME-A and FRAME-B window, and fails on six $\chi$ windows.
The T1-combo route does not carry ($E_\pi$ spikes at kz$111$/$117$).
Scramble seed~$1$ named-breaks $(L)$ on the $M_3$ polylog.
No RH claim.
\end{abstract}

\section{Recollection}\label{sec:rec}

Write $M=\sqrt{5/7}$, $H=\max(2,\lceil\sqrt{m}\rceil)$,
$\pref=(m+H-1)/H$, $D=\sum_j P_j^2$, $L_1=\sum_j\lvert P_j\rvert$,
and $S_F=(1/H)\sum_t W_t^2$ for the Fej\'er energy of a real
block field $P$ of length $m$ \cite{r378}.
The reduced compose implication is
\[
  S_F\le R_0 D
  \ \wedge\
  L_1\le\Lambda
  \ \wedge\
  \lvert Z_{\loc}\rvert\le Z_0<M
  \ \wedge\
  M_3\le\phi(m)
  \implies
  \lvert Z\rvert<M,
\]
with
$\phi(m)=(M-Z_0)^4/(\pref(m)^2 R_0^2\Lambda^4)$
and $\pref(m)\le\sqrt{m}+1$.
If $L_1\le C m^\gamma$ replaces the absolute $\Lambda$, the
same algebra demands $2\beta=1+4\gamma$; T1-floor supplies
$\beta=1$, which beats this if and only if $\gamma<1/4$
(at $\gamma=1/4$ one needs $(\log m)^2$ bounded).
Hypothesis H5 is a sister route, not an extra premise.
Claim boundary: finite identities and named reductions.
Not evidence for or against the Riemann hypothesis.

\section{Shared machinery}\label{sec:shared}

\begin{lemma}[Fej\'er identity and van der Corput]\label{lem:fejer}
For every real sequence $P$ of length $m$ and every
$H\in\{1,\ldots,m\}$,
$\lvert\sum P\rvert^2\le\pref\,S_F$ with
$S_F=(1/H)\sum_t W_t^2=\sum_{\lvert h\rvert<H}(1-\lvert h\rvert/H)A(h)\ge 0$.
The bilinear form $B(u,v)=(1/H)\sum_t W_t(u)W_t(v)$ is a Gram form:
$B(u,v)^2\le B(u,u)B(v,v)$.  Consequently $S_F\le H D$.
\end{lemma}

\noindent
The bound $S_F\le H D$ is a theorem and is too coarse for
composition ($H$ grows like $\sqrt{m}$).  Companion: exact
period-$2$ witness $P=(1,-1)^{8}$, $H=4$: $S_F=1$, $D=16$,
$R=1/16$; DC witness $P=1^{16}$: $R=3.6875\le H=4$.

\begin{lemma}[The $r298$ split]\label{lem:split}
On the frozen positional Fej\'er kernel,
$S_F=B(\omega,\omega)+B(\Delta,\omega+\beta)$ identically,
with $\Delta=\beta-\omega$.
The geometric edge of the canonical split satisfies
$Z=Z_{\loc}+t_{\mathrm{bulk}}$ (Lean \texttt{canonical\_split});
$t_{\mathrm{bulk}}$ is the bulk chain sum, not $\sum P_\Delta$.
\end{lemma}

\begin{lemma}[Exact period-$2$ fold]\label{lem:fold}
If $P=(a,-a,\ldots,a,-a)$ has even length and $H$ is even, then
every complete $H$-window sums to $0$ and $S_F/D=O(1/m)$
(the residual is the $H-1$ incomplete edge windows).
In particular the window main term $B(\omega,\omega)$ would be
$O(D_\omega/m)$ if $P_\omega$ were exactly period-$2$.
\end{lemma}

\begin{remark}\label{rem:fold}
Lemma~\ref{lem:fold} is \emph{not} the live $P_\omega$.
On $181$ windows the adjacent-pair residual
$\sum\lvert P_{\omega,2k}+P_{\omega,2k+1}\rvert/
\sum\lvert P_{\omega,2k}\rvert+\lvert P_{\omega,2k+1}\rvert$
has median $0.512$.  The measured emptiness
$\mathrm{med}\,B(\omega,\omega)/S_F=0.0086$ is a low-pass
census (window lag-$1$ correlation median $-0.459$), not a
pairing identity.  Fold-parity as a SATZ for the live field is
therefore refuted; MAIN-empty remains a census.
\end{remark}

\begin{lemma}[$L^2$ identity]\label{lem:l2}
Writing $q_i=\lvert P_i\rvert/L_1$, $g_i$ the normalised local
gap, $F_i=(m q_i/\log m)\,g_i$, $T_2=\sum_i q_i/g_i^2$ and
$\pi_i=(q_i/g_i^2)/T_2$, one has
$M_3=(({\log m})^2/m^2)\,T_2\,E_\pi(F^2)$ identically.
\end{lemma}

\begin{lemma}[Von Mangoldt triangle]\label{lem:lambda}
$\Lambda(n)\le\log n$ for every integer $n\ge 2$
(trial division: $\Lambda(p^k)=\log p$).
The triangle $L_1\le\sum_{\mathrm{atoms}}\lvert\Delta c_a\rvert$
therefore costs at most one $\log$ per atom, not a power
$\gamma<1/4$.
\end{lemma}

\begin{lemma}[$H$-rule and $\phi$]\label{lem:phi}
At $H=\max(2,\lceil\sqrt{m}\rceil)$ one has $\pref\le\sqrt{m}+1$.
If $g_i\ge 3/8$ and $F_i\le C_K$ then
$M_3\le\bigl((8/3)C_K(\log m)/m\bigr)^2$.
Under $(R)(L)(Z)$ the sufficient $M_3$-cap is
$\phi(m)=(M-Z_0)^4/(\pref^2 R_0^2\Lambda^4)$.
\end{lemma}

\section{Premise $(R)$}\label{sec:R}

\begin{prop}[$(R)$, reduced]\label{prop:R}
On the $181$-window surface (FRAME-A $89$ $+$ FRAME-B $8$ $+$
$\chi_3$ $42$ $+$ $\chi_4$ $42$),
$S_F\le 4\,D$.  The measured supremum is $2.916788$ at FRAME-A
kz$37$ ($H=15$).  The constant $R_0=4$ is at most twice that
supremum and strictly below the trivial envelope
($H_{\max}=29$).  The mutant $R_0/2=2$ fails on $17$ windows
and on the DC toy.  A cofinal absolute $R_0$ is not proved:
Lemma~\ref{lem:fold} does not apply to the live $P_\omega$,
and $S_F\le H D$ grows.
\end{prop}

\noindent
Outcome: \textbf{REDUCED} to the census $R_0=4$, with the
cofinal ratio a named remainder.  $\chi$ is not an exception
for $(R)$.

\section{Premise $(L)$}\label{sec:L}

Two routes were available.

\textit{T1-combo.}  Lemma~\ref{lem:l2} is a theorem
(worst relative residual $1.4\cdot 10^{-15}$ on $181$ rows).
Under the floor $g_i\ge 3/8$ one has $T_2\le(8/3)^2$, and
indeed $T_2\le 1.329$ on the surface.  The remaining factor is
$E_\pi(F^2)$.  It is \emph{not} family-uniform polylog:
$E_\pi=52.276$ at FRAME-A kz$111$ and $E_\pi=75.691$ at
FRAME-B kz$117$ (the $r368$ spikes).  The T1-combo therefore
does \emph{not} kill $(L)$ and T1 together.

\textit{Von Mangoldt / FAB.}  Lemma~\ref{lem:lambda} gives
$\Lambda\le\log$, hence a triangle.  The live total variation
has log-log slope $+0.307$ on FRAME-A, which is \emph{not}
$<1/4$.  The triangle does not prove $\gamma<1/4$.

\textit{Census.}  $L_1\le 3$ on all $181$ windows (maximum
$2.126$ at FRAME-B kz$124$).  The log-log slope of $L_1$ is
$+0.202$ on the $42$-rung ladder and $+0.223$ on FRAME-A ---
measured, strictly below $1/4$, not a theorem.  Because the
slope is positive, no cofinal absolute $\Lambda$ follows from
the census.

\begin{prop}[$(L)$, reduced]\label{prop:L}
$L_1\le 3$ on the $181$-window surface.  The T1-combo is a
theorem of the identity and a refutation of the $E_\pi$ bound.
The von~Mangoldt triangle does not yield $\gamma<1/4$.
The mutant $\gamma=1/4$ fails the T1-versus-$\phi$ algebra
($(\log m)^2$ unbounded).  Scramble seed~$1$ at $w=9$
named-breaks the $r306$ polylog
$M_3\le 1.069\,(\log m)^2/m^2$ ($2.288\cdot 10^{-2}>1.374\cdot 10^{-2}$)
while leaving $R_0=4$ and $\Lambda=3$ intact --- the named
$(L)$ break.
\end{prop}

\noindent
Outcome: \textbf{REDUCED}.  Remaining: a cofinal $\gamma<1/4$
or a family-uniform $E_\pi(F^2)$ bound.

\section{Premise $(Z)$}\label{sec:Z}

$Z_{\loc}=t_{\loc}+\mathrm{chain}$ is the $F=0.20$ geometric
edge of the canonical split (Lean \texttt{canonical\_split} is
the algebraic triangle, already proved).  The geometric edge
contains $167$ to $4689$ atoms on the surface --- it is not a
finite von~Mangoldt head.  A Chebyshev/Rosser bound
$\psi(x)\le 1.04x$ therefore does not supply a finite sum.

\begin{prop}[$(Z)$, reduced on FRAME-A, refuted on $\chi$]\label{prop:Z}
On FRAME-A, $\lvert Z_{\loc}\rvert\le 4/5<M$ holds on $89/89$
windows (maximum $0.755675$ at kz$16$).  FRAME-B: $8/8$
(maximum $0.540$).  The $42$-rung ladder: $42/42$.
On $\chi$ the bound $\lvert Z_{\loc}\rvert<M$ fails on six
windows:
$\chi_3$ kz$\{15,19,23,33,39\}$ and $\chi_4$ kz$20$
(maximum $1.021$ at $\chi_3$ kz$15$).
Thus $175/181$ windows satisfy $\lvert Z_{\loc}\rvert<M$.
A cofinal PNT-free bound is not proved.
\end{prop}

\noindent
Outcome: \textbf{REDUCED} on the canonical FRAME-A family with
$Z_0=4/5$; \textbf{REFUTED} as a family-uniform statement on
$\chi$.  Finite-head Chebyshev: \textbf{REFUTED}.

\section{The contract}\label{sec:contract}

Taking the census constants $R_0=4$, $\Lambda=3$, $Z_0=4/5$
inside the r378 implication, COMPOSE$^-$ is a theorem of those
three inequalities plus $M_3\le\phi(m)$ on every FRAME-A window
of the $181$-surface.  The three inequalities are census SATZ
on that surface, not cofinal.  $\chi$ blocks only $(Z)$.
T1 remains open as a family-uniform $C_K$ or $E_\pi$ bound;
the identity of Lemma~\ref{lem:l2} stands.

H5/\texttt{PairMarginLaw} is untouched (sister route).
The $r263$ dictionary remains a named \texttt{sorry}.
The $42$-rung $q_N<1$ certificate is already sealed and is not
re-proved here.
