\begin{abstract}\noindent
After rounds 400--403 the bulk of $R_{N-3}-\half I$ has a
single defect candidate and the edge is an exact $3\times 3$
normal form $\Phi_{\mathrm{edge}}(a,b)$ with living/dead
coordinate $\mathrm{sch}$.
The reviewer (DCCLXIX) asks that the missing Euler null mode
be written $\ell=Vc$ from geometric recurrences, that
$1-\|c\|^2$ be a manifest positive source sum, and that this
reserve equal $\kappa(a,b)\,(-\mathrm{sch})$ in the r401
Sylvester chart, with dead $\chi$ appearing as $\|c\|>1$.
The Euler tail, the disk Parseval identity, and the Woodbury
identity $\Delta=\kappa_{\mathrm{closed}}(-\mathrm{sch})$ are
finite SATZ.
The source-explicit lift of the \emph{constant} on $Y$ has
residual~$0$.
The Sherman--Morrison border is not the aggregated tail
$z_p^{K_p+1}$; dead $\chi$ keep $\|c\|<1$ (death remains
$\mathrm{sch}>0$); $\kappa$ is not a function of $(a,b)$
alone; the geometric sum and Woodbury $\Delta$ differ by
two to four orders of magnitude.
The $A_0$-defect lift stays open (round 404: class~3 not
reached; $|\cos|(v_-,1)=0.007$).
No RH claim.
\end{abstract}

\section{What is asked}\label{sec:ask}

Write $A_0=R_{N-3}-\half I$ on the hole nodes $Y$ and
$M^\dagger=A^\dagger+UJU^T$ for the mixed form of round 369,
with $J=\operatorname{diag}(1,1,1/\mathrm{den})$ and
$\Phi=J^{-1}+U^T(A^\dagger)^{-1}U$ the r401 $3\times 3$.
If a source-pure $H\succ 0$ and a source-pure $\ell$ satisfy
$A=H-\ell\ell^T$ on the same space as $U$, Woodbury gives
\begin{equation}\label{eq:D}
\Delta
\;=\;
1-\eta+r^T K_W^{-1}r,
\qquad
\eta=\ell^T H^{-1}\ell,
\quad
r=U^T H^{-1}\ell,
\quad
K_W=J^{-1}+U^T H^{-1}U,
\end{equation}
and $M=H+UJU^T-\ell\ell^T\succ 0$ if and only if $\Delta>0$.
The constructive form is $\ell=Vc$ with
$V=[F_\Lambda,F_\Gamma,F_{\mathrm{Uv}},UJ^{1/2}]$ and
$\|c\|<1$.
The Euler tail
\begin{equation}\label{eq:tail}
1-z_p^{K_p+1}
\;=\;
(1-z_p)\sum_{r=0}^{K_p}z_p^r,
\qquad
z_p=p^{-1/2}e^{i\vartheta_p},
\end{equation}
is the proposed source of $c$, with $1-\|c\|^2$ equal to
\begin{equation}\label{eq:sum}
\eta_\Gamma+\eta_{\mathrm{pole}}
+\sum_p(1-p^{-1})|b_p|^2
+\sum_p p^{-(K_p+1)}|t_p|^2
\end{equation}
and, in the r401 chart, to $\kappa(a,b)\,(-\mathrm{sch})$.

\section{Finite identities}\label{sec:satz}

\begin{lemma}[Euler tail]\label{lem:tail}
For every $z$ and every integer $K\ge 0$,
$1-z^{K+1}=(1-z)\sum_{r=0}^K z^r$.
Over $\mathbb{Q}$, at $z=1/3$ and $K=5$, both sides equal
$728/729$.
Omitting the tail leaves the remainder $z^{K+1}\neq 0$.
\end{lemma}

\begin{lemma}[disk Parseval]\label{lem:disk}
For $0<q<1$ and $K\ge 0$,
$(1-q)\sum_{r=0}^K q^r=1-q^{K+1}$.
At $q=1/2$ and $K=6$ the left side is $127/128$ and the
reserve $q^{K+1}$ is $1/128$.
With $q=p^{-1}=|z_p|^2$ this is~\eqref{eq:sum} for a single
prime ($b_p=1$, no Gamma/pole).
\end{lemma}

\begin{lemma}[Woodbury / Schur]\label{lem:kappa}
Let $H\succ 0$ (or merely invertible), $A=H-\ell\ell^T$
invertible, and $\Phi=J^{-1}+U^T A^{-1}U$.
Write $\mathrm{sch}$ for the scalar Schur complement of the
leading $2\times 2$ of $\Phi$, and $K_2$ for that block.
Then
\[
\Delta
\;=\;
(1-\eta)\frac{\det\Phi}{\det K_W}
\;=\;
\kappa_{\mathrm{closed}}\,(-\mathrm{sch}),
\qquad
\kappa_{\mathrm{closed}}
\;=\;
-\,(1-\eta)\,\frac{\det K_2}{\det K_W}.
\]
Over $\mathbb{Q}$, on $H=\operatorname{diag}(2,3,4)$,
$\ell=(1,1,1)$, $U=I$, $J^{-1}=\operatorname{diag}(1,1,2)$:
$\eta=13/12$, $\Delta=7/36$, $\mathrm{sch}=21/8$,
$\kappa_{\mathrm{closed}}=-2/27$, and the identity is exact.
After the r401 Sylvester map $D$, $\det K_2=\det J_2/\det(D_2)^2$,
so $\kappa_{\mathrm{closed}}$ still depends on the bulk split
$(H,\ell)$ through $\eta$ and $K_W$: it is not a function of
the chart coordinates $(a,b)$ alone.
\end{lemma}

\begin{lemma}[Cauchy Gram]\label{lem:cauchy}
The $2\times 2$ Cauchy Gram $G_{ij}=1/(z_i z_j)$ at
$z=(1,2)$ has $\det G=1/72>0$.
This is the $F_\Gamma$ block over $\mathbb{Q}$; the lift below
takes its coefficients to be zero.
\end{lemma}

\section{The geometric lift of the constant}\label{sec:E1}

On $Y$ set $\vartheta_j=\arccos(y_j)$ and
$z_p(j)=p^{-1/2}e^{i\vartheta_j\log p}$.
Let $\nu_p=\log p/\sum\log p$ over a finite prime list and
$E_p=(1-z_p)\sum_{r=0}^K z_p^r$, $T_p=z_p^{K+1}$.
Then~\eqref{eq:tail} gives $E_p+T_p=1$ pointwise, hence
\begin{equation}\label{eq:ones}
\mathbf{1}
\;=\;
\sum_p \nu_p\,(E_p+T_p)
\end{equation}
with explicit $c$ (the weights $\nu_p$ on the real/imaginary
parts of $E_p$ and $T_p$).  No optimiser.

\begin{prop}[ones-lift]\label{prop:ones}
On FRAME-A, $\|\mathbf{1}-\sum\nu(E+T)\|_2\le 10^{-12}$.
Omitting $T$ leaves residual $0.049$.
The same identity continues to hold after a $10^{-4}$ jitter
of the nodes (it is an identity in $z$).
The bulk defect $v_-$ of $A_0$ is not this vector:
$|\cos|(v_-,\mathbf{1})=0.007$ and the residual of $v_-$
against the geometric $V$ is $0.81$.
Round 404 independently records that the ones-mode is not
the null mode of $A_0$.
\end{prop}

\begin{prop}[border is not the tail]\label{prop:border}
The $Y$-component $s$ of the Sherman--Morrison border of
round 401 has relative residual $0.996$ against
$\operatorname{span}\{\operatorname{Re}T,\operatorname{Im}T\}$
and correlation $0.07$ with $\operatorname{Re}T$, on the
named phase $\vartheta_j\log p$.
Every other source-explicit phase convention tested (shared
$\arccos y$, $y\log p$, Chebyshev $T_5$) leaves relative
residual $\ge 0.96$.
The identification ``border $=$ aggregated $z^{K+1}$'' is
refuted for these conventions.
\end{prop}

\section{Woodbury $\Delta$ of the ones-split, and worlds}\label{sec:E2}

The mixed $A^\dagger=\operatorname{blkdiag}(A_0,-1/2)$ is not
a rank-one update of a positive $H$ (two negatives).
The ones-split used for measurement is the $Y$-only object
$H=A_0+\mathbf{1}\mathbf{1}^T$ (normalised) with the two CD
columns, which is the bulk half of~\eqref{eq:D}.
Then $\Delta$ and $\|c_{\min}\|^2=\ell^T B^{-1}\ell=1-\Delta$
are closed forms, not an optimisation.

\begin{prop}[ones-split census]\label{prop:worlds}
On FRAME-A, $\Delta=2.110\cdot 10^{-3}$,
$\|c\|^2=0.99789$, $\kappa=\Delta/(-\mathrm{sch})=0.0315$
with $\mathrm{sch}=-0.06696$.
On core-$42$: $\Delta>0$, $\kappa>0$, and $0<\|c\|^2<1$ on
$42/42$, with $\Delta\in[6.7\cdot 10^{-6},2.11\cdot 10^{-3}]$.
Living $\chi_3$: $37/37$ have $\kappa>0$ and $\|c\|<1$.
Dead $\chi_3$: $5/5$ have $\mathrm{sch}>0$, $\kappa<0$,
\emph{and} $\|c\|<1$.
The predicted overflow $\|c\|>1$ on dead $\chi$ is refuted
for this source $c$.
Death remains $\mathrm{sch}>0$ (round 401).
The geometric disk sum~\eqref{eq:sum} is $O(10^{-1})$ on
the named primes; Woodbury $\Delta$ is
$O(10^{-5}$--$10^{-3})$: they are not the same scalar, so
``sum form $=\kappa(-\mathrm{sch})$'' is refuted on scale.
Weight permutation already yields $\indm A_0\ge 15$
(round 403): factorisation~(G) breaks before any lift.
Scramble: $\indm=21$.  Wrong-sign $J$: mixed residual
$867/115\neq 0$.
\end{prop}

\section{What this is not}\label{sec:not}

This is not a proof of $R^\dagger\succ\half I$, not a
source-explicit $A_0=H-\ell\ell^T$ (round 404 stopped at
class~3), and not a unique $\kappa(a,b)$ in the r401 chart.
The finite SATZ are the tail, the disk identity, and the
Woodbury--Schur determinant formula.
The $A_0$-defect lift, and the identification of the mixed
border with an Euler tail, remain open or refuted as named.
No RH claim.
