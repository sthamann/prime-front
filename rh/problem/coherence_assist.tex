\begin{abstract}\noindent
Round 385 reduced geometric quietness $Q_k<1$ to the pair
$(\Delta,C)$, and named $C$ as the r285 factor
$\lambda_{\max}(E_k)=\mathrm{maxdiag}_k\cdot(1+\mathrm{assist}_k)$.
This note asks whether $\mathrm{assist}_k$ admits a source-defined
upper bound, from $\nu$-angle geometry on the cosine grid, that
pushes $\lambda_{\max}<1$ through $k=N-1$.
What is proved: the Chebyshev--$T$ Christoffel--Darboux kernel is
the Dirichlet combination
$[D_{k-1}(\theta-\phi)+D_{k-1}(\theta+\phi)]/(2\pi)$;
the bookkeeping identity; the Gershgorin majorant;
the cosine-mesh identity $\Delta_{\mathrm{mesh}}=\pi/S$;
and that the two-period adversary is a \emph{single arithmetic
progression} of $\nu$-angles ($\rho_{\mathrm{AP}}=1$).
What is \emph{not} proved: Gershgorin plus the $3/8$-floor
closes the budget $\mathrm{assist}<(1-\mathrm{maxdiag})/\mathrm{maxdiag}$.
On FRAME-A $w=9$, Gershgorin crosses $1$ at $k=21$ while
$n_0=\lfloor 2N/5\rfloor=73$ and the wall sits at $k=183$;
core-$42$ has $k^*_{\mathrm{Gershgorin}}/N\le 0.114<2/5$.
The $3/8$-floor does not upgrade $\nu$-angle gaps:
$d_{\min}=1$ mesh step and $d_{\mathrm{med}}=3$ mesh steps on
$w=9$, so $d_{\min}/d_{\mathrm{med}}=1/3<3/8$.
The hydra-legal remainder is the \emph{signed} $\mu$-CD
off-diagonal (true assist $0.040$ at the $w=9$ wall against
Gershgorin assist $13.3$, cancellation $99.7\,\%$).
No RH claim.
\end{abstract}

\section{The object}\label{sec:obj}

Write $B_k$ for the $\nu$-dressed matrix of the first $k$
$\mu$-orthonormal polynomials and $E_k=B_k B_k^T$, so
\begin{equation}\label{eq:E}
(E_k)_{ij}
=
\sqrt{v_i v_j}\,K_k(y_i,y_j),
\end{equation}
with $K_k$ the $\mu$-Christoffel--Darboux kernel of degree $k$.
The r285 identity
\begin{equation}\label{eq:book}
\lambda_{\max}(E_k)
=
\mathrm{maxdiag}_k\cdot(1+\mathrm{assist}_k),
\qquad
\mathrm{assist}_k
=
\lambda_{\max}/\mathrm{maxdiag}_k-1
\end{equation}
is exact.  $L^*$ is $\lambda_{\max}(E_{N_w})<1$.  Round 382
reduced a flank-mass entry to $n_0=\lfloor 2N/5\rfloor$ and named
this assist as the $L^2$ remainder for $n_0=N-1$.  Round 385
showed $Q_k<1$ is not sufficient (two-period $c=2/3$).

A parenthetical correction: r385 quoted ``assist $1.5$--$4$'' at
$k=183$ against $\max Q\approx 0.39$.  That mixed the
\emph{single-degree} sampling $Q_k=\langle\pi_k^2\rangle_\nu$ with
$\mathrm{maxdiag}(E_k)=\max_j v_j K_k(y_j,y_j)$.  The wall values
are $\mathrm{maxdiag}_{183}=0.9614$ and $\mathrm{assist}_{183}=0.0399$,
inside the r285 ladder $0.0059$--$0.0702$.

The candidate of this round: a geometric bound on the off-diagonal
of~\eqref{eq:E} from Dirichlet evaluations at $\nu$-angle
differences, using the $3/8$-floor and F1 run-lengths, feeding
Gershgorin or Schur so that $\lambda_{\max}<1$ through $k=N-1$.

\section{What the grid sees}\label{sec:grid}

Write $D_n(\alpha)=\sin\bigl((n+\tfrac12)\alpha\bigr)/\sin(\alpha/2)$
for the Dirichlet kernel ($D_n(0)=2n+1$).

\begin{lemma}[Chebyshev--Dirichlet]\label{lem:cd}
For the Chebyshev-$T$ orthonormal system on $[-1,1]$
($p_0=\pi^{-1/2}$, $p_n=\sqrt{2/\pi}\,T_n$),
\[
K_n(\cos\theta,\cos\phi)
=
\frac{D_{n-1}(\theta-\phi)+D_{n-1}(\theta+\phi)}{2\pi}.
\]
This is an identity of trigonometric polynomials, independent of
any window.  It is the Chebyshev \emph{proxy} of~\eqref{eq:E}, not
the $\mu$-CD kernel.
\end{lemma}

\begin{lemma}[bookkeeping]\label{lem:book}
Identity~\eqref{eq:book} holds for every Hermitian $E$ with
positive diagonal: it is the definition of $\mathrm{assist}$.
On the $2\times 2$ equality case
$E=\bigl(\begin{smallmatrix}2&1\\1&2\end{smallmatrix}\bigr)$
one has $\lambda_{\max}=3$, $\mathrm{maxdiag}=2$,
$\mathrm{assist}=\tfrac12$.
\end{lemma}

\begin{lemma}[Gershgorin]\label{lem:gersh}
$\lambda_{\max}(E)\le \max_i\bigl(E_{ii}+\sum_{j\neq i}\lvert E_{ij}\rvert\bigr)$.
On the $2\times 2$ of Lemma~\ref{lem:book} this is equality
(the all-ones mode).  The same number is the Schur $/\ell^\infty$
bound.  It uses \emph{absolute} off-diagonals.
\end{lemma}

\begin{lemma}[cosine mesh]\label{lem:mesh}
On the construction grid $x_j=\cos(\pi j/S)$ the angles are
$\theta_j=\pi j/S$ and consecutive mesh steps equal $\pi/S$.
A $\nu$-run of length $2$ (permitted by F1) realises
$d_{\min}(\nu)=\pi/S$.  The $3/8$-floor of the log-sieve
controls fold gaps, not this mesh step: on FRAME-A $w=9$ one has
$d_{\min}=1$ mesh, $d_{\mathrm{med}}=3$ mesh, hence
$d_{\min}/d_{\mathrm{med}}=1/3<3/8$.
\end{lemma}

\begin{lemma}[two-period AP]\label{lem:ap}
The two-period toy of amplitude $c$ on $S$ nodes occupies every
other cosine index.  Its $\nu$-angles form a single arithmetic
progression of common difference $2\pi/S$, so
$\rho_{\mathrm{AP}}:=L_{\mathrm{AP}}/n_\nu=1$ and consecutive
gaps are unique.  This is the occupation of one residue class.
\end{lemma}

\section{The geometric candidate, measured}\label{sec:geom}

Write $k^*_{\mathrm{G}}$ for the last degree with Gershgorin
majorant $<1$, and
$\rho_{\mathrm{AP}}=L_{\mathrm{AP}}/n_\nu$ for the longest
arithmetic progression in the $\nu$-angle set, divided by the
number of $\nu$-atoms.

\begin{prop}[census]\label{prop:census}
On the sealed constructors:
\begin{enumerate}
\item FRAME-A $w=9$: $N=184$, $n_\nu=104$,
  $k^*_{\mathrm{G}}=20$ (first crossing $k=21$),
  $n_0=73$, $k^*_{\mathrm{G}}/N=0.109$.
  At $k=183$: $\lambda_{\max}=0.999830$,
  $\mathrm{maxdiag}=0.9614$, $\mathrm{assist}=0.0399$,
  Gershgorin assist $13.32$, cancellation
  $1-\mathrm{assist}/A_{\mathrm{G}}=0.997$.
  $\rho_{\mathrm{AP}}=0.0769$, gap-CV $0.813$,
  $d_{\min}/d_{\mathrm{med}}=1/3$.
  Dirichlet max-row at $k=22$: $0.291$.
\item Core-$42$: $\rho_{\mathrm{AP}}\in[0.034,0.168]<1/5$;
  gap-CV $\in[0.536,0.892]>1/2$;
  $k_{\mathrm{Gershgorin}}\in[8,32]$, all strictly below
  $n_0$, and $k_{\mathrm{G}}/N\le 0.114<2/5$.
  The mutant bar $\rho_{\mathrm{AP}}\le 0.03$ fails
  (max $0.168$).
\item EXT-heavy seven $\{69,96,97,99,107,117,129\}$:
  $\rho_{\mathrm{AP}}\in[0.006,0.033]<0.05$,
  gap-CV $\ge 0.56$, and $\lambda_{\max}(E_{80})<1$
  (sharpest $0.9965$ at kz$97$).
\end{enumerate}
Gershgorin does not improve the r382 entry
$n_0=\lfloor 2N/5\rfloor$.  The unsigned Dirichlet row-sum
grows with $k$ and crosses $1$ near $0.11\,N$.
\end{prop}

The $3/8$-floor cannot feed a better $d_{\min}$: F1 already
permits a run of two, which saturates the mesh.  A crude
Dirichlet cap $1/\sin(d_{\min}/2)\sim 2S/\pi$ is $\Theta(N)$ and
useless past $k\sim 10$.

\section{Why the two-period dies}\label{sec:ap}

The two-period of amplitude $c=2/3$ on $S=81$ has
$Q\equiv\alpha=0.683<1$ and $\lambda_{\max}(E_{22})=1.029>1$.
Its $\nu$-atoms are equidistant: $\rho_{\mathrm{AP}}=1$,
gap-CV $0.078$ (wrap residual), Dirichlet max-row at $k=22$
equal to $0.858$, about three times the source.
Dirichlet peaks add.  Amplitude $c=1$ dies at $k=1$
($\lambda_{\max}=1.025$).

The source is not a residue class.  Log-near prime powers plus
F1 (run-lengths $\le 2$) produce a spread of consecutive
$\nu$-gaps ($w=9$: multipliers $0.375,0.625,1,1.375,\ldots$ in
eighths of the median; twelve distinct values against the
two-period's one).  That is the named property:
\begin{equation}\label{eq:rho}
\rho_{\mathrm{AP}}
\;<\;
\tfrac15
\qquad\text{on canonical windows.}
\end{equation}
It is construction-level (the fold image of the prime-power comb
is not one residue class) and is census-grade on core-$42$ and
the EXT-heavy seven.  It is \emph{not} sufficient for
$\lambda_{\max}<1$: scramble seed $1$ has $\rho_{\mathrm{AP}}=0.064$
like the source and still crosses $\lambda_{\max}\ge 1$ at $k=22$
(maxdiag crosses $1$ later, at $k=50$).  Scramble is a diagonal /
product kill, not an AP kill.

F1 alone does not kill the two-period: the two-period has
max-run $1$.  F1 does kill the clustered run-of-$3$
($k_{\lambda}=15<n_0=16$ on $S=81$, $\rho_{\mathrm{AP}}=1$).
A clustered run-of-$2$ has $k_{\lambda}=21>n_0$ --- F1 permits it,
and the r382 entry survives.

\section{The reduction}\label{sec:red}

\begin{prop}[reduction]\label{prop:red}
The candidate ``$\mathrm{assist}_k\le(1-\mathrm{maxdiag})/\mathrm{maxdiag}$
from $3/8$-separation + Gershgorin'' is strictly larger than the
signed remainder:
\begin{enumerate}
\item \emph{Chebyshev--Dirichlet CD, bookkeeping, Gershgorin, mesh.}
  SATZ (Lemmas~\ref{lem:cd}--\ref{lem:mesh}).
\item \emph{Two-period $=$ global AP.}  SATZ
  (Lemma~\ref{lem:ap}), with Dirichlet max-row $\sim 3\times$ the
  source at the r385 witness degree $k=22$.
\item \emph{Unsigned geometry does not close.}
  $k^*_{\mathrm{G}}\sim 0.11\,N<\lfloor 2N/5\rfloor$ on every
  tested window; $d_{\min}/d_{\mathrm{med}}=1/3<3/8$.
\item \emph{Named property} $\rho_{\mathrm{AP}}<1/5$: kills the
  two-period, holds on EXT-heavy seven, holds on scramble
  (which dies elsewhere).
\item \emph{Remainder, strictly smaller:} the signed
  $\mu$-CD off-diagonal --- the true $\mathrm{assist}_k$, whose
  unsigned majorant is closed as too crude.  At the $w=9$ wall
  this is a $0.040$ spectral excess against a $13.3$ Gershgorin
  excess.  An assist theorem that would dock to
  \texttt{ChristoffelPivotBound} (r384) has to bound this
  \emph{signed} object.  $n_0=\lfloor 2N/5\rfloor$ is not
  improved; $n_0=N-1$ stays open with that remainder.
\end{enumerate}
\end{prop}

No Lean file is edited in this round.  The r384 named Prop
\texttt{ChristoffelPivotBound} already accepts a Rayleigh bound
on $E_{k+1}$; a proved assist estimate would be a hypothesis
of that Prop, not a new sorry.
