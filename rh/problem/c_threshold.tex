\begin{abstract}\noindent
Round 407 reduced P1 to $\lambda_2(C)\ge 1$, with $C$ the
$\mu$-only dual Christoffel--Darboux Gram on the hole nodes
$Y$.
The candidate lemma of this round is that $C$ has at most one
eigenvalue below $1$ because half-filling is Nyquist-critical.
That lemma is refuted as a theorem: source hole density is
$0.28$ not $\tfrac12$; the periodic $1010$ tile at density
$\tfrac12$ has a kernel and three or four eigenvalues at $0$;
$\mathrm{diag}(C)$ is far above $1$.
Finite linear-algebra identities remain: $2\times 2$ coherence,
Rayleigh, rank deficiency, Cauchy interlacing.
On the source family, $\#\{\lambda(C)<1\}\in\{0,1\}$ is a
census ($42/42$), not a proof.
No RH claim.
\end{abstract}

\section{The object}\label{sec:obj}

Write $u^\vee$ for the Borodin dual of $|u|$ (r356) and
$n=N_w-3$.
Let $P_0,\dots,P_{n-1}$ be the $u^\vee|_\mu$-orthonormal
orthogonal polynomials and $B_{ki}=\sqrt{u^\vee(y_k)}\,P_i(y_k)$.
Then
\begin{equation}\label{eq:C}
  C \;=\; BB^T,
  \qquad
  C_{ij}
  \;=\;
  \sqrt{u^\vee(y_i)\,u^\vee(y_j)}\;
  K_n^\mu(y_i,y_j),
\end{equation}
the dual CD kernel of the positive $\mu$-chain, evaluated on
$Y$ and dressed by $\sqrt{u^\vee|_Y}$.
Round 407: $R_{N-3}=C(I+C)^{-1}$ and
$\indm A_0=\#\{\lambda(C)<1\}$, so P1 $\iff\lambda_2(C)\ge 1$.

\section{Finite identities}\label{sec:satz}

\begin{lemma}[Two-by-two coherence]\label{lem:coh}
For $C=\bigl(\begin{smallmatrix}a&c\\c&b\end{smallmatrix}\bigr)$
over $\mathbb{Q}$ with $a=2$, $b=3$, $c=2$,
$\det(C-I)=-2<0$.
Hence $C-I$ has eigenvalues of opposite sign, so exactly one
eigenvalue of $C$ lies below $1$.
In general $\det(C-I)=(a-1)(b-1)-c^2$, so the $1$-threshold
on a $2\times 2$ Gram is a coherence bound, not a diagonal
bound.
\end{lemma}

\begin{lemma}[Rayleigh]\label{lem:rayleigh}
$\lambda_{\min}(C)\le\min_i C_{ii}$.
In particular $C_{ii}<1$ for some $i$ forces
$\#\{\lambda(C)<1\}\ge 1$.
\end{lemma}

\begin{lemma}[Rank deficiency]\label{lem:rank}
If $B$ is $|Y|\times n$ then $\mathrm{rank}\,C\le\min(|Y|,n)$.
Over $\mathbb{Q}$, a $3\times 2$ example has $\det C=0$.
Thus $|Y|>n$ produces at least $|Y|-n$ zeros, all strictly
below $1$.
\end{lemma}

\begin{lemma}[Cauchy interlacing]\label{lem:cauchy}
A principal submatrix $C_I$ of a Hermitian $C$ satisfies
$\lambda_{\min}(C_I)\ge\lambda_{\min}(C)$.
Deleting hole coordinates of a \emph{frozen} $C$ cannot
create new eigenvalues below $1$; it can only raise
$\lambda_{\min}$.
\end{lemma}

\section{Why Nyquist-at-$\tfrac12$ fails}\label{sec:no}

The half-filling slogan $S=2N_w-1$ is a statement about the
cosine grid versus CD depth, not about $|Y|/S$.
On FRAME-A, $|Y|/S=104/367=0.283$ and $n=181>|Y|$, so $C$ is
typically full rank.
The periodic $1010$ tile (even/odd split, density $\tfrac12$)
has $n<|Y|$ and therefore a kernel.

\begin{prop}[Periodic $1010$ is not $\lambda_2=1$]\label{prop:1010}
On the two-period mesh of r367, $S=20$ and $c=1$ give
density $\tfrac12$, $\#\{\lambda(C)<1\}=3$ and
$\lambda_{\min}=0$ (at least two exact zeros).
$S=21$, $c=\tfrac23$ gives four eigenvalues at $0$.
This is Lemma~\ref{lem:rank}, not a critical frame with
exactly one soft mode.
\end{prop}

\begin{prop}[FRAME-A anatomy]\label{prop:anatomy}
On window $kz=9$: $C_{\min}=0.85712$, $\lambda_2(C)=1.00018$,
$\#\{\lambda<1\}=1$.
Every diagonal entry exceeds $1.65$ (mean $33.1$).
The exceptional eigenvector is not the ones-mode
($|\langle v,1\rangle|=0.007$); it is hole-Nyquist
($k_{\mathrm{peak}}=|Y|/2=52$) with endpoint mass $0.53$.
\end{prop}

Thinning $Y$ to $30\%$ raises $C_{\min}$ above $1$
(rebuilt, and also frozen-Cauchy).
Flipping ten $\mu$-atoms into holes produces eight
eigenvalues below $1$.
The defect is not caused by too few holes.

\section{Worlds and census}\label{sec:worlds}

\begin{prop}[Named exclusion]\label{prop:worlds}
Scramble: $\#\{\lambda<1\}=21$, $\min\mathrm{diag}(C)=0.37<1$,
fifteen adjacent-hole gaps --- $d_{\min}<1$ plus F1-clusters,
at \emph{lower} density than the source.
Permute: the same $Y$ as the source, but
$\#\{\lambda<1\}=20$ and $d_{\min}=0.15$; the dual weights,
not the hole mask, break $d_{\min}>1$.
Live $\chi_3$ ($kz=9$): $\#\{\lambda<1\}=0$.
Dead $\chi_3$ ($kz=15$): $\#\{\lambda<1\}=1$ --- the
threshold dictionary holds; death remains the edge
signature.
\end{prop}

\begin{prop}[Source census]\label{prop:census}
On the admissible core of $42$ windows,
$\#\{\lambda(C)<1\}\in\{0,1\}$ (fourteen vacuous, twenty-eight
with one), never greater;
$\min\mathrm{diag}(C)\in[1.097,1.656]$.
Every one-defect window has hole-Nyquist peak
$k_{\mathrm{peak}}/(|Y|/2)\ge 0.85$.
Every vacuous window has endpoint mass $1$ on the lowest
mode (DC sitting at or above $1$).
This is a census of the source family, not a sampling
theorem, and not an RH claim.
\end{prop}

\begin{remark}
P1 remains equivalent to $\lambda_2(C)\ge 1$ by r407.
The present round does not prove the right-hand side.
F1-run and gap-histogram statistics do not control
$C_{\min}$ on the one-defect subsample (inherited from
r393/r395: those coordinates do not carry).
Unconditional SATZ is Lemmas~\ref{lem:coh}--\ref{lem:cauchy}.
The census is source-weight-adjacent and is not gated as
PNT or RH-near.
\end{remark}
